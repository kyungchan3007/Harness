import {
  Catalog,
  ErrorCode,
  EventView,
  ID,
  Lab,
  Qty,
  Receipt,
  ReservationView,
  Result,
  SampleView,
  Snapshot,
} from "./types.js";

export function createLab(catalog: Catalog): Lab {
  const locationSet = new Set(catalog.locationIds);
  const custodianSet = new Set(catalog.custodianIds);
  const containerMap = new Map<ID, { locationId: ID; capacity: Qty }>();
  for (const c of catalog.containers) {
    containerMap.set(c.containerId, { locationId: c.locationId, capacity: c.capacity });
  }

  const samples = new Map<ID, SampleView>();
  const reservations = new Map<ID, ReservationView>();
  const containerOccupant = new Map<ID, ID>(); // containerId -> sampleId
  const events: EventView[] = [];
  const eventIdempotency = new Map<ID, { args: unknown; result: Receipt }>();

  let seq = 0;

  function normalizeInput(input: unknown): string {
    if (typeof input === "string" || typeof input === "number" || typeof input === "boolean") {
      return String(input);
    }
    if (input === null) return "null";
    if (Array.isArray(input)) {
      return `[${input.map(normalizeInput).join(",")}]`;
    }
    if (typeof input === "object") {
      const pairs = Object.keys(input as Record<string, unknown>)
        .sort()
        .map((k) => `${k}:${normalizeInput((input as Record<string, unknown>)[k])}`);
      return `{${pairs.join(",")}}`;
    }
    return String(input);
  }

  function fail<T>(code: ErrorCode): Result<T> {
    return { ok: false, code };
  }

  function success(eventId: ID, op: EventView["op"], args: unknown): Result<Receipt> {
    seq++;
    const receipt = { eventId, seq };
    events.push({ seq, eventId, op, args });
    eventIdempotency.set(eventId, { args, result: receipt });
    return { ok: true, value: receipt };
  }

  function checkIdempotent(eventId: ID, args: unknown): Result<Receipt> | null {
    const cached = eventIdempotency.get(eventId);
    if (!cached) return null;
    if (normalizeInput(cached.args) === normalizeInput(args)) {
      return { ok: true, value: cached.result };
    }
    return fail("EVENT_CONFLICT");
  }

  function validateQty(qty: Qty): boolean {
    return Number.isInteger(qty) && qty >= 1 && qty <= 1000000;
  }

  function getAvailable(sample: SampleView): Qty {
    const reserved = Array.from(reservations.values())
      .filter((r) => r.sampleId === sample.sampleId && r.status === "ACTIVE")
      .reduce((sum, r) => sum + r.qty, 0);
    return sample.remaining - reserved;
  }

  const lab: Lab = {
    receive(eventId: ID, input: { sampleId: ID; containerId: ID; custodianId: ID; qty: Qty }): Result<Receipt> {
      const cached = checkIdempotent(eventId, { sampleId: input.sampleId, containerId: input.containerId, custodianId: input.custodianId, qty: input.qty });
      if (cached !== null) return cached;

      if (!validateQty(input.qty)) return fail("INVALID_INPUT");
      if (samples.has(input.sampleId)) return fail("ID_TAKEN");
      if (!containerMap.has(input.containerId)) return fail("UNKNOWN_CONTAINER");
      if (!custodianSet.has(input.custodianId)) return fail("UNKNOWN_CUSTODIAN");
      if (containerOccupant.has(input.containerId)) return fail("CONTAINER_OCCUPIED");
      if (input.qty > containerMap.get(input.containerId)!.capacity) return fail("CAPACITY_EXCEEDED");

      const sample: SampleView = {
        sampleId: input.sampleId,
        parentId: null,
        rootId: input.sampleId,
        remaining: input.qty,
        reserved: 0,
        consumedQty: 0,
        discardedQty: 0,
        status: "ACTIVE",
        containerId: input.containerId,
        custodianId: input.custodianId,
      };
      samples.set(input.sampleId, sample);
      containerOccupant.set(input.containerId, input.sampleId);
      return success(eventId, "receive", { sampleId: input.sampleId, containerId: input.containerId, custodianId: input.custodianId, qty: input.qty });
    },

    split(eventId: ID, parentId: ID, children: Array<{ sampleId: ID; containerId: ID; qty: Qty }>): Result<Receipt> {
      const cached = checkIdempotent(eventId, { parentId, children });
      if (cached !== null) return cached;

      if (children.length === 0) return fail("INVALID_INPUT");

      const parent = samples.get(parentId);
      if (!parent) return fail("UNKNOWN_SAMPLE");
      if (parent.status !== "ACTIVE") return fail("NOT_ACTIVE");

      let totalQty = 0;
      const childrenToCreate: Array<{ sampleId: ID; containerId: ID; qty: Qty }> = [];

      for (const child of children) {
        if (!validateQty(child.qty)) return fail("INVALID_INPUT");
        if (samples.has(child.sampleId)) return fail("ID_TAKEN");
        if (!containerMap.has(child.containerId)) return fail("UNKNOWN_CONTAINER");
        if (containerOccupant.has(child.containerId)) return fail("CONTAINER_OCCUPIED");
        if (child.containerId === parent.containerId) return fail("CONTAINER_OCCUPIED");
        const container = containerMap.get(child.containerId)!;
        if (child.qty > container.capacity) return fail("CAPACITY_EXCEEDED");
        totalQty += child.qty;
        childrenToCreate.push(child);
      }

      const available = getAvailable(parent);
      if (totalQty > available) return fail("INSUFFICIENT");

      // All validations passed, create children
      for (const child of childrenToCreate) {
        const newSample: SampleView = {
          sampleId: child.sampleId,
          parentId: parent.sampleId,
          rootId: parent.rootId,
          remaining: child.qty,
          reserved: 0,
          consumedQty: 0,
          discardedQty: 0,
          status: "ACTIVE",
          containerId: child.containerId,
          custodianId: parent.custodianId,
        };
        samples.set(child.sampleId, newSample);
        containerOccupant.set(child.containerId, child.sampleId);
      }

      // Update parent
      parent.remaining -= totalQty;
      if (parent.remaining === 0) {
        parent.status = "DEPLETED";
        containerOccupant.delete(parent.containerId!);
        parent.containerId = null;
      }

      return success(eventId, "split", { parentId, children });
    },

    reserve(eventId: ID, input: { reservationId: ID; sampleId: ID; qty: Qty }): Result<Receipt> {
      const cached = checkIdempotent(eventId, { reservationId: input.reservationId, sampleId: input.sampleId, qty: input.qty });
      if (cached !== null) return cached;

      if (!validateQty(input.qty)) return fail("INVALID_INPUT");
      if (reservations.has(input.reservationId)) return fail("ID_TAKEN");

      const sample = samples.get(input.sampleId);
      if (!sample) return fail("UNKNOWN_SAMPLE");
      if (sample.status !== "ACTIVE") return fail("NOT_ACTIVE");

      const available = getAvailable(sample);
      if (input.qty > available) return fail("INSUFFICIENT");

      const reservation: ReservationView = {
        reservationId: input.reservationId,
        sampleId: input.sampleId,
        qty: input.qty,
        status: "ACTIVE",
      };
      reservations.set(input.reservationId, reservation);
      sample.reserved += input.qty;

      return success(eventId, "reserve", { reservationId: input.reservationId, sampleId: input.sampleId, qty: input.qty });
    },

    cancel(eventId: ID, reservationId: ID): Result<Receipt> {
      const cached = checkIdempotent(eventId, { reservationId });
      if (cached !== null) return cached;

      const reservation = reservations.get(reservationId);
      if (!reservation) return fail("UNKNOWN_RESERVATION");
      if (reservation.status !== "ACTIVE") return fail("RESERVATION_CLOSED");

      const sample = samples.get(reservation.sampleId);
      if (!sample) return fail("UNKNOWN_SAMPLE");

      reservation.status = "CANCELED";
      sample.reserved -= reservation.qty;

      return success(eventId, "cancel", { reservationId });
    },

    consume(eventId: ID, reservationId: ID): Result<Receipt> {
      const cached = checkIdempotent(eventId, { reservationId });
      if (cached !== null) return cached;

      const reservation = reservations.get(reservationId);
      if (!reservation) return fail("UNKNOWN_RESERVATION");
      if (reservation.status !== "ACTIVE") return fail("RESERVATION_CLOSED");

      const sample = samples.get(reservation.sampleId);
      if (!sample) return fail("UNKNOWN_SAMPLE");

      sample.remaining -= reservation.qty;
      sample.reserved -= reservation.qty;
      sample.consumedQty += reservation.qty;
      reservation.status = "CONSUMED";

      if (sample.remaining === 0) {
        sample.status = "DEPLETED";
        if (sample.containerId) {
          containerOccupant.delete(sample.containerId);
        }
        sample.containerId = null;
      }

      return success(eventId, "consume", { reservationId });
    },

    discard(eventId: ID, sampleId: ID): Result<Receipt> {
      const cached = checkIdempotent(eventId, { sampleId });
      if (cached !== null) return cached;

      const sample = samples.get(sampleId);
      if (!sample) return fail("UNKNOWN_SAMPLE");
      if (sample.status !== "ACTIVE") return fail("NOT_ACTIVE");

      const hasActiveReservation = Array.from(reservations.values()).some(
        (r) => r.sampleId === sampleId && r.status === "ACTIVE"
      );
      if (hasActiveReservation) return fail("HAS_RESERVATION");

      sample.discardedQty += sample.remaining;
      sample.remaining = 0;
      sample.status = "DISCARDED";
      if (sample.containerId) {
        containerOccupant.delete(sample.containerId);
        sample.containerId = null;
      }

      return success(eventId, "discard", { sampleId });
    },

    move(eventId: ID, sampleId: ID, targetContainerId: ID): Result<Receipt> {
      const cached = checkIdempotent(eventId, { sampleId, targetContainerId });
      if (cached !== null) return cached;

      const sample = samples.get(sampleId);
      if (!sample) return fail("UNKNOWN_SAMPLE");
      if (sample.status !== "ACTIVE") return fail("NOT_ACTIVE");

      if (!containerMap.has(targetContainerId)) return fail("UNKNOWN_CONTAINER");
      const targetContainer = containerMap.get(targetContainerId)!;

      if (sample.containerId === targetContainerId) return fail("CONTAINER_OCCUPIED");
      if (containerOccupant.has(targetContainerId)) return fail("CONTAINER_OCCUPIED");
      if (sample.remaining > targetContainer.capacity) return fail("CAPACITY_EXCEEDED");

      const oldContainerId = sample.containerId;
      if (oldContainerId) {
        containerOccupant.delete(oldContainerId);
      }
      sample.containerId = targetContainerId;
      containerOccupant.set(targetContainerId, sampleId);

      return success(eventId, "move", { sampleId, targetContainerId });
    },

    transferCustody(eventId: ID, sampleId: ID, custodianId: ID): Result<Receipt> {
      const cached = checkIdempotent(eventId, { sampleId, custodianId });
      if (cached !== null) return cached;

      const sample = samples.get(sampleId);
      if (!sample) return fail("UNKNOWN_SAMPLE");
      if (sample.status !== "ACTIVE") return fail("NOT_ACTIVE");
      if (!custodianSet.has(custodianId)) return fail("UNKNOWN_CUSTODIAN");

      sample.custodianId = custodianId;
      return success(eventId, "transferCustody", { sampleId, custodianId });
    },

    snapshot(): Snapshot {
      return {
        samples: Array.from(samples.values()),
        reservations: Array.from(reservations.values()),
        events,
      };
    },
  };

  return lab;
}
