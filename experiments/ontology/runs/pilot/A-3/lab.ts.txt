import {
  Catalog,
  ErrorCode,
  EventView,
  ID,
  Lab,
  Qty,
  Receipt,
  ReservationStatus,
  ReservationView,
  Result,
  SampleStatus,
  SampleView,
  Snapshot,
} from "./types.js";

interface SampleRecord {
  sampleId: ID;
  parentId: ID | null;
  rootId: ID;
  remaining: Qty;
  reserved: Qty;
  consumedQty: Qty;
  discardedQty: Qty;
  status: SampleStatus;
  containerId: ID | null;
  custodianId: ID;
}

interface ReservationRecord {
  reservationId: ID;
  sampleId: ID;
  qty: Qty;
  status: ReservationStatus;
}

interface ContainerRecord {
  containerId: ID;
  locationId: ID;
  capacity: Qty;
  occupantSampleId: ID | null;
}

interface EventLogEntry {
  op: EventView["op"];
  args: unknown;
  seq: number;
}

export function createLab(catalog: Catalog): Lab {
  const samples = new Map<ID, SampleRecord>();
  const reservations = new Map<ID, ReservationRecord>();
  const containers = new Map<ID, ContainerRecord>();
  const custodianIds = new Set<ID>(catalog.custodianIds);
  const containerIds = new Set<ID>(catalog.containers.map((c) => c.containerId));
  const eventLog = new Map<ID, EventLogEntry>();

  let seq = 0;

  for (const c of catalog.containers) {
    const container: ContainerRecord = {
      containerId: c.containerId,
      locationId: c.locationId,
      capacity: c.capacity,
      occupantSampleId: null,
    };
    containers.set(c.containerId, container);
  }

  function deepEqual(a: unknown, b: unknown): boolean {
    if (typeof a !== typeof b) return false;
    if (a === null || b === null) return a === b;

    if (Array.isArray(a) && Array.isArray(b)) {
      if (a.length !== b.length) return false;
      return a.every((v, i) => deepEqual(v, b[i]));
    }

    if (typeof a === "object" && typeof b === "object") {
      const keysA = Object.keys(a).sort();
      const keysB = Object.keys(b).sort();
      if (keysA.length !== keysB.length) return false;
      if (!keysA.every((k, i) => k === keysB[i])) return false;
      return keysA.every((k) => deepEqual((a as Record<string, unknown>)[k], (b as Record<string, unknown>)[k]));
    }

    return a === b;
  }

  function normalizeArgs(args: unknown): unknown {
    if (Array.isArray(args)) {
      return args.map(normalizeArgs);
    }
    if (typeof args === "object" && args !== null) {
      const obj = args as Record<string, unknown>;
      const keys = Object.keys(obj).sort();
      const normalized: Record<string, unknown> = {};
      for (const k of keys) {
        normalized[k] = normalizeArgs(obj[k]);
      }
      return normalized;
    }
    return args;
  }

  function checkEventLog(
    eventId: ID,
    op: EventView["op"],
    args: unknown
  ): Result<Receipt> | null {
    const entry = eventLog.get(eventId);
    if (!entry) return null;

    if (entry.op !== op || !deepEqual(normalizeArgs(entry.args), normalizeArgs(args))) {
      return { ok: false, code: "EVENT_CONFLICT" };
    }

    return {
      ok: true,
      value: { eventId, seq: entry.seq },
    };
  }

  function recordEvent(eventId: ID, op: EventView["op"], args: unknown): void {
    seq += 1;
    eventLog.set(eventId, {
      op,
      args,
      seq,
    });
  }

  function fail<T>(code: ErrorCode): Result<T> {
    return { ok: false, code };
  }

  function success<T>(value: T): Result<T> {
    return { ok: true, value };
  }

  function getAvailableQty(sampleId: ID): Qty {
    const sample = samples.get(sampleId);
    if (!sample) return 0;
    return Math.max(0, sample.remaining - sample.reserved);
  }

  const receive = (eventId: ID, input: { sampleId: ID; containerId: ID; custodianId: ID; qty: Qty }): Result<Receipt> => {
    const cached = checkEventLog(eventId, "receive", input);
    if (cached !== null) return cached;

    if (
      !Number.isInteger(input.qty) ||
      input.qty < 1 ||
      input.qty > 1000000
    ) {
      return fail("INVALID_INPUT");
    }

    if (samples.has(input.sampleId)) {
      return fail("ID_TAKEN");
    }

    if (!containerIds.has(input.containerId)) {
      return fail("UNKNOWN_CONTAINER");
    }

    if (!custodianIds.has(input.custodianId)) {
      return fail("UNKNOWN_CUSTODIAN");
    }

    const container = containers.get(input.containerId)!;
    if (container.occupantSampleId !== null) {
      return fail("CONTAINER_OCCUPIED");
    }

    if (input.qty > container.capacity) {
      return fail("CAPACITY_EXCEEDED");
    }

    const sample: SampleRecord = {
      sampleId: input.sampleId,
      parentId: null,
      rootId: input.sampleId,
      remaining: input.qty,
      reserved: 0,
      consumedQty: 0,
      discardedQty: 0,
      status: "ACTIVE",
      containerId: input.containerId,
      custodianId: input.custodianId,
    };

    samples.set(input.sampleId, sample);
    container.occupantSampleId = input.sampleId;

    recordEvent(eventId, "receive", input);
    return success({ eventId, seq });
  };

  const split = (
    eventId: ID,
    parentId: ID,
    children: Array<{ sampleId: ID; containerId: ID; qty: Qty }>
  ): Result<Receipt> => {
    const cached = checkEventLog(eventId, "split", { parentId, children });
    if (cached !== null) return cached;

    if (children.length === 0) {
      return fail("INVALID_INPUT");
    }

    for (const child of children) {
      if (!Number.isInteger(child.qty) || child.qty < 1 || child.qty > 1000000) {
        return fail("INVALID_INPUT");
      }
    }

    const parent = samples.get(parentId);
    if (!parent) {
      return fail("UNKNOWN_SAMPLE");
    }

    if (parent.status !== "ACTIVE") {
      return fail("NOT_ACTIVE");
    }

    const totalChildQty = children.reduce((sum, c) => sum + c.qty, 0);
    const availableQty = getAvailableQty(parentId);

    if (totalChildQty > availableQty) {
      return fail("INSUFFICIENT");
    }

    const usedContainers = new Set<ID>([parent.containerId!]);
    for (const child of children) {
      if (!containerIds.has(child.containerId)) {
        return fail("UNKNOWN_CONTAINER");
      }

      if (usedContainers.has(child.containerId)) {
        return fail("CONTAINER_OCCUPIED");
      }

      const container = containers.get(child.containerId)!;
      if (container.occupantSampleId !== null) {
        return fail("CONTAINER_OCCUPIED");
      }

      if (child.qty > container.capacity) {
        return fail("CAPACITY_EXCEEDED");
      }

      if (samples.has(child.sampleId)) {
        return fail("ID_TAKEN");
      }

      usedContainers.add(child.containerId);
    }

    for (const child of children) {
      const childSample: SampleRecord = {
        sampleId: child.sampleId,
        parentId,
        rootId: parent.rootId,
        remaining: child.qty,
        reserved: 0,
        consumedQty: 0,
        discardedQty: 0,
        status: "ACTIVE",
        containerId: child.containerId,
        custodianId: parent.custodianId,
      };
      samples.set(child.sampleId, childSample);
      const container = containers.get(child.containerId)!;
      container.occupantSampleId = child.sampleId;
    }

    parent.remaining -= totalChildQty;
    if (parent.remaining === 0) {
      parent.status = "DEPLETED";
      const oldContainerId = parent.containerId;
      parent.containerId = null;
      if (oldContainerId) {
        const oldContainer = containers.get(oldContainerId)!;
        oldContainer.occupantSampleId = null;
      }
    }

    recordEvent(eventId, "split", { parentId, children });
    return success({ eventId, seq });
  };

  const reserve = (eventId: ID, input: { reservationId: ID; sampleId: ID; qty: Qty }): Result<Receipt> => {
    const cached = checkEventLog(eventId, "reserve", input);
    if (cached !== null) return cached;

    if (!Number.isInteger(input.qty) || input.qty < 1 || input.qty > 1000000) {
      return fail("INVALID_INPUT");
    }

    if (reservations.has(input.reservationId)) {
      return fail("ID_TAKEN");
    }

    const sample = samples.get(input.sampleId);
    if (!sample) {
      return fail("UNKNOWN_SAMPLE");
    }

    if (sample.status !== "ACTIVE") {
      return fail("NOT_ACTIVE");
    }

    const availableQty = getAvailableQty(input.sampleId);
    if (input.qty > availableQty) {
      return fail("INSUFFICIENT");
    }

    const reservation: ReservationRecord = {
      reservationId: input.reservationId,
      sampleId: input.sampleId,
      qty: input.qty,
      status: "ACTIVE",
    };

    reservations.set(input.reservationId, reservation);
    sample.reserved += input.qty;

    recordEvent(eventId, "reserve", input);
    return success({ eventId, seq });
  };

  const cancel = (eventId: ID, reservationId: ID): Result<Receipt> => {
    const cached = checkEventLog(eventId, "cancel", { reservationId });
    if (cached !== null) return cached;

    const reservation = reservations.get(reservationId);
    if (!reservation) {
      return fail("UNKNOWN_RESERVATION");
    }

    if (reservation.status !== "ACTIVE") {
      return fail("RESERVATION_CLOSED");
    }

    const sample = samples.get(reservation.sampleId);
    if (!sample) {
      return fail("UNKNOWN_SAMPLE");
    }

    reservation.status = "CANCELED";
    sample.reserved -= reservation.qty;

    recordEvent(eventId, "cancel", { reservationId });
    return success({ eventId, seq });
  };

  const consume = (eventId: ID, reservationId: ID): Result<Receipt> => {
    const cached = checkEventLog(eventId, "consume", { reservationId });
    if (cached !== null) return cached;

    const reservation = reservations.get(reservationId);
    if (!reservation) {
      return fail("UNKNOWN_RESERVATION");
    }

    if (reservation.status !== "ACTIVE") {
      return fail("RESERVATION_CLOSED");
    }

    const sample = samples.get(reservation.sampleId);
    if (!sample) {
      return fail("UNKNOWN_SAMPLE");
    }

    reservation.status = "CONSUMED";
    sample.remaining -= reservation.qty;
    sample.reserved -= reservation.qty;
    sample.consumedQty += reservation.qty;

    if (sample.remaining === 0) {
      sample.status = "DEPLETED";
      const oldContainerId = sample.containerId;
      sample.containerId = null;
      if (oldContainerId) {
        const container = containers.get(oldContainerId)!;
        container.occupantSampleId = null;
      }
    }

    recordEvent(eventId, "consume", { reservationId });
    return success({ eventId, seq });
  };

  const discard = (eventId: ID, sampleId: ID): Result<Receipt> => {
    const cached = checkEventLog(eventId, "discard", { sampleId });
    if (cached !== null) return cached;

    const sample = samples.get(sampleId);
    if (!sample) {
      return fail("UNKNOWN_SAMPLE");
    }

    if (sample.status !== "ACTIVE") {
      return fail("NOT_ACTIVE");
    }

    if (sample.reserved > 0) {
      return fail("HAS_RESERVATION");
    }

    sample.discardedQty += sample.remaining;
    sample.remaining = 0;
    sample.status = "DISCARDED";
    const oldContainerId = sample.containerId;
    sample.containerId = null;
    if (oldContainerId) {
      const container = containers.get(oldContainerId)!;
      container.occupantSampleId = null;
    }

    recordEvent(eventId, "discard", { sampleId });
    return success({ eventId, seq });
  };

  const move = (eventId: ID, sampleId: ID, targetContainerId: ID): Result<Receipt> => {
    const cached = checkEventLog(eventId, "move", { sampleId, targetContainerId });
    if (cached !== null) return cached;

    const sample = samples.get(sampleId);
    if (!sample) {
      return fail("UNKNOWN_SAMPLE");
    }

    if (sample.status !== "ACTIVE") {
      return fail("NOT_ACTIVE");
    }

    if (!containerIds.has(targetContainerId)) {
      return fail("UNKNOWN_CONTAINER");
    }

    const targetContainer = containers.get(targetContainerId)!;
    if (targetContainer.occupantSampleId !== null) {
      return fail("CONTAINER_OCCUPIED");
    }

    if (sample.remaining > targetContainer.capacity) {
      return fail("CAPACITY_EXCEEDED");
    }

    if (sample.containerId === targetContainerId) {
      return fail("CONTAINER_OCCUPIED");
    }

    const oldContainerId = sample.containerId;
    if (oldContainerId) {
      const oldContainer = containers.get(oldContainerId)!;
      oldContainer.occupantSampleId = null;
    }

    sample.containerId = targetContainerId;
    targetContainer.occupantSampleId = sampleId;

    recordEvent(eventId, "move", { sampleId, targetContainerId });
    return success({ eventId, seq });
  };

  const transferCustody = (eventId: ID, sampleId: ID, custodianId: ID): Result<Receipt> => {
    const cached = checkEventLog(eventId, "transferCustody", { sampleId, custodianId });
    if (cached !== null) return cached;

    const sample = samples.get(sampleId);
    if (!sample) {
      return fail("UNKNOWN_SAMPLE");
    }

    if (sample.status !== "ACTIVE") {
      return fail("NOT_ACTIVE");
    }

    if (!custodianIds.has(custodianId)) {
      return fail("UNKNOWN_CUSTODIAN");
    }

    sample.custodianId = custodianId;

    recordEvent(eventId, "transferCustody", { sampleId, custodianId });
    return success({ eventId, seq });
  };

  const snapshot = (): Snapshot => {
    const sampleArray: SampleView[] = Array.from(samples.values());
    const reservationArray: ReservationView[] = Array.from(reservations.values());
    const eventArray: EventView[] = Array.from(eventLog.entries()).map(([eventId, entry]) => ({
      seq: entry.seq,
      eventId,
      op: entry.op,
      args: entry.args,
    }));

    return {
      samples: sampleArray,
      reservations: reservationArray,
      events: eventArray.sort((a, b) => a.seq - b.seq),
    };
  };

  return {
    receive,
    split,
    reserve,
    cancel,
    consume,
    discard,
    move,
    transferCustody,
    snapshot,
  };
}
