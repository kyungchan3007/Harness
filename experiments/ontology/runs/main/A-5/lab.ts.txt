import { Lab, Result, Receipt, Catalog, SampleView, ReservationView, EventView, Snapshot, ID, Qty, SampleStatus, ReservationStatus } from './types.js';

export function createLab(catalog: Catalog): Lab {
  const locationSet = new Set(catalog.locationIds);
  const custodianSet = new Set(catalog.custodianIds);
  const containerMap = new Map<ID, { locationId: ID; capacity: Qty }>();

  for (const c of catalog.containers) {
    containerMap.set(c.containerId, { locationId: c.locationId, capacity: c.capacity });
  }

  interface Sample {
    sampleId: ID;
    parentId: ID | null;
    rootId: ID;
    remaining: Qty;
    reserved: Qty;
    consumedQty: Qty;
    discardedQty: Qty;
    status: SampleStatus;
    containerId: ID | null;
    custodianId: ID;
  }

  interface Reservation {
    reservationId: ID;
    sampleId: ID;
    qty: Qty;
    status: ReservationStatus;
  }

  interface EventRecord {
    seq: number;
    eventId: ID;
    op: string;
    args: unknown;
  }

  const samples = new Map<ID, Sample>();
  const reservations = new Map<ID, Reservation>();
  const events: EventRecord[] = [];
  const eventIdMap = new Map<ID, number>();
  let seq = 0;

  function isValidQty(qty: any): qty is Qty {
    return typeof qty === 'number' && Number.isInteger(qty) && qty >= 1 && qty <= 1_000_000;
  }

  function sortKeys(obj: any): any {
    if (obj === null || typeof obj !== 'object') return obj;
    if (Array.isArray(obj)) return obj.map(sortKeys);
    const sorted: any = {};
    Object.keys(obj).sort().forEach(k => {
      sorted[k] = sortKeys(obj[k]);
    });
    return sorted;
  }

  function argsEqual(a: unknown, b: unknown): boolean {
    const aStr = JSON.stringify(sortKeys(a));
    const bStr = JSON.stringify(sortKeys(b));
    return aStr === bStr;
  }

  function recalculateReserved(sampleId: ID): Qty {
    let sum = 0;
    for (const res of reservations.values()) {
      if (res.sampleId === sampleId && res.status === 'ACTIVE') {
        sum += res.qty;
      }
    }
    return sum;
  }

  function getAvailable(sample: Sample): Qty {
    return Math.max(0, sample.remaining - sample.reserved);
  }

  function checkDuplicate(eventId: ID, op: string, args: unknown): Result<Receipt> | null {
    if (eventIdMap.has(eventId)) {
      const index = eventIdMap.get(eventId)!;
      const event = events[index - 1]!;
      if (event.op === op && argsEqual(event.args, args)) {
        return { ok: true, value: { eventId, seq: event.seq } };
      } else {
        return { ok: false, code: 'EVENT_CONFLICT' };
      }
    }
    return null;
  }

  function recordEvent(eventId: ID, op: string, args: unknown): void {
    seq++;
    eventIdMap.set(eventId, seq);
    events.push({ seq, eventId, op, args });
  }

  function isContainerOccupied(containerId: ID): boolean {
    for (const s of samples.values()) {
      if (s.containerId === containerId && s.status === 'ACTIVE') {
        return true;
      }
    }
    return false;
  }

  const lab: Lab = {
    receive(eventId: ID, input: { sampleId: ID; containerId: ID; custodianId: ID; qty: Qty }): Result<Receipt> {
      const dup = checkDuplicate(eventId, 'receive', input);
      if (dup) return dup;

      if (!isValidQty(input.qty)) return { ok: false, code: 'INVALID_INPUT' };
      if (samples.has(input.sampleId)) return { ok: false, code: 'ID_TAKEN' };
      if (!containerMap.has(input.containerId)) return { ok: false, code: 'UNKNOWN_CONTAINER' };
      if (!custodianSet.has(input.custodianId)) return { ok: false, code: 'UNKNOWN_CUSTODIAN' };

      if (isContainerOccupied(input.containerId)) return { ok: false, code: 'CONTAINER_OCCUPIED' };

      const container = containerMap.get(input.containerId)!;
      if (input.qty > container.capacity) {
        return { ok: false, code: 'CAPACITY_EXCEEDED' };
      }

      const sample: Sample = {
        sampleId: input.sampleId,
        parentId: null,
        rootId: input.sampleId,
        remaining: input.qty,
        reserved: 0,
        consumedQty: 0,
        discardedQty: 0,
        status: 'ACTIVE',
        containerId: input.containerId,
        custodianId: input.custodianId,
      };

      samples.set(input.sampleId, sample);
      recordEvent(eventId, 'receive', input);

      return { ok: true, value: { eventId, seq } };
    },

    split(eventId: ID, parentId: ID, children: Array<{ sampleId: ID; containerId: ID; qty: Qty }>): Result<Receipt> {
      const args = { parentId, children };
      const dup = checkDuplicate(eventId, 'split', args);
      if (dup) return dup;

      if (!samples.has(parentId)) return { ok: false, code: 'UNKNOWN_SAMPLE' };
      if (children.length === 0) return { ok: false, code: 'INVALID_INPUT' };

      const parent = samples.get(parentId)!;
      if (parent.status !== 'ACTIVE') return { ok: false, code: 'NOT_ACTIVE' };

      const childIds = new Set<ID>();
      let totalChildQty = 0;
      const usedContainers = new Set<ID>();
      if (parent.containerId) usedContainers.add(parent.containerId);

      for (const child of children) {
        if (!isValidQty(child.qty)) return { ok: false, code: 'INVALID_INPUT' };
        if (childIds.has(child.sampleId)) return { ok: false, code: 'ID_TAKEN' };
        if (samples.has(child.sampleId)) return { ok: false, code: 'ID_TAKEN' };
        if (!containerMap.has(child.containerId)) return { ok: false, code: 'UNKNOWN_CONTAINER' };

        if (usedContainers.has(child.containerId)) return { ok: false, code: 'CONTAINER_OCCUPIED' };
        if (isContainerOccupied(child.containerId)) return { ok: false, code: 'CONTAINER_OCCUPIED' };

        const container = containerMap.get(child.containerId)!;
        if (child.qty > container.capacity) {
          return { ok: false, code: 'CAPACITY_EXCEEDED' };
        }

        childIds.add(child.sampleId);
        usedContainers.add(child.containerId);
        totalChildQty += child.qty;
      }

      parent.reserved = recalculateReserved(parentId);
      const available = getAvailable(parent);
      if (totalChildQty > available) {
        return { ok: false, code: 'INSUFFICIENT' };
      }

      for (const child of children) {
        const newSample: Sample = {
          sampleId: child.sampleId,
          parentId: parentId,
          rootId: parent.rootId,
          remaining: child.qty,
          reserved: 0,
          consumedQty: 0,
          discardedQty: 0,
          status: 'ACTIVE',
          containerId: child.containerId,
          custodianId: parent.custodianId,
        };
        samples.set(child.sampleId, newSample);
      }

      parent.remaining -= totalChildQty;
      if (parent.remaining === 0) {
        parent.status = 'DEPLETED';
        parent.containerId = null;
      }

      recordEvent(eventId, 'split', args);

      return { ok: true, value: { eventId, seq } };
    },

    reserve(eventId: ID, input: { reservationId: ID; sampleId: ID; qty: Qty }): Result<Receipt> {
      const dup = checkDuplicate(eventId, 'reserve', input);
      if (dup) return dup;

      if (!isValidQty(input.qty)) return { ok: false, code: 'INVALID_INPUT' };
      if (!samples.has(input.sampleId)) return { ok: false, code: 'UNKNOWN_SAMPLE' };
      if (reservations.has(input.reservationId)) return { ok: false, code: 'ID_TAKEN' };

      const sample = samples.get(input.sampleId)!;
      if (sample.status !== 'ACTIVE') return { ok: false, code: 'NOT_ACTIVE' };

      sample.reserved = recalculateReserved(input.sampleId);
      const available = getAvailable(sample);
      if (input.qty > available) {
        return { ok: false, code: 'INSUFFICIENT' };
      }

      const reservation: Reservation = {
        reservationId: input.reservationId,
        sampleId: input.sampleId,
        qty: input.qty,
        status: 'ACTIVE',
      };
      reservations.set(input.reservationId, reservation);
      sample.reserved += input.qty;

      recordEvent(eventId, 'reserve', input);

      return { ok: true, value: { eventId, seq } };
    },

    cancel(eventId: ID, reservationId: ID): Result<Receipt> {
      const args = { reservationId };
      const dup = checkDuplicate(eventId, 'cancel', args);
      if (dup) return dup;

      if (!reservations.has(reservationId)) return { ok: false, code: 'UNKNOWN_RESERVATION' };

      const res = reservations.get(reservationId)!;
      if (res.status !== 'ACTIVE') return { ok: false, code: 'RESERVATION_CLOSED' };

      const sample = samples.get(res.sampleId)!;
      res.status = 'CANCELED';
      sample.reserved -= res.qty;

      recordEvent(eventId, 'cancel', args);

      return { ok: true, value: { eventId, seq } };
    },

    consume(eventId: ID, reservationId: ID): Result<Receipt> {
      const args = { reservationId };
      const dup = checkDuplicate(eventId, 'consume', args);
      if (dup) return dup;

      if (!reservations.has(reservationId)) return { ok: false, code: 'UNKNOWN_RESERVATION' };

      const res = reservations.get(reservationId)!;
      if (res.status !== 'ACTIVE') return { ok: false, code: 'RESERVATION_CLOSED' };

      const sample = samples.get(res.sampleId)!;
      res.status = 'CONSUMED';
      sample.remaining -= res.qty;
      sample.reserved -= res.qty;
      sample.consumedQty += res.qty;

      if (sample.remaining === 0) {
        sample.status = 'DEPLETED';
        sample.containerId = null;
      }

      recordEvent(eventId, 'consume', args);

      return { ok: true, value: { eventId, seq } };
    },

    discard(eventId: ID, sampleId: ID): Result<Receipt> {
      const args = { sampleId };
      const dup = checkDuplicate(eventId, 'discard', args);
      if (dup) return dup;

      if (!samples.has(sampleId)) return { ok: false, code: 'UNKNOWN_SAMPLE' };

      const sample = samples.get(sampleId)!;
      if (sample.status !== 'ACTIVE') return { ok: false, code: 'NOT_ACTIVE' };

      for (const res of reservations.values()) {
        if (res.sampleId === sampleId && res.status === 'ACTIVE') {
          return { ok: false, code: 'HAS_RESERVATION' };
        }
      }

      sample.discardedQty += sample.remaining;
      sample.remaining = 0;
      sample.status = 'DISCARDED';
      sample.containerId = null;

      recordEvent(eventId, 'discard', args);

      return { ok: true, value: { eventId, seq } };
    },

    move(eventId: ID, sampleId: ID, targetContainerId: ID): Result<Receipt> {
      const args = { sampleId, targetContainerId };
      const dup = checkDuplicate(eventId, 'move', args);
      if (dup) return dup;

      if (!samples.has(sampleId)) return { ok: false, code: 'UNKNOWN_SAMPLE' };
      if (!containerMap.has(targetContainerId)) return { ok: false, code: 'UNKNOWN_CONTAINER' };

      const sample = samples.get(sampleId)!;
      if (sample.status !== 'ACTIVE') return { ok: false, code: 'NOT_ACTIVE' };

      if (sample.containerId === targetContainerId) {
        return { ok: false, code: 'CONTAINER_OCCUPIED' };
      }

      if (isContainerOccupied(targetContainerId)) return { ok: false, code: 'CONTAINER_OCCUPIED' };

      const container = containerMap.get(targetContainerId)!;
      if (sample.remaining > container.capacity) {
        return { ok: false, code: 'CAPACITY_EXCEEDED' };
      }

      sample.containerId = targetContainerId;

      recordEvent(eventId, 'move', args);

      return { ok: true, value: { eventId, seq } };
    },

    transferCustody(eventId: ID, sampleId: ID, custodianId: ID): Result<Receipt> {
      const args = { sampleId, custodianId };
      const dup = checkDuplicate(eventId, 'transferCustody', args);
      if (dup) return dup;

      if (!samples.has(sampleId)) return { ok: false, code: 'UNKNOWN_SAMPLE' };
      if (!custodianSet.has(custodianId)) return { ok: false, code: 'UNKNOWN_CUSTODIAN' };

      const sample = samples.get(sampleId)!;
      if (sample.status !== 'ACTIVE') return { ok: false, code: 'NOT_ACTIVE' };

      sample.custodianId = custodianId;

      recordEvent(eventId, 'transferCustody', args);

      return { ok: true, value: { eventId, seq } };
    },

    snapshot(): Snapshot {
      const samplesArray: SampleView[] = Array.from(samples.values()).map(s => ({
        sampleId: s.sampleId,
        parentId: s.parentId,
        rootId: s.rootId,
        remaining: s.remaining,
        reserved: s.reserved,
        consumedQty: s.consumedQty,
        discardedQty: s.discardedQty,
        status: s.status,
        containerId: s.containerId,
        custodianId: s.custodianId,
      }));

      const reservationsArray: ReservationView[] = Array.from(reservations.values()).map(r => ({
        reservationId: r.reservationId,
        sampleId: r.sampleId,
        qty: r.qty,
        status: r.status,
      }));

      const eventsArray: EventView[] = events.map(e => ({
        seq: e.seq,
        eventId: e.eventId,
        op: e.op as any,
        args: e.args,
      }));

      return {
        samples: samplesArray,
        reservations: reservationsArray,
        events: eventsArray,
      };
    },
  };

  return lab;
}
