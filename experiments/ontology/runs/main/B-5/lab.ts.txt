import {
  ID,
  Qty,
  Result,
  Receipt,
  SampleStatus,
  ReservationStatus,
  SampleView,
  ReservationView,
  EventView,
  Snapshot,
  Catalog,
  Lab,
} from "./types.js";

interface Sample {
  sampleId: ID;
  parentId: ID | null;
  rootId: ID;
  remaining: Qty;
  consumedQty: Qty;
  discardedQty: Qty;
  status: SampleStatus;
  containerId: ID | null;
  custodianId: ID;
}

interface Container {
  containerId: ID;
  locationId: ID;
  capacity: Qty;
  sampleId: ID | null;
}

interface Reservation {
  reservationId: ID;
  sampleId: ID;
  qty: Qty;
  status: ReservationStatus;
}

interface EventRecord {
  eventId: ID;
  op: string;
  args: unknown;
  seq: number;
}

export function createLab(catalog: Catalog): Lab {
  const samples = new Map<ID, Sample>();
  const containers = new Map<ID, Container>();
  const reservations = new Map<ID, Reservation>();
  const eventHistory = new Map<ID, EventRecord>();
  const events: EventView[] = [];

  const locationSet = new Set(catalog.locationIds);
  const custodianSet = new Set(catalog.custodianIds);
  const containerMap = new Map<ID, { locationId: ID; capacity: Qty }>();
  const sampleIdTaken = new Set<ID>();
  const reservationIdTaken = new Set<ID>();

  for (const c of catalog.containers) {
    containerMap.set(c.containerId, { locationId: c.locationId, capacity: c.capacity });
    containers.set(c.containerId, {
      containerId: c.containerId,
      locationId: c.locationId,
      capacity: c.capacity,
      sampleId: null,
    });
  }

  let seq = 0;

  function deepCopy<T>(obj: T): T {
    return JSON.parse(JSON.stringify(obj));
  }

  function getReserved(sampleId: ID): Qty {
    let sum = 0;
    for (const res of reservations.values()) {
      if (res.sampleId === sampleId && res.status === "ACTIVE") {
        sum += res.qty;
      }
    }
    return sum;
  }

  function updateSampleStatus(sample: Sample): void {
    if (sample.status === "ACTIVE" && sample.remaining === 0) {
      sample.status = "DEPLETED";
      sample.containerId = null;
    }
  }

  function argsEqual(a: unknown, b: unknown): boolean {
    return JSON.stringify(a) === JSON.stringify(b);
  }

  function recordEvent(
    eventId: ID,
    op: string,
    args: unknown,
    newSeq: number
  ): void {
    eventHistory.set(eventId, { eventId, op, args: deepCopy(args), seq: newSeq });
    events.push({
      seq: newSeq,
      eventId,
      op: op as any,
      args: deepCopy(args),
    });
  }

  const lab: Lab = {
    receive(eventId: ID, input: { sampleId: ID; containerId: ID; custodianId: ID; qty: Qty }): Result<Receipt> {
      const existing = eventHistory.get(eventId);
      if (existing && existing.op === "receive" && argsEqual(existing.args, input)) {
        return { ok: true, value: { eventId, seq: existing.seq } };
      }
      if (existing) {
        return { ok: false, code: "EVENT_CONFLICT" };
      }

      if (!input.sampleId || !input.containerId || !input.custodianId || !Number.isInteger(input.qty) || input.qty < 1 || input.qty > 1_000_000) {
        return { ok: false, code: "INVALID_INPUT" };
      }

      if (sampleIdTaken.has(input.sampleId)) {
        return { ok: false, code: "ID_TAKEN" };
      }

      if (!containerMap.has(input.containerId)) {
        return { ok: false, code: "UNKNOWN_CONTAINER" };
      }

      if (!custodianSet.has(input.custodianId)) {
        return { ok: false, code: "UNKNOWN_CUSTODIAN" };
      }

      const container = containers.get(input.containerId)!;
      if (container.sampleId !== null) {
        return { ok: false, code: "CONTAINER_OCCUPIED" };
      }

      if (container.capacity < input.qty) {
        return { ok: false, code: "CAPACITY_EXCEEDED" };
      }

      seq++;
      const sample: Sample = {
        sampleId: input.sampleId,
        parentId: null,
        rootId: input.sampleId,
        remaining: input.qty,
        consumedQty: 0,
        discardedQty: 0,
        status: "ACTIVE",
        containerId: input.containerId,
        custodianId: input.custodianId,
      };

      samples.set(input.sampleId, sample);
      sampleIdTaken.add(input.sampleId);
      container.sampleId = input.sampleId;

      recordEvent(eventId, "receive", input, seq);
      return { ok: true, value: { eventId, seq } };
    },

    split(
      eventId: ID,
      parentId: ID,
      children: Array<{ sampleId: ID; containerId: ID; qty: Qty }>
    ): Result<Receipt> {
      const existing = eventHistory.get(eventId);
      if (existing && existing.op === "split" && argsEqual(existing.args, { parentId, children })) {
        return { ok: true, value: { eventId, seq: existing.seq } };
      }
      if (existing) {
        return { ok: false, code: "EVENT_CONFLICT" };
      }

      if (!parentId || !children || children.length === 0) {
        return { ok: false, code: "INVALID_INPUT" };
      }

      const parent = samples.get(parentId);
      if (!parent) {
        return { ok: false, code: "UNKNOWN_SAMPLE" };
      }

      if (parent.status !== "ACTIVE") {
        return { ok: false, code: "NOT_ACTIVE" };
      }

      let totalChildQty = 0;
      const childSampleIds = new Set<ID>();
      const childContainerIds = new Set<ID>();

      for (const child of children) {
        if (!child.sampleId || !child.containerId || !Number.isInteger(child.qty) || child.qty < 1 || child.qty > 1_000_000) {
          return { ok: false, code: "INVALID_INPUT" };
        }
        totalChildQty += child.qty;

        if (childSampleIds.has(child.sampleId)) {
          return { ok: false, code: "ID_TAKEN" };
        }
        if (childContainerIds.has(child.containerId)) {
          return { ok: false, code: "CONTAINER_OCCUPIED" };
        }
        if (sampleIdTaken.has(child.sampleId)) {
          return { ok: false, code: "ID_TAKEN" };
        }
        if (!containerMap.has(child.containerId)) {
          return { ok: false, code: "UNKNOWN_CONTAINER" };
        }

        childSampleIds.add(child.sampleId);
        childContainerIds.add(child.containerId);

        const container = containers.get(child.containerId)!;
        if (container.sampleId !== null) {
          return { ok: false, code: "CONTAINER_OCCUPIED" };
        }

        if (child.containerId === parent.containerId) {
          return { ok: false, code: "CONTAINER_OCCUPIED" };
        }

        if (container.capacity < child.qty) {
          return { ok: false, code: "CAPACITY_EXCEEDED" };
        }
      }

      const reserved = getReserved(parentId);
      const available = parent.remaining - reserved;
      if (available < totalChildQty) {
        return { ok: false, code: "INSUFFICIENT" };
      }

      seq++;
      for (const child of children) {
        const childSample: Sample = {
          sampleId: child.sampleId,
          parentId: parentId,
          rootId: parent.rootId,
          remaining: child.qty,
          consumedQty: 0,
          discardedQty: 0,
          status: "ACTIVE",
          containerId: child.containerId,
          custodianId: parent.custodianId,
        };

        samples.set(child.sampleId, childSample);
        sampleIdTaken.add(child.sampleId);
        containers.get(child.containerId)!.sampleId = child.sampleId;
      }

      parent.remaining -= totalChildQty;
      updateSampleStatus(parent);

      recordEvent(eventId, "split", { parentId, children: deepCopy(children) }, seq);
      return { ok: true, value: { eventId, seq } };
    },

    reserve(eventId: ID, input: { reservationId: ID; sampleId: ID; qty: Qty }): Result<Receipt> {
      const existing = eventHistory.get(eventId);
      if (existing && existing.op === "reserve" && argsEqual(existing.args, input)) {
        return { ok: true, value: { eventId, seq: existing.seq } };
      }
      if (existing) {
        return { ok: false, code: "EVENT_CONFLICT" };
      }

      if (!input.reservationId || !input.sampleId || !Number.isInteger(input.qty) || input.qty < 1 || input.qty > 1_000_000) {
        return { ok: false, code: "INVALID_INPUT" };
      }

      if (reservationIdTaken.has(input.reservationId)) {
        return { ok: false, code: "ID_TAKEN" };
      }

      const sample = samples.get(input.sampleId);
      if (!sample) {
        return { ok: false, code: "UNKNOWN_SAMPLE" };
      }

      if (sample.status !== "ACTIVE") {
        return { ok: false, code: "NOT_ACTIVE" };
      }

      const reserved = getReserved(input.sampleId);
      const available = sample.remaining - reserved;
      if (available < input.qty) {
        return { ok: false, code: "INSUFFICIENT" };
      }

      seq++;
      const reservation: Reservation = {
        reservationId: input.reservationId,
        sampleId: input.sampleId,
        qty: input.qty,
        status: "ACTIVE",
      };

      reservations.set(input.reservationId, reservation);
      reservationIdTaken.add(input.reservationId);

      recordEvent(eventId, "reserve", input, seq);
      return { ok: true, value: { eventId, seq } };
    },

    cancel(eventId: ID, reservationId: ID): Result<Receipt> {
      const existing = eventHistory.get(eventId);
      if (existing && existing.op === "cancel" && argsEqual(existing.args, { reservationId })) {
        return { ok: true, value: { eventId, seq: existing.seq } };
      }
      if (existing) {
        return { ok: false, code: "EVENT_CONFLICT" };
      }

      if (!reservationId) {
        return { ok: false, code: "INVALID_INPUT" };
      }

      const reservation = reservations.get(reservationId);
      if (!reservation) {
        return { ok: false, code: "UNKNOWN_RESERVATION" };
      }

      if (reservation.status !== "ACTIVE") {
        return { ok: false, code: "RESERVATION_CLOSED" };
      }

      seq++;
      reservation.status = "CANCELED";

      recordEvent(eventId, "cancel", { reservationId }, seq);
      return { ok: true, value: { eventId, seq } };
    },

    consume(eventId: ID, reservationId: ID): Result<Receipt> {
      const existing = eventHistory.get(eventId);
      if (existing && existing.op === "consume" && argsEqual(existing.args, { reservationId })) {
        return { ok: true, value: { eventId, seq: existing.seq } };
      }
      if (existing) {
        return { ok: false, code: "EVENT_CONFLICT" };
      }

      if (!reservationId) {
        return { ok: false, code: "INVALID_INPUT" };
      }

      const reservation = reservations.get(reservationId);
      if (!reservation) {
        return { ok: false, code: "UNKNOWN_RESERVATION" };
      }

      if (reservation.status !== "ACTIVE") {
        return { ok: false, code: "RESERVATION_CLOSED" };
      }

      const sample = samples.get(reservation.sampleId);
      if (!sample) {
        return { ok: false, code: "UNKNOWN_SAMPLE" };
      }

      seq++;
      reservation.status = "CONSUMED";
      sample.remaining -= reservation.qty;
      sample.consumedQty += reservation.qty;
      updateSampleStatus(sample);

      recordEvent(eventId, "consume", { reservationId }, seq);
      return { ok: true, value: { eventId, seq } };
    },

    discard(eventId: ID, sampleId: ID): Result<Receipt> {
      const existing = eventHistory.get(eventId);
      if (existing && existing.op === "discard" && argsEqual(existing.args, { sampleId })) {
        return { ok: true, value: { eventId, seq: existing.seq } };
      }
      if (existing) {
        return { ok: false, code: "EVENT_CONFLICT" };
      }

      if (!sampleId) {
        return { ok: false, code: "INVALID_INPUT" };
      }

      const sample = samples.get(sampleId);
      if (!sample) {
        return { ok: false, code: "UNKNOWN_SAMPLE" };
      }

      if (sample.status !== "ACTIVE") {
        return { ok: false, code: "NOT_ACTIVE" };
      }

      for (const res of reservations.values()) {
        if (res.sampleId === sampleId && res.status === "ACTIVE") {
          return { ok: false, code: "HAS_RESERVATION" };
        }
      }

      seq++;
      sample.discardedQty += sample.remaining;
      sample.remaining = 0;
      sample.status = "DISCARDED";

      const containerId = sample.containerId;
      sample.containerId = null;

      if (containerId) {
        containers.get(containerId)!.sampleId = null;
      }

      recordEvent(eventId, "discard", { sampleId }, seq);
      return { ok: true, value: { eventId, seq } };
    },

    move(eventId: ID, sampleId: ID, targetContainerId: ID): Result<Receipt> {
      const existing = eventHistory.get(eventId);
      if (existing && existing.op === "move" && argsEqual(existing.args, { sampleId, targetContainerId })) {
        return { ok: true, value: { eventId, seq: existing.seq } };
      }
      if (existing) {
        return { ok: false, code: "EVENT_CONFLICT" };
      }

      if (!sampleId || !targetContainerId) {
        return { ok: false, code: "INVALID_INPUT" };
      }

      const sample = samples.get(sampleId);
      if (!sample) {
        return { ok: false, code: "UNKNOWN_SAMPLE" };
      }

      if (sample.status !== "ACTIVE") {
        return { ok: false, code: "NOT_ACTIVE" };
      }

      if (!containerMap.has(targetContainerId)) {
        return { ok: false, code: "UNKNOWN_CONTAINER" };
      }

      const targetContainer = containers.get(targetContainerId)!;

      if (targetContainer.sampleId !== null) {
        return { ok: false, code: "CONTAINER_OCCUPIED" };
      }

      if (targetContainer.capacity < sample.remaining) {
        return { ok: false, code: "CAPACITY_EXCEEDED" };
      }

      if (targetContainerId === sample.containerId) {
        return { ok: false, code: "CONTAINER_OCCUPIED" };
      }

      seq++;
      const oldContainerId = sample.containerId;
      if (oldContainerId) {
        containers.get(oldContainerId)!.sampleId = null;
      }

      sample.containerId = targetContainerId;
      targetContainer.sampleId = sampleId;

      recordEvent(eventId, "move", { sampleId, targetContainerId }, seq);
      return { ok: true, value: { eventId, seq } };
    },

    transferCustody(eventId: ID, sampleId: ID, custodianId: ID): Result<Receipt> {
      const existing = eventHistory.get(eventId);
      if (existing && existing.op === "transferCustody" && argsEqual(existing.args, { sampleId, custodianId })) {
        return { ok: true, value: { eventId, seq: existing.seq } };
      }
      if (existing) {
        return { ok: false, code: "EVENT_CONFLICT" };
      }

      if (!sampleId || !custodianId) {
        return { ok: false, code: "INVALID_INPUT" };
      }

      const sample = samples.get(sampleId);
      if (!sample) {
        return { ok: false, code: "UNKNOWN_SAMPLE" };
      }

      if (sample.status !== "ACTIVE") {
        return { ok: false, code: "NOT_ACTIVE" };
      }

      if (!custodianSet.has(custodianId)) {
        return { ok: false, code: "UNKNOWN_CUSTODIAN" };
      }

      seq++;
      sample.custodianId = custodianId;

      recordEvent(eventId, "transferCustody", { sampleId, custodianId }, seq);
      return { ok: true, value: { eventId, seq } };
    },

    snapshot(): Snapshot {
      const sampleViews: SampleView[] = Array.from(samples.values()).map((s) => ({
        sampleId: s.sampleId,
        parentId: s.parentId,
        rootId: s.rootId,
        remaining: s.remaining,
        reserved: getReserved(s.sampleId),
        consumedQty: s.consumedQty,
        discardedQty: s.discardedQty,
        status: s.status,
        containerId: s.containerId,
        custodianId: s.custodianId,
      }));

      const reservationViews: ReservationView[] = Array.from(reservations.values()).map((r) => ({
        reservationId: r.reservationId,
        sampleId: r.sampleId,
        qty: r.qty,
        status: r.status,
      }));

      return {
        samples: sampleViews,
        reservations: reservationViews,
        events: [...events],
      };
    },
  };

  return lab;
}
