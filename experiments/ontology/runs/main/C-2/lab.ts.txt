import {
  ID,
  Qty,
  ErrorCode,
  Result,
  Receipt,
  SampleStatus,
  ReservationStatus,
  SampleView,
  ReservationView,
  EventView,
  Snapshot,
  Catalog,
  Lab,
} from "./types.js";

interface SampleState {
  sampleId: ID;
  parentId: ID | null;
  rootId: ID;
  remaining: Qty;
  reserved: Qty;
  consumedQty: Qty;
  discardedQty: Qty;
  status: SampleStatus;
  containerId: ID | null;
  custodianId: ID;
}

interface ReservationState {
  reservationId: ID;
  sampleId: ID;
  qty: Qty;
  status: ReservationStatus;
}

class LabImpl implements Lab {
  private catalog: Catalog;
  private samples: Map<ID, SampleState>;
  private reservations: Map<ID, ReservationState>;
  private containerUsage: Map<ID, ID | null>;
  private successfulEvents: Map<ID, { op: string; args: unknown; receipt: Receipt }>;
  private events: EventView[];
  private nextSeq: number;

  constructor(catalog: Catalog) {
    this.catalog = catalog;
    this.samples = new Map();
    this.reservations = new Map();
    this.containerUsage = new Map();
    this.successfulEvents = new Map();
    this.events = [];
    this.nextSeq = 1;

    // 모든 용기를 처음 상태(사용 안 함)로 초기화
    for (const container of catalog.containers) {
      this.containerUsage.set(container.containerId, null as null);
    }
  }

  // ============== 헬퍼 함수들 ==============

  private deepCopy(obj: unknown): unknown {
    if (obj === null || typeof obj !== "object") return obj;
    if (Array.isArray(obj)) return obj.map((x) => this.deepCopy(x));
    const copy: any = {};
    for (const [k, v] of Object.entries(obj)) {
      copy[k] = this.deepCopy(v);
    }
    return copy;
  }

  private argsEqual(a: unknown, b: unknown): boolean {
    if (typeof a !== typeof b) return false;
    if (Array.isArray(a) && Array.isArray(b)) {
      return (
        a.length === b.length && a.every((x, i) => this.argsEqual(x, b[i]))
      );
    }
    if (typeof a === "object" && a !== null && typeof b === "object" && b !== null) {
      const aKeys = Object.keys(a).sort();
      const bKeys = Object.keys(b).sort();
      return (
        aKeys.length === bKeys.length &&
        aKeys.every((k) => this.argsEqual((a as any)[k], (b as any)[k]))
      );
    }
    return a === b;
  }

  private checkIdempotency(
    eventId: ID,
    op: string,
    args: unknown
  ): Result<Receipt | null> {
    if (this.successfulEvents.has(eventId)) {
      const record = this.successfulEvents.get(eventId)!;
      if (record.op !== op || !this.argsEqual(record.args, args)) {
        return { ok: false, code: "EVENT_CONFLICT" };
      }
      return { ok: true, value: record.receipt };
    }
    return { ok: true, value: null };
  }

  private executeCommand<T>(
    eventId: ID,
    op: string,
    args: unknown,
    validate: () => Result<T>,
    apply: (payload: T) => void
  ): Result<Receipt> {
    // 멱등성 확인 (R7.1) — 다른 모든 검사보다 먼저
    const idem = this.checkIdempotency(eventId, op, args);
    if (!idem.ok) return idem;
    if (idem.value !== null) {
      return { ok: true, value: idem.value };
    }

    // 검증 (부작용 없음)
    const result = validate();
    if (!result.ok) return result;

    // 상태 변경 (원자성 보장)
    apply(result.value);

    // 이력 기록 (R8.1, R8.4)
    const seq = this.nextSeq++;
    this.events.push({
      seq,
      eventId,
      op,
      args: this.deepCopy(args),
    });
    this.successfulEvents.set(eventId, {
      op,
      args: this.deepCopy(args),
      receipt: { eventId, seq },
    });

    return { ok: true, value: { eventId, seq } };
  }

  private validateQty(qty: Qty): Result<null> {
    if (!Number.isInteger(qty) || qty < 1 || qty > 1_000_000) {
      return { ok: false, code: "INVALID_INPUT" };
    }
    return { ok: true, value: null };
  }

  private getContainer(containerId: ID) {
    return this.catalog.containers.find((c) => c.containerId === containerId);
  }

  private isCustodianValid(custodianId: ID): boolean {
    return this.catalog.custodianIds.includes(custodianId);
  }

  // ============== 명령 메서드들 ==============

  receive(
    eventId: ID,
    input: { sampleId: ID; containerId: ID; custodianId: ID; qty: Qty }
  ): Result<Receipt> {
    return this.executeCommand(
      eventId,
      "receive",
      input,
      () => {
        // 검증: R1, R2, R5
        const { sampleId, containerId, custodianId, qty } = input;

        // R1.2: qty 유효성
        const qtyCheck = this.validateQty(qty);
        if (!qtyCheck.ok) return qtyCheck;

        // R1.1: sampleId는 신규여야 함
        if (this.samples.has(sampleId)) {
          return { ok: false, code: "ID_TAKEN" };
        }

        // R1.1: containerId 유효성
        const container = this.getContainer(containerId);
        if (!container) {
          return { ok: false, code: "UNKNOWN_CONTAINER" };
        }

        // R1.1: custodianId 유효성
        if (!this.isCustodianValid(custodianId)) {
          return { ok: false, code: "UNKNOWN_CUSTODIAN" };
        }

        // R1.3: 용기 점유 확인
        if (this.containerUsage.get(containerId) !== null) {
          return { ok: false, code: "CONTAINER_OCCUPIED" };
        }

        // R1.4: 용기 용량
        if (container.capacity < qty) {
          return { ok: false, code: "CAPACITY_EXCEEDED" };
        }

        return { ok: true, value: null };
      },
      () => {
        const { sampleId, containerId, custodianId, qty } = input;
        this.samples.set(sampleId, {
          sampleId,
          parentId: null,
          rootId: sampleId,
          remaining: qty,
          reserved: 0,
          consumedQty: 0,
          discardedQty: 0,
          status: "ACTIVE",
          containerId,
          custodianId,
        });
        this.containerUsage.set(containerId, sampleId);
      }
    );
  }

  split(
    eventId: ID,
    parentId: ID,
    children: Array<{ sampleId: ID; containerId: ID; qty: Qty }>
  ): Result<Receipt> {
    return this.executeCommand(
      eventId,
      "split",
      { parentId, children },
      () => {
        // 검증: R1, R2, R5, R8.3
        // R1.2: children 배열 비어 있음
        if (children.length === 0) {
          return { ok: false, code: "INVALID_INPUT" };
        }

        // R1.2: 각 child의 qty 유효성
        for (const child of children) {
          const qtyCheck = this.validateQty(child.qty);
          if (!qtyCheck.ok) return qtyCheck;
        }

        // R1.1: parentId 유효성
        const parent = this.samples.get(parentId);
        if (!parent) {
          return { ok: false, code: "UNKNOWN_SAMPLE" };
        }

        // R4.3: 부모가 ACTIVE 상태인지 확인
        if (parent.status !== "ACTIVE") {
          return { ok: false, code: "NOT_ACTIVE" };
        }

        // R1.1: 각 child의 sampleId는 신규여야 함
        for (const child of children) {
          if (this.samples.has(child.sampleId)) {
            return { ok: false, code: "ID_TAKEN" };
          }
        }

        // R1.1: 각 child의 containerId 유효성
        for (const child of children) {
          if (!this.getContainer(child.containerId)) {
            return { ok: false, code: "UNKNOWN_CONTAINER" };
          }
        }

        // R5.3: child의 containerId 중복 확인 및 부모 용기 포함 확인
        const childContainers = new Set<ID>();
        for (const child of children) {
          if (
            childContainers.has(child.containerId) ||
            child.containerId === parent.containerId
          ) {
            return { ok: false, code: "CONTAINER_OCCUPIED" };
          }
          childContainers.add(child.containerId);
        }

        // R5.3: 각 child의 containerId가 비어 있는지 확인
        for (const child of children) {
          if (this.containerUsage.get(child.containerId) !== null) {
            return { ok: false, code: "CONTAINER_OCCUPIED" };
          }
        }

        // R2.2: 분취 양이 부모의 이용 가능량을 초과하지 않는지
        const parentAvailable = parent.remaining - parent.reserved;
        const totalChildQty = children.reduce((sum, c) => sum + c.qty, 0);
        if (totalChildQty > parentAvailable) {
          return { ok: false, code: "INSUFFICIENT" };
        }

        // R1.4: 각 child의 용기 용량 확인
        for (const child of children) {
          const container = this.getContainer(child.containerId)!;
          if (container.capacity < child.qty) {
            return { ok: false, code: "CAPACITY_EXCEEDED" };
          }
        }

        return { ok: true, value: null };
      },
      () => {
        // R8.3: split 원자성 - 모두 또는 무
        const parent = this.samples.get(parentId)!;
        for (const child of children) {
          this.samples.set(child.sampleId, {
            sampleId: child.sampleId,
            parentId,
            rootId: parent.rootId,
            remaining: child.qty,
            reserved: 0,
            consumedQty: 0,
            discardedQty: 0,
            status: "ACTIVE",
            containerId: child.containerId,
            custodianId: parent.custodianId,
          });
          this.containerUsage.set(child.containerId, child.sampleId);
        }

        // 부모의 상태 업데이트 (R2.2)
        const totalChildQty = children.reduce((sum, c) => sum + c.qty, 0);
        parent.remaining -= totalChildQty;
        if (parent.remaining === 0) {
          parent.status = "DEPLETED";
          parent.containerId = null;
        }
      }
    );
  }

  reserve(
    eventId: ID,
    input: { reservationId: ID; sampleId: ID; qty: Qty }
  ): Result<Receipt> {
    return this.executeCommand(
      eventId,
      "reserve",
      input,
      () => {
        const { reservationId, sampleId, qty } = input;

        // R1.2: qty 유효성
        const qtyCheck = this.validateQty(qty);
        if (!qtyCheck.ok) return qtyCheck;

        // R1.1: reservationId는 신규여야 함
        if (this.reservations.has(reservationId)) {
          return { ok: false, code: "ID_TAKEN" };
        }

        // R1.1: sampleId 유효성
        const sample = this.samples.get(sampleId);
        if (!sample) {
          return { ok: false, code: "UNKNOWN_SAMPLE" };
        }

        // R3.2: NOT_ACTIVE 시료에 reserve 불가
        if (sample.status !== "ACTIVE") {
          return { ok: false, code: "NOT_ACTIVE" };
        }

        // R3.2: available >= qty (이용 가능량)
        const available = sample.remaining - sample.reserved;
        if (available < qty) {
          return { ok: false, code: "INSUFFICIENT" };
        }

        return { ok: true, value: null };
      },
      () => {
        const { reservationId, sampleId, qty } = input;
        const sample = this.samples.get(sampleId)!;
        this.reservations.set(reservationId, {
          reservationId,
          sampleId,
          qty,
          status: "ACTIVE",
        });
        sample.reserved += qty;
      }
    );
  }

  cancel(eventId: ID, reservationId: ID): Result<Receipt> {
    return this.executeCommand(
      eventId,
      "cancel",
      { reservationId },
      () => {
        // R1.1: reservationId 유효성
        const reservation = this.reservations.get(reservationId);
        if (!reservation) {
          return { ok: false, code: "UNKNOWN_RESERVATION" };
        }

        // R3.5: 이미 CANCELED/CONSUMED인 예약에 cancel 불가
        if (reservation.status !== "ACTIVE") {
          return { ok: false, code: "RESERVATION_CLOSED" };
        }

        return { ok: true, value: null };
      },
      () => {
        const reservation = this.reservations.get(reservationId)!;
        const sample = this.samples.get(reservation.sampleId)!;
        reservation.status = "CANCELED";
        sample.reserved -= reservation.qty;
      }
    );
  }

  consume(eventId: ID, reservationId: ID): Result<Receipt> {
    return this.executeCommand(
      eventId,
      "consume",
      { reservationId },
      () => {
        // R1.1: reservationId 유효성
        const reservation = this.reservations.get(reservationId);
        if (!reservation) {
          return { ok: false, code: "UNKNOWN_RESERVATION" };
        }

        // R3.5: 이미 CANCELED/CONSUMED인 예약에 consume 불가
        if (reservation.status !== "ACTIVE") {
          return { ok: false, code: "RESERVATION_CLOSED" };
        }

        return { ok: true, value: null };
      },
      () => {
        const reservation = this.reservations.get(reservationId)!;
        const sample = this.samples.get(reservation.sampleId)!;
        reservation.status = "CONSUMED";
        sample.remaining -= reservation.qty;
        sample.reserved -= reservation.qty;
        sample.consumedQty += reservation.qty;

        // R4.1: remaining이 0이면 DEPLETED
        if (sample.remaining === 0) {
          sample.status = "DEPLETED";
          sample.containerId = null;
        }
      }
    );
  }

  discard(eventId: ID, sampleId: ID): Result<Receipt> {
    return this.executeCommand(
      eventId,
      "discard",
      { sampleId },
      () => {
        // R1.1: sampleId 유효성
        const sample = this.samples.get(sampleId);
        if (!sample) {
          return { ok: false, code: "UNKNOWN_SAMPLE" };
        }

        // R4.3: NOT_ACTIVE 시료에 discard 불가
        if (sample.status !== "ACTIVE") {
          return { ok: false, code: "NOT_ACTIVE" };
        }

        // R4.2: ACTIVE 예약이 있으면 discard 불가
        const hasActiveReservation = Array.from(this.reservations.values()).some(
          (r) => r.sampleId === sampleId && r.status === "ACTIVE"
        );
        if (hasActiveReservation) {
          return { ok: false, code: "HAS_RESERVATION" };
        }

        return { ok: true, value: null };
      },
      () => {
        const sample = this.samples.get(sampleId)!;
        sample.discardedQty += sample.remaining;
        sample.remaining = 0;
        sample.status = "DISCARDED";
        sample.containerId = null;
      }
    );
  }

  move(
    eventId: ID,
    sampleId: ID,
    targetContainerId: ID
  ): Result<Receipt> {
    return this.executeCommand(
      eventId,
      "move",
      { sampleId, targetContainerId },
      () => {
        // R1.1: sampleId 유효성
        const sample = this.samples.get(sampleId);
        if (!sample) {
          return { ok: false, code: "UNKNOWN_SAMPLE" };
        }

        // R1.1: targetContainerId 유효성
        const targetContainer = this.getContainer(targetContainerId);
        if (!targetContainer) {
          return { ok: false, code: "UNKNOWN_CONTAINER" };
        }

        // R4.3: NOT_ACTIVE 시료에 move 불가
        if (sample.status !== "ACTIVE") {
          return { ok: false, code: "NOT_ACTIVE" };
        }

        // R5.2: targetContainerId가 현재 용기와 같으면 안 됨
        if (sample.containerId === targetContainerId) {
          return { ok: false, code: "CONTAINER_OCCUPIED" };
        }

        // R5.2: targetContainerId가 비어 있어야 함
        if (this.containerUsage.get(targetContainerId) !== null) {
          return { ok: false, code: "CONTAINER_OCCUPIED" };
        }

        // R5.2: targetContainer 용량이 충분해야 함
        if (targetContainer.capacity < sample.remaining) {
          return { ok: false, code: "CAPACITY_EXCEEDED" };
        }

        return { ok: true, value: null };
      },
      () => {
        const sample = this.samples.get(sampleId)!;
        const oldContainerId = sample.containerId!;
        this.containerUsage.set(oldContainerId, null);
        this.containerUsage.set(targetContainerId, sample.sampleId);
        sample.containerId = targetContainerId;
      }
    );
  }

  transferCustody(
    eventId: ID,
    sampleId: ID,
    custodianId: ID
  ): Result<Receipt> {
    return this.executeCommand(
      eventId,
      "transferCustody",
      { sampleId, custodianId },
      () => {
        // R1.1: sampleId 유효성
        const sample = this.samples.get(sampleId);
        if (!sample) {
          return { ok: false, code: "UNKNOWN_SAMPLE" };
        }

        // R1.1: custodianId 유효성
        if (!this.isCustodianValid(custodianId)) {
          return { ok: false, code: "UNKNOWN_CUSTODIAN" };
        }

        // R4.3: NOT_ACTIVE 시료에 transferCustody 불가
        if (sample.status !== "ACTIVE") {
          return { ok: false, code: "NOT_ACTIVE" };
        }

        return { ok: true, value: null };
      },
      () => {
        const sample = this.samples.get(sampleId)!;
        sample.custodianId = custodianId;
      }
    );
  }

  snapshot(): Snapshot {
    const samples: SampleView[] = Array.from(this.samples.values());
    const reservations: ReservationView[] = Array.from(
      this.reservations.values()
    );
    return {
      samples,
      reservations,
      events: this.events,
    };
  }
}

export function createLab(catalog: Catalog): Lab {
  return new LabImpl(catalog);
}
