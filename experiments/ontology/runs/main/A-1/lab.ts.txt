import {
  ID,
  Qty,
  ErrorCode,
  Result,
  Receipt,
  SampleStatus,
  ReservationStatus,
  SampleView,
  ReservationView,
  EventView,
  Snapshot,
  Catalog,
  Lab,
} from "./types.js";

type SampleState = SampleView;

type ReservationState = ReservationView;

interface SuccessfulEvent {
  op: EventView["op"];
  args: unknown;
  seq: number;
}

export function createLab(catalog: Catalog): Lab {
  const samples = new Map<ID, SampleState>();
  const reservations = new Map<ID, ReservationState>();
  const events = new Map<ID, SuccessfulEvent>();
  const eventLog: EventView[] = [];

  let seq = 0;

  const locations = new Set(catalog.locationIds);
  const custodians = new Set(catalog.custodianIds);
  const containerMap = new Map<ID, { containerId: ID; locationId: ID; capacity: Qty }>(
    catalog.containers.map((c) => [c.containerId, c])
  );
  const containerToSample = new Map<ID, ID>();

  function normalizeArgs(args: unknown): string {
    return JSON.stringify(args);
  }

  function checkEventIdConflict(
    eventId: ID,
    currentOp: EventView["op"],
    currentArgs: unknown
  ): Result<Receipt> | null {
    if (events.has(eventId)) {
      const prev = events.get(eventId)!;
      if (
        prev.op === currentOp &&
        normalizeArgs(prev.args) === normalizeArgs(currentArgs)
      ) {
        return {
          ok: true,
          value: { eventId, seq: prev.seq },
        };
      }
      return { ok: false, code: "EVENT_CONFLICT" };
    }
    return null;
  }

  function fail<T>(code: ErrorCode): Result<T> {
    return { ok: false, code };
  }

  function success(eventId: ID, op: EventView["op"], args: unknown): Receipt {
    seq++;
    events.set(eventId, { op, args, seq });
    eventLog.push({ seq, eventId, op, args });
    return { eventId, seq };
  }

  const lab: Lab = {
    receive(eventId: ID, input: { sampleId: ID; containerId: ID; custodianId: ID; qty: Qty }) {
      const existingCheck = checkEventIdConflict(eventId, "receive", input);
      if (existingCheck !== null) return existingCheck;

      const { sampleId, containerId, custodianId, qty } = input;

      if (
        !Number.isInteger(qty) ||
        qty < 1 ||
        qty > 1000000
      ) {
        return fail("INVALID_INPUT");
      }

      if (samples.has(sampleId)) {
        return fail("ID_TAKEN");
      }

      if (!containerMap.has(containerId)) {
        return fail("UNKNOWN_CONTAINER");
      }

      if (!custodians.has(custodianId)) {
        return fail("UNKNOWN_CUSTODIAN");
      }

      const container = containerMap.get(containerId)!;
      if (containerToSample.has(containerId)) {
        return fail("CONTAINER_OCCUPIED");
      }

      if (qty > container.capacity) {
        return fail("CAPACITY_EXCEEDED");
      }

      samples.set(sampleId, {
        sampleId,
        parentId: null,
        rootId: sampleId,
        remaining: qty,
        reserved: 0,
        consumedQty: 0,
        discardedQty: 0,
        status: "ACTIVE",
        containerId,
        custodianId,
      });

      containerToSample.set(containerId, sampleId);

      return { ok: true, value: success(eventId, "receive", input) };
    },

    split(eventId: ID, parentId: ID, children: Array<{ sampleId: ID; containerId: ID; qty: Qty }>) {
      const splitArgs = { parentId, children };
      const existingCheck = checkEventIdConflict(eventId, "split", splitArgs);
      if (existingCheck !== null) return existingCheck;

      if (!Array.isArray(children) || children.length === 0) {
        return fail("INVALID_INPUT");
      }

      for (const child of children) {
        const { qty } = child;
        if (!Number.isInteger(qty) || qty < 1 || qty > 1000000) {
          return fail("INVALID_INPUT");
        }
      }

      if (!samples.has(parentId)) {
        return fail("UNKNOWN_SAMPLE");
      }

      const parent = samples.get(parentId)!;
      if (parent.status !== "ACTIVE") {
        return fail("NOT_ACTIVE");
      }

      const totalChildQty = children.reduce((sum, c) => sum + c.qty, 0);
      const availableQty = parent.remaining - parent.reserved;
      if (totalChildQty > availableQty) {
        return fail("INSUFFICIENT");
      }

      const usedContainers = new Set<ID>();
      usedContainers.add(parent.containerId!);

      for (const child of children) {
        const { sampleId, containerId, qty } = child;

        if (samples.has(sampleId)) {
          return fail("ID_TAKEN");
        }

        if (!containerMap.has(containerId)) {
          return fail("UNKNOWN_CONTAINER");
        }

        if (containerToSample.has(containerId)) {
          return fail("CONTAINER_OCCUPIED");
        }

        if (usedContainers.has(containerId)) {
          return fail("CONTAINER_OCCUPIED");
        }

        const container = containerMap.get(containerId)!;
        if (qty > container.capacity) {
          return fail("CAPACITY_EXCEEDED");
        }

        usedContainers.add(containerId);
      }

      for (const child of children) {
        const { sampleId, containerId, qty } = child;
        samples.set(sampleId, {
          sampleId,
          parentId,
          rootId: parent.rootId,
          remaining: qty,
          reserved: 0,
          consumedQty: 0,
          discardedQty: 0,
          status: "ACTIVE",
          containerId,
          custodianId: parent.custodianId,
        });
        containerToSample.set(containerId, sampleId);
      }

      parent.remaining -= totalChildQty;
      if (parent.remaining === 0) {
        parent.status = "DEPLETED";
        containerToSample.delete(parent.containerId!);
        parent.containerId = null;
      }

      return {
        ok: true,
        value: success(eventId, "split", splitArgs),
      };
    },

    reserve(eventId: ID, input: { reservationId: ID; sampleId: ID; qty: Qty }) {
      const existingCheck = checkEventIdConflict(eventId, "reserve", input);
      if (existingCheck !== null) return existingCheck;

      const { reservationId, sampleId, qty } = input;

      if (!Number.isInteger(qty) || qty < 1 || qty > 1000000) {
        return fail("INVALID_INPUT");
      }

      if (reservations.has(reservationId)) {
        return fail("ID_TAKEN");
      }

      if (!samples.has(sampleId)) {
        return fail("UNKNOWN_SAMPLE");
      }

      const sample = samples.get(sampleId)!;
      if (sample.status !== "ACTIVE") {
        return fail("NOT_ACTIVE");
      }

      const availableQty = sample.remaining - sample.reserved;
      if (qty > availableQty) {
        return fail("INSUFFICIENT");
      }

      reservations.set(reservationId, {
        reservationId,
        sampleId,
        qty,
        status: "ACTIVE",
      });

      sample.reserved += qty;

      return { ok: true, value: success(eventId, "reserve", input) };
    },

    cancel(eventId: ID, reservationId: ID) {
      const cancelArgs = { reservationId };
      const existingCheck = checkEventIdConflict(eventId, "cancel", cancelArgs);
      if (existingCheck !== null) return existingCheck;

      if (!reservations.has(reservationId)) {
        return fail("UNKNOWN_RESERVATION");
      }

      const reservation = reservations.get(reservationId)!;
      if (reservation.status !== "ACTIVE") {
        return fail("RESERVATION_CLOSED");
      }

      const sample = samples.get(reservation.sampleId)!;
      sample.reserved -= reservation.qty;

      reservation.status = "CANCELED";

      return {
        ok: true,
        value: success(eventId, "cancel", cancelArgs),
      };
    },

    consume(eventId: ID, reservationId: ID) {
      const consumeArgs = { reservationId };
      const existingCheck = checkEventIdConflict(eventId, "consume", consumeArgs);
      if (existingCheck !== null) return existingCheck;

      if (!reservations.has(reservationId)) {
        return fail("UNKNOWN_RESERVATION");
      }

      const reservation = reservations.get(reservationId)!;
      if (reservation.status !== "ACTIVE") {
        return fail("RESERVATION_CLOSED");
      }

      const sample = samples.get(reservation.sampleId)!;
      const consumeQty = reservation.qty;

      sample.remaining -= consumeQty;
      sample.reserved -= consumeQty;
      sample.consumedQty += consumeQty;

      if (sample.remaining === 0) {
        sample.status = "DEPLETED";
        containerToSample.delete(sample.containerId!);
        sample.containerId = null;
      }

      reservation.status = "CONSUMED";

      return {
        ok: true,
        value: success(eventId, "consume", consumeArgs),
      };
    },

    discard(eventId: ID, sampleId: ID) {
      const discardArgs = { sampleId };
      const existingCheck = checkEventIdConflict(eventId, "discard", discardArgs);
      if (existingCheck !== null) return existingCheck;

      if (!samples.has(sampleId)) {
        return fail("UNKNOWN_SAMPLE");
      }

      const sample = samples.get(sampleId)!;
      if (sample.status !== "ACTIVE") {
        return fail("NOT_ACTIVE");
      }

      if (sample.reserved > 0) {
        return fail("HAS_RESERVATION");
      }

      sample.discardedQty += sample.remaining;
      sample.remaining = 0;
      sample.status = "DISCARDED";
      containerToSample.delete(sample.containerId!);
      sample.containerId = null;

      return {
        ok: true,
        value: success(eventId, "discard", discardArgs),
      };
    },

    move(eventId: ID, sampleId: ID, targetContainerId: ID) {
      const moveArgs = { sampleId, targetContainerId };
      const existingCheck = checkEventIdConflict(eventId, "move", moveArgs);
      if (existingCheck !== null) return existingCheck;

      if (!samples.has(sampleId)) {
        return fail("UNKNOWN_SAMPLE");
      }

      const sample = samples.get(sampleId)!;
      if (sample.status !== "ACTIVE") {
        return fail("NOT_ACTIVE");
      }

      if (!containerMap.has(targetContainerId)) {
        return fail("UNKNOWN_CONTAINER");
      }

      if (targetContainerId === sample.containerId) {
        return fail("CONTAINER_OCCUPIED");
      }

      if (containerToSample.has(targetContainerId)) {
        return fail("CONTAINER_OCCUPIED");
      }

      const targetContainer = containerMap.get(targetContainerId)!;
      if (sample.remaining > targetContainer.capacity) {
        return fail("CAPACITY_EXCEEDED");
      }

      const oldContainerId = sample.containerId!;
      sample.containerId = targetContainerId;
      containerToSample.delete(oldContainerId);
      containerToSample.set(targetContainerId, sampleId);

      return {
        ok: true,
        value: success(eventId, "move", moveArgs),
      };
    },

    transferCustody(eventId: ID, sampleId: ID, custodianId: ID) {
      const transferArgs = { sampleId, custodianId };
      const existingCheck = checkEventIdConflict(
        eventId,
        "transferCustody",
        transferArgs
      );
      if (existingCheck !== null) return existingCheck;

      if (!samples.has(sampleId)) {
        return fail("UNKNOWN_SAMPLE");
      }

      const sample = samples.get(sampleId)!;
      if (sample.status !== "ACTIVE") {
        return fail("NOT_ACTIVE");
      }

      if (!custodians.has(custodianId)) {
        return fail("UNKNOWN_CUSTODIAN");
      }

      sample.custodianId = custodianId;

      return {
        ok: true,
        value: success(eventId, "transferCustody", transferArgs),
      };
    },

    snapshot(): Snapshot {
      return {
        samples: Array.from(samples.values()),
        reservations: Array.from(reservations.values()),
        events: eventLog,
      };
    },
  };

  return lab;
}
