import {
  ID,
  Qty,
  Result,
  Receipt,
  SampleStatus,
  ReservationStatus,
  SampleView,
  ReservationView,
  EventView,
  Snapshot,
  Catalog,
  Lab,
  ErrorCode,
} from "./types.js";

type InternalSample = {
  sampleId: ID;
  parentId: ID | null;
  rootId: ID;
  remaining: Qty;
  reserved: Qty;
  consumedQty: Qty;
  discardedQty: Qty;
  status: SampleStatus;
  containerId: ID | null;
  custodianId: ID;
};

type InternalReservation = {
  reservationId: ID;
  sampleId: ID;
  qty: Qty;
  status: ReservationStatus;
};

type CachedEvent = {
  eventId: ID;
  seq: number;
  op: string;
  args: unknown;
};

function deepCopy(obj: unknown): unknown {
  if (obj === null || obj === undefined) return obj;
  if (typeof obj !== "object") return obj;
  if (obj instanceof Date) return new Date(obj);
  if (Array.isArray(obj)) return obj.map(deepCopy);
  const copy: Record<string, unknown> = {};
  for (const key in obj) {
    if (Object.prototype.hasOwnProperty.call(obj, key)) {
      copy[key] = deepCopy((obj as Record<string, unknown>)[key]);
    }
  }
  return copy;
}

function deepEqual(a: unknown, b: unknown): boolean {
  if (a === b) return true;
  if (a === null || b === null) return false;
  if (typeof a !== typeof b) return false;
  if (typeof a !== "object") return a === b;

  const aArray = Array.isArray(a);
  const bArray = Array.isArray(b);

  if (aArray !== bArray) return false;

  if (aArray && bArray) {
    if (a.length !== b.length) return false;
    return (a as unknown[]).every((item, idx) => deepEqual(item, (b as unknown[])[idx]));
  }

  const aKeys = Object.keys(a as Record<string, unknown>).sort();
  const bKeys = Object.keys(b as Record<string, unknown>).sort();

  if (aKeys.length !== bKeys.length) return false;
  if (!aKeys.every((k, idx) => k === bKeys[idx])) return false;

  return aKeys.every((key) => deepEqual((a as any)[key], (b as any)[key]));
}

export function createLab(catalog: Catalog): Lab {
  const validLocations = new Set(catalog.locationIds);
  const validCustodians = new Set(catalog.custodianIds);
  const validContainers = new Map<ID, { locationId: ID; capacity: Qty }>();

  for (const container of catalog.containers) {
    validContainers.set(container.containerId, {
      locationId: container.locationId,
      capacity: container.capacity,
    });
  }

  const samples = new Map<ID, InternalSample>();
  const reservations = new Map<ID, InternalReservation>();
  const containerToSample = new Map<ID, ID | null>();
  const usedEventIds = new Map<ID, CachedEvent>();
  const events: EventView[] = [];
  let seq = 0;

  for (const containerId of validContainers.keys()) {
    containerToSample.set(containerId, null);
  }

  function error<T>(code: ErrorCode): Result<T> {
    return { ok: false, code };
  }

  function success<T>(value: T): Result<T> {
    return { ok: true, value };
  }

  function recordEvent(eventId: ID, op: EventView["op"], args: unknown): void {
    events.push({
      seq: ++seq,
      eventId,
      op,
      args: deepCopy(args),
    });

    usedEventIds.set(eventId, {
      eventId,
      seq,
      op,
      args: deepCopy(args),
    });
  }

  function isValidQty(qty: unknown): boolean {
    return typeof qty === "number" && qty >= 1 && qty <= 1000000 && Number.isInteger(qty);
  }

  const lab: Lab = {
    receive(eventId: ID, input: { sampleId: ID; containerId: ID; custodianId: ID; qty: Qty }) {
      if (usedEventIds.has(eventId)) {
        const cached = usedEventIds.get(eventId)!;
        if (
          cached.op === "receive" &&
          deepEqual(cached.args, { sampleId: input.sampleId, containerId: input.containerId, custodianId: input.custodianId, qty: input.qty })
        ) {
          return success({ eventId, seq: cached.seq });
        }
        return error("EVENT_CONFLICT");
      }

      if (!isValidQty(input.qty)) {
        return error("INVALID_INPUT");
      }

      if (samples.has(input.sampleId)) {
        return error("ID_TAKEN");
      }

      if (!validCustodians.has(input.custodianId)) {
        return error("UNKNOWN_CUSTODIAN");
      }

      if (!validContainers.has(input.containerId)) {
        return error("UNKNOWN_CONTAINER");
      }

      if (containerToSample.get(input.containerId) !== null) {
        return error("CONTAINER_OCCUPIED");
      }

      const container = validContainers.get(input.containerId)!;
      if (input.qty > container.capacity) {
        return error("CAPACITY_EXCEEDED");
      }

      const sample: InternalSample = {
        sampleId: input.sampleId,
        parentId: null,
        rootId: input.sampleId,
        remaining: input.qty,
        reserved: 0,
        consumedQty: 0,
        discardedQty: 0,
        status: "ACTIVE",
        containerId: input.containerId,
        custodianId: input.custodianId,
      };

      samples.set(input.sampleId, sample);
      containerToSample.set(input.containerId, input.sampleId);

      recordEvent(eventId, "receive", {
        sampleId: input.sampleId,
        containerId: input.containerId,
        custodianId: input.custodianId,
        qty: input.qty,
      });

      return success({ eventId, seq });
    },

    split(eventId: ID, parentId: ID, children: Array<{ sampleId: ID; containerId: ID; qty: Qty }>) {
      if (usedEventIds.has(eventId)) {
        const cached = usedEventIds.get(eventId)!;
        if (
          cached.op === "split" &&
          deepEqual(cached.args, { parentId, children })
        ) {
          return success({ eventId, seq: cached.seq });
        }
        return error("EVENT_CONFLICT");
      }

      if (!Array.isArray(children) || children.length === 0) {
        return error("INVALID_INPUT");
      }

      const childContainerIds = new Set<ID>();
      for (const child of children) {
        if (!isValidQty(child.qty)) {
          return error("INVALID_INPUT");
        }
        if (samples.has(child.sampleId)) {
          return error("ID_TAKEN");
        }
        if (!validContainers.has(child.containerId)) {
          return error("UNKNOWN_CONTAINER");
        }
        if (childContainerIds.has(child.containerId)) {
          return error("CONTAINER_OCCUPIED");
        }
        childContainerIds.add(child.containerId);
      }

      if (!samples.has(parentId)) {
        return error("UNKNOWN_SAMPLE");
      }

      const parent = samples.get(parentId)!;
      if (parent.status !== "ACTIVE") {
        return error("NOT_ACTIVE");
      }

      if (parent.containerId !== null && childContainerIds.has(parent.containerId)) {
        return error("CONTAINER_OCCUPIED");
      }

      for (const child of children) {
        if (containerToSample.get(child.containerId) !== null) {
          return error("CONTAINER_OCCUPIED");
        }
        const targetContainer = validContainers.get(child.containerId)!;
        if (child.qty > targetContainer.capacity) {
          return error("CAPACITY_EXCEEDED");
        }
      }

      const totalQty = children.reduce((sum, c) => sum + c.qty, 0);
      const availableQty = parent.remaining - parent.reserved;

      if (totalQty > availableQty) {
        return error("INSUFFICIENT");
      }

      parent.remaining -= totalQty;

      for (const child of children) {
        const newSample: InternalSample = {
          sampleId: child.sampleId,
          parentId: parentId,
          rootId: parent.rootId,
          remaining: child.qty,
          reserved: 0,
          consumedQty: 0,
          discardedQty: 0,
          status: "ACTIVE",
          containerId: child.containerId,
          custodianId: parent.custodianId,
        };

        samples.set(child.sampleId, newSample);
        containerToSample.set(child.containerId, child.sampleId);
      }

      if (parent.remaining === 0) {
        parent.status = "DEPLETED";
        if (parent.containerId !== null) {
          containerToSample.set(parent.containerId, null);
          parent.containerId = null;
        }
      }

      recordEvent(eventId, "split", { parentId, children: deepCopy(children) });

      return success({ eventId, seq });
    },

    reserve(eventId: ID, input: { reservationId: ID; sampleId: ID; qty: Qty }) {
      if (usedEventIds.has(eventId)) {
        const cached = usedEventIds.get(eventId)!;
        if (
          cached.op === "reserve" &&
          deepEqual(cached.args, {
            reservationId: input.reservationId,
            sampleId: input.sampleId,
            qty: input.qty,
          })
        ) {
          return success({ eventId, seq: cached.seq });
        }
        return error("EVENT_CONFLICT");
      }

      if (!isValidQty(input.qty)) {
        return error("INVALID_INPUT");
      }

      if (reservations.has(input.reservationId)) {
        return error("ID_TAKEN");
      }

      if (!samples.has(input.sampleId)) {
        return error("UNKNOWN_SAMPLE");
      }

      const sample = samples.get(input.sampleId)!;

      if (sample.status !== "ACTIVE") {
        return error("NOT_ACTIVE");
      }

      const availableQty = sample.remaining - sample.reserved;
      if (input.qty > availableQty) {
        return error("INSUFFICIENT");
      }

      const reservation: InternalReservation = {
        reservationId: input.reservationId,
        sampleId: input.sampleId,
        qty: input.qty,
        status: "ACTIVE",
      };

      reservations.set(input.reservationId, reservation);
      sample.reserved += input.qty;

      recordEvent(eventId, "reserve", {
        reservationId: input.reservationId,
        sampleId: input.sampleId,
        qty: input.qty,
      });

      return success({ eventId, seq });
    },

    cancel(eventId: ID, reservationId: ID) {
      if (usedEventIds.has(eventId)) {
        const cached = usedEventIds.get(eventId)!;
        if (
          cached.op === "cancel" &&
          deepEqual(cached.args, { reservationId })
        ) {
          return success({ eventId, seq: cached.seq });
        }
        return error("EVENT_CONFLICT");
      }

      if (!reservations.has(reservationId)) {
        return error("UNKNOWN_RESERVATION");
      }

      const reservation = reservations.get(reservationId)!;

      if (reservation.status !== "ACTIVE") {
        return error("RESERVATION_CLOSED");
      }

      const sample = samples.get(reservation.sampleId)!;
      sample.reserved -= reservation.qty;
      reservation.status = "CANCELED";

      recordEvent(eventId, "cancel", { reservationId });

      return success({ eventId, seq });
    },

    consume(eventId: ID, reservationId: ID) {
      if (usedEventIds.has(eventId)) {
        const cached = usedEventIds.get(eventId)!;
        if (
          cached.op === "consume" &&
          deepEqual(cached.args, { reservationId })
        ) {
          return success({ eventId, seq: cached.seq });
        }
        return error("EVENT_CONFLICT");
      }

      if (!reservations.has(reservationId)) {
        return error("UNKNOWN_RESERVATION");
      }

      const reservation = reservations.get(reservationId)!;

      if (reservation.status !== "ACTIVE") {
        return error("RESERVATION_CLOSED");
      }

      const sample = samples.get(reservation.sampleId)!;
      sample.remaining -= reservation.qty;
      sample.reserved -= reservation.qty;
      sample.consumedQty += reservation.qty;
      reservation.status = "CONSUMED";

      if (sample.remaining === 0) {
        sample.status = "DEPLETED";
        if (sample.containerId !== null) {
          containerToSample.set(sample.containerId, null);
          sample.containerId = null;
        }
      }

      recordEvent(eventId, "consume", { reservationId });

      return success({ eventId, seq });
    },

    discard(eventId: ID, sampleId: ID) {
      if (usedEventIds.has(eventId)) {
        const cached = usedEventIds.get(eventId)!;
        if (
          cached.op === "discard" &&
          deepEqual(cached.args, { sampleId })
        ) {
          return success({ eventId, seq: cached.seq });
        }
        return error("EVENT_CONFLICT");
      }

      if (!samples.has(sampleId)) {
        return error("UNKNOWN_SAMPLE");
      }

      const sample = samples.get(sampleId)!;

      if (sample.status !== "ACTIVE") {
        return error("NOT_ACTIVE");
      }

      if (sample.reserved > 0) {
        return error("HAS_RESERVATION");
      }

      sample.discardedQty += sample.remaining;
      sample.remaining = 0;
      sample.status = "DISCARDED";

      if (sample.containerId !== null) {
        containerToSample.set(sample.containerId, null);
        sample.containerId = null;
      }

      recordEvent(eventId, "discard", { sampleId });

      return success({ eventId, seq });
    },

    move(eventId: ID, sampleId: ID, targetContainerId: ID) {
      if (usedEventIds.has(eventId)) {
        const cached = usedEventIds.get(eventId)!;
        if (
          cached.op === "move" &&
          deepEqual(cached.args, { sampleId, targetContainerId })
        ) {
          return success({ eventId, seq: cached.seq });
        }
        return error("EVENT_CONFLICT");
      }

      if (!samples.has(sampleId)) {
        return error("UNKNOWN_SAMPLE");
      }

      const sample = samples.get(sampleId)!;

      if (sample.status !== "ACTIVE") {
        return error("NOT_ACTIVE");
      }

      if (!validContainers.has(targetContainerId)) {
        return error("UNKNOWN_CONTAINER");
      }

      if (sample.containerId === targetContainerId) {
        return error("CONTAINER_OCCUPIED");
      }

      if (containerToSample.get(targetContainerId) !== null) {
        return error("CONTAINER_OCCUPIED");
      }

      const targetContainer = validContainers.get(targetContainerId)!;
      if (sample.remaining > targetContainer.capacity) {
        return error("CAPACITY_EXCEEDED");
      }

      const oldContainerId = sample.containerId;
      if (oldContainerId !== null) {
        containerToSample.set(oldContainerId, null);
      }
      containerToSample.set(targetContainerId, sampleId);
      sample.containerId = targetContainerId;

      recordEvent(eventId, "move", { sampleId, targetContainerId });

      return success({ eventId, seq });
    },

    transferCustody(eventId: ID, sampleId: ID, custodianId: ID) {
      if (usedEventIds.has(eventId)) {
        const cached = usedEventIds.get(eventId)!;
        if (
          cached.op === "transferCustody" &&
          deepEqual(cached.args, { sampleId, custodianId })
        ) {
          return success({ eventId, seq: cached.seq });
        }
        return error("EVENT_CONFLICT");
      }

      if (!samples.has(sampleId)) {
        return error("UNKNOWN_SAMPLE");
      }

      const sample = samples.get(sampleId)!;

      if (sample.status !== "ACTIVE") {
        return error("NOT_ACTIVE");
      }

      if (!validCustodians.has(custodianId)) {
        return error("UNKNOWN_CUSTODIAN");
      }

      sample.custodianId = custodianId;

      recordEvent(eventId, "transferCustody", { sampleId, custodianId });

      return success({ eventId, seq });
    },

    snapshot(): Snapshot {
      const sampleViews: SampleView[] = Array.from(samples.values()).map((s) => ({
        sampleId: s.sampleId,
        parentId: s.parentId,
        rootId: s.rootId,
        remaining: s.remaining,
        reserved: s.reserved,
        consumedQty: s.consumedQty,
        discardedQty: s.discardedQty,
        status: s.status,
        containerId: s.containerId,
        custodianId: s.custodianId,
      }));

      const reservationViews: ReservationView[] = Array.from(reservations.values()).map((r) => ({
        reservationId: r.reservationId,
        sampleId: r.sampleId,
        qty: r.qty,
        status: r.status,
      }));

      return {
        samples: sampleViews,
        reservations: reservationViews,
        events: events.slice(),
      };
    },
  };

  return lab;
}
