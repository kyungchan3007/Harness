import { describe, expect, it } from "vitest";
import { createLab } from "./lab.js";
import type { Catalog, Lab, Result } from "./types.js";

const catalog: Catalog = {
  locationIds: ["L1", "L2"],
  custodianIds: ["kim", "lee"],
  containers: [
    { containerId: "A", locationId: "L1", capacity: 100, supportedGrades: ["AMBIENT"] },
    { containerId: "B", locationId: "L1", capacity: 100, supportedGrades: ["AMBIENT", "CHILLED"] },
    { containerId: "C", locationId: "L2", capacity: 100, supportedGrades: ["FROZEN"] },
    { containerId: "D", locationId: "L2", capacity: 100, supportedGrades: ["AMBIENT"] },
    { containerId: "E", locationId: "L2", capacity: 100, supportedGrades: ["AMBIENT"] },
    { containerId: "S", locationId: "L2", capacity: 5, supportedGrades: ["AMBIENT"] },
  ],
};

let n = 0;
const id = () => `e${++n}`;
const code = (r: Result<unknown>) => (r.ok ? "OK" : r.code);
function setup(qty = 50): Lab {
  const lab = createLab(catalog);
  expect(lab.receive(id(), { sampleId: "s1", containerId: "A", custodianId: "kim", qty, storageGrade: "AMBIENT" }).ok).toBe(true);
  return lab;
}
const sample = (lab: Lab, sid: string) => lab.snapshot().samples.find((x) => x.sampleId === sid)!;

describe("격리", () => {
  it("설정·해제, 같은 값 반복도 성공, 이벤트 기록", () => {
    const lab = setup();
    expect(code(lab.quarantine(id(), "s1"))).toBe("OK");
    expect(code(lab.quarantine(id(), "s1"))).toBe("OK");
    expect(sample(lab, "s1").quarantined).toBe(true);
    expect(code(lab.release(id(), "s1"))).toBe("OK");
    expect(code(lab.release(id(), "s1"))).toBe("OK");
    expect(sample(lab, "s1").quarantined).toBe(false);
    expect(lab.snapshot().events.map((e) => e.op).slice(1)).toEqual(["quarantine", "quarantine", "release", "release"]);
    expect(code(lab.quarantine(id(), "nope"))).toBe("UNKNOWN_SAMPLE");
  });
  it("격리 중 분취·새 예약·소비 거절, 예약은 그대로, 막지 않는 명령은 동작", () => {
    const lab = setup();
    lab.reserve(id(), { reservationId: "r1", sampleId: "s1", qty: 10, expiresAt: 5 });
    lab.quarantine(id(), "s1");
    expect(sample(lab, "s1").reserved).toBe(10);
    expect(code(lab.split(id(), "s1", [{ sampleId: "c1", containerId: "D", qty: 5 }]))).toBe("QUARANTINED");
    expect(code(lab.reserve(id(), { reservationId: "r2", sampleId: "s1", qty: 1, expiresAt: null }))).toBe("QUARANTINED");
    expect(code(lab.consume(id(), "r1"))).toBe("QUARANTINED");
    expect(code(lab.move(id(), "s1", "B"))).toBe("OK");
    expect(code(lab.transferCustody(id(), "s1", "lee"))).toBe("OK");
    expect(code(lab.advanceTime(id(), 5))).toBe("OK");
    const snap = lab.snapshot();
    expect(snap.reservations[0]?.status).toBe("EXPIRED");
    expect(snap.samples[0]?.quarantined).toBe(true);
    expect(code(lab.discard(id(), "s1"))).toBe("OK");
  });
  it("격리 중 예약 취소는 예약 양만 푼다", () => {
    const lab = setup();
    lab.reserve(id(), { reservationId: "r1", sampleId: "s1", qty: 10, expiresAt: null });
    lab.quarantine(id(), "s1");
    expect(code(lab.cancel(id(), "r1"))).toBe("OK");
    expect(sample(lab, "s1")).toMatchObject({ reserved: 0, remaining: 50, quarantined: true });
  });
  it("끝난 시료는 NOT_ACTIVE", () => {
    const lab = setup();
    lab.discard(id(), "s1");
    expect(code(lab.quarantine(id(), "s1"))).toBe("NOT_ACTIVE");
  });
});

describe("보관 등급", () => {
  it("받기·이동은 정확한 지원 등급만 허용", () => {
    const lab = createLab(catalog);
    expect(code(lab.receive(id(), { sampleId: "x", containerId: "A", custodianId: "kim", qty: 5, storageGrade: "CHILLED" }))).toBe("STORAGE_INCOMPATIBLE");
    expect(code(lab.receive(id(), { sampleId: "x", containerId: "C", custodianId: "kim", qty: 5, storageGrade: "CHILLED" }))).toBe("STORAGE_INCOMPATIBLE");
    expect(code(lab.receive(id(), { sampleId: "x", containerId: "B", custodianId: "kim", qty: 5, storageGrade: "CHILLED" }))).toBe("OK");
    expect(code(lab.move(id(), "x", "A"))).toBe("STORAGE_INCOMPATIBLE");
    expect(code(lab.move(id(), "x", "C"))).toBe("STORAGE_INCOMPATIBLE");
    expect(lab.snapshot().samples).toHaveLength(1);
  });
  it("분취 자식은 부모 등급을 상속하고 용기가 맞아야 한다(원자적)", () => {
    const lab = createLab(catalog);
    lab.receive(id(), { sampleId: "f", containerId: "C", custodianId: "kim", qty: 50, storageGrade: "FROZEN" });
    expect(code(lab.split(id(), "f", [{ sampleId: "c1", containerId: "C2" as never, qty: 5 }]))).toBe("UNKNOWN_CONTAINER");
    expect(code(lab.split(id(), "f", [{ sampleId: "c1", containerId: "D", qty: 5 }]))).toBe("STORAGE_INCOMPATIBLE");
    expect(lab.snapshot().samples).toHaveLength(1);
    const lab2 = createLab({ ...catalog, containers: [...catalog.containers, { containerId: "C2", locationId: "L2", capacity: 100, supportedGrades: ["FROZEN"] }] });
    lab2.receive(id(), { sampleId: "f", containerId: "C", custodianId: "kim", qty: 50, storageGrade: "FROZEN" });
    expect(code(lab2.split(id(), "f", [{ sampleId: "c1", containerId: "C2", qty: 5 }]))).toBe("OK");
    expect(sample(lab2, "c1").storageGrade).toBe("FROZEN");
  });
});

describe("예약 만료", () => {
  it("시각은 절대값, 과거·잘못된 값은 INVALID_INPUT, 같은 시각은 허용", () => {
    const lab = setup();
    expect(code(lab.advanceTime(id(), 10))).toBe("OK");
    expect(code(lab.advanceTime(id(), 9))).toBe("INVALID_INPUT");
    expect(code(lab.advanceTime(id(), 10))).toBe("OK");
    for (const bad of [-1, 1.5, NaN, Infinity, Number.MAX_SAFE_INTEGER + 1]) expect(code(lab.advanceTime(id(), bad))).toBe("INVALID_INPUT");
    expect(lab.snapshot().now).toBe(10);
    expect(lab.snapshot().events).toHaveLength(3);
  });
  it("expiresAt 검증: null 또는 현재보다 미래", () => {
    const lab = setup();
    lab.advanceTime(id(), 10);
    for (const bad of [10, 9, 1.5, -1, NaN, undefined as never]) expect(code(lab.reserve(id(), { reservationId: "r", sampleId: "s1", qty: 1, expiresAt: bad }))).toBe("INVALID_INPUT");
    expect(code(lab.reserve(id(), { reservationId: "r", sampleId: "s1", qty: 1, expiresAt: 11 }))).toBe("OK");
    expect(code(lab.reserve(id(), { reservationId: "r2", sampleId: "s1", qty: 1, expiresAt: null }))).toBe("OK");
  });
  it("정확히 만료 시각에 만료, 양만 풀고 실제 양은 불변, 이벤트 하나, 이후 cancel·consume은 RESERVATION_CLOSED", () => {
    const lab = setup();
    lab.reserve(id(), { reservationId: "r1", sampleId: "s1", qty: 10, expiresAt: 5 });
    lab.reserve(id(), { reservationId: "r2", sampleId: "s1", qty: 20, expiresAt: 5 });
    lab.reserve(id(), { reservationId: "r3", sampleId: "s1", qty: 5, expiresAt: 6 });
    lab.reserve(id(), { reservationId: "r4", sampleId: "s1", qty: 5, expiresAt: null });
    const before = lab.snapshot().events.length;
    lab.advanceTime(id(), 4);
    expect(sample(lab, "s1").reserved).toBe(40);
    lab.advanceTime(id(), 5);
    const s = lab.snapshot();
    expect(s.reservations.map((r) => r.status)).toEqual(["EXPIRED", "EXPIRED", "ACTIVE", "ACTIVE"]);
    expect(s.samples[0]).toMatchObject({ reserved: 10, remaining: 50, consumedQty: 0, discardedQty: 0 });
    expect(s.events.length).toBe(before + 2);
    expect(code(lab.cancel(id(), "r1"))).toBe("RESERVATION_CLOSED");
    expect(code(lab.consume(id(), "r2"))).toBe("RESERVATION_CLOSED");
    expect(code(lab.reserve(id(), { reservationId: "r1", sampleId: "s1", qty: 1, expiresAt: null }))).toBe("ID_TAKEN");
    expect(code(lab.reserve(id(), { reservationId: "r9", sampleId: "s1", qty: 40, expiresAt: null }))).toBe("OK");
  });
});

describe("운송", () => {
  it("시작은 목적 용기를 확보하고 출발 용기는 점유를 유지, 도착은 옮김", () => {
    const lab = setup();
    expect(code(lab.startTransit(id(), { transitId: "t1", sampleId: "s1", targetContainerId: "D" }))).toBe("OK");
    expect(sample(lab, "s1").containerId).toBe("A");
    expect(code(lab.receive(id(), { sampleId: "x", containerId: "A", custodianId: "kim", qty: 1, storageGrade: "AMBIENT" }))).toBe("CONTAINER_OCCUPIED");
    expect(code(lab.receive(id(), { sampleId: "x", containerId: "D", custodianId: "kim", qty: 1, storageGrade: "AMBIENT" }))).toBe("CONTAINER_OCCUPIED");
    lab.receive(id(), { sampleId: "y", containerId: "B", custodianId: "kim", qty: 1, storageGrade: "AMBIENT" });
    expect(code(lab.move(id(), "y", "D"))).toBe("CONTAINER_OCCUPIED");
    expect(code(lab.startTransit(id(), { transitId: "t2", sampleId: "y", targetContainerId: "D" }))).toBe("CONTAINER_OCCUPIED");
    expect(code(lab.confirmArrival(id(), "t1"))).toBe("OK");
    expect(sample(lab, "s1").containerId).toBe("D");
    expect(lab.snapshot().transits[0]?.status).toBe("ARRIVED");
    expect(code(lab.receive(id(), { sampleId: "x", containerId: "A", custodianId: "kim", qty: 1, storageGrade: "AMBIENT" }))).toBe("OK");
    expect(code(lab.receive(id(), { sampleId: "z", containerId: "D", custodianId: "kim", qty: 1, storageGrade: "AMBIENT" }))).toBe("CONTAINER_OCCUPIED");
  });
  it("취소는 시료를 출발 용기에 두고 목적 확보를 푼다", () => {
    const lab = setup();
    lab.startTransit(id(), { transitId: "t1", sampleId: "s1", targetContainerId: "D" });
    expect(code(lab.cancelTransit(id(), "t1"))).toBe("OK");
    expect(sample(lab, "s1").containerId).toBe("A");
    expect(lab.snapshot().transits[0]?.status).toBe("CANCELED");
    expect(code(lab.receive(id(), { sampleId: "x", containerId: "D", custodianId: "kim", qty: 1, storageGrade: "AMBIENT" }))).toBe("OK");
    expect(code(lab.move(id(), "s1", "B"))).toBe("OK");
  });
  it("운송 중 분취·소비·폐기·이동·추가 운송 거절, 허용 명령은 동작", () => {
    const lab = setup();
    lab.reserve(id(), { reservationId: "r1", sampleId: "s1", qty: 10, expiresAt: 3 });
    lab.startTransit(id(), { transitId: "t1", sampleId: "s1", targetContainerId: "D" });
    const before = lab.snapshot();
    expect(code(lab.split(id(), "s1", [{ sampleId: "c", containerId: "B", qty: 1 }]))).toBe("IN_TRANSIT");
    expect(code(lab.consume(id(), "r1"))).toBe("IN_TRANSIT");
    expect(code(lab.discard(id(), "s1"))).toBe("IN_TRANSIT");
    expect(code(lab.move(id(), "s1", "B"))).toBe("IN_TRANSIT");
    expect(code(lab.startTransit(id(), { transitId: "t2", sampleId: "s1", targetContainerId: "E" }))).toBe("IN_TRANSIT");
    expect(lab.snapshot()).toEqual(before);
    expect(code(lab.reserve(id(), { reservationId: "r2", sampleId: "s1", qty: 1, expiresAt: null }))).toBe("OK");
    expect(code(lab.quarantine(id(), "s1"))).toBe("OK");
    expect(code(lab.transferCustody(id(), "s1", "lee"))).toBe("OK");
    expect(code(lab.advanceTime(id(), 3))).toBe("OK");
    expect(lab.snapshot().reservations[0]?.status).toBe("EXPIRED");
    expect(code(lab.cancel(id(), "r2"))).toBe("OK");
    expect(code(lab.confirmArrival(id(), "t1"))).toBe("OK");
    expect(sample(lab, "s1")).toMatchObject({ quarantined: true, custodianId: "lee", containerId: "D", remaining: 50 });
  });
  it("검증: 등급·용량·존재·ID", () => {
    const lab = setup();
    expect(code(lab.startTransit(id(), { transitId: "t1", sampleId: "s1", targetContainerId: "C" }))).toBe("STORAGE_INCOMPATIBLE");
    expect(code(lab.startTransit(id(), { transitId: "t1", sampleId: "s1", targetContainerId: "S" }))).toBe("CAPACITY_EXCEEDED");
    expect(code(lab.startTransit(id(), { transitId: "t1", sampleId: "s1", targetContainerId: "Z" }))).toBe("UNKNOWN_CONTAINER");
    expect(code(lab.startTransit(id(), { transitId: "t1", sampleId: "s1", targetContainerId: "A" }))).toBe("CONTAINER_OCCUPIED");
    expect(code(lab.startTransit(id(), { transitId: "t1", sampleId: "q", targetContainerId: "D" }))).toBe("UNKNOWN_SAMPLE");
    expect(lab.snapshot().transits).toHaveLength(0);
    expect(lab.snapshot().events).toHaveLength(1);
    expect(code(lab.startTransit(id(), { transitId: "t1", sampleId: "s1", targetContainerId: "D" }))).toBe("OK");
    lab.cancelTransit(id(), "t1");
    expect(code(lab.startTransit(id(), { transitId: "t1", sampleId: "s1", targetContainerId: "D" }))).toBe("ID_TAKEN");
    expect(code(lab.confirmArrival(id(), "t1"))).toBe("TRANSIT_CLOSED");
    expect(code(lab.cancelTransit(id(), "t1"))).toBe("TRANSIT_CLOSED");
    expect(code(lab.confirmArrival(id(), "none"))).toBe("UNKNOWN_TRANSIT");
  });
  it("끝난 운송의 같은 이벤트 재전송은 처음 Receipt, 다른 인자는 EVENT_CONFLICT", () => {
    const lab = setup();
    lab.startTransit("st", { transitId: "t1", sampleId: "s1", targetContainerId: "D" });
    const a = lab.confirmArrival("ca", "t1");
    const len = lab.snapshot().events.length;
    expect(lab.confirmArrival("ca", "t1")).toEqual(a);
    expect(lab.startTransit("st", { targetContainerId: "D", sampleId: "s1", transitId: "t1" }).ok).toBe(true);
    expect(code(lab.startTransit("st", { transitId: "t1", sampleId: "s1", targetContainerId: "E" }))).toBe("EVENT_CONFLICT");
    expect(code(lab.cancelTransit("ca", "t1"))).toBe("EVENT_CONFLICT");
    expect(lab.snapshot().events).toHaveLength(len);
  });
  it("만료·격리는 운송 확보를 바꾸지 않는다", () => {
    const lab = setup();
    lab.reserve(id(), { reservationId: "r1", sampleId: "s1", qty: 10, expiresAt: 2 });
    lab.startTransit(id(), { transitId: "t1", sampleId: "s1", targetContainerId: "D" });
    lab.advanceTime(id(), 9);
    expect(code(lab.receive(id(), { sampleId: "x", containerId: "D", custodianId: "kim", qty: 1, storageGrade: "AMBIENT" }))).toBe("CONTAINER_OCCUPIED");
  });
});

describe("재전송·이력", () => {
  it("실패해도 먼저 성공한 재전송은 처음 Receipt, 실패한 eventId는 소모되지 않는다", () => {
    const lab = setup();
    const r = lab.quarantine("q1", "s1");
    lab.discard(id(), "s1");
    expect(lab.quarantine("q1", "s1")).toEqual(r);
    expect(code(lab.quarantine("q2", "s1"))).toBe("NOT_ACTIVE");
    expect(code(lab.release("q2", "s1"))).toBe("NOT_ACTIVE");
    expect(code(lab.advanceTime("t", -1))).toBe("INVALID_INPUT");
    expect(code(lab.advanceTime("t", 3))).toBe("OK");
  });
  it("이력 args는 호출자 객체 수정과 분리된다", () => {
    const lab = setup();
    const input = { transitId: "t1", sampleId: "s1", targetContainerId: "D" };
    lab.startTransit(id(), input);
    input.targetContainerId = "E";
    const snap = lab.snapshot();
    expect(snap.events[1]?.args).toEqual({ transitId: "t1", sampleId: "s1", targetContainerId: "D" });
    expect(snap.events.map((e) => e.seq)).toEqual([1, 2]);
  });
});
