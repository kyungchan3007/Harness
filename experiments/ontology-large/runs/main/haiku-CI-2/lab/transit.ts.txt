// 운송 — 시작·도착 확정·취소 (docs/transit.md)
import { active, inTransit } from "./state.js";
import type { Engine } from "./events.js";
import type { Storage } from "./storage.js";
import type { Lab } from "./types.js";

export function transitCommands(engine: Engine, storage: Storage): Pick<Lab, "startTransit" | "confirmArrival" | "cancelTransit"> {
  return {
    startTransit(eventId, input) {
      return engine.run(eventId, "startTransit", input, (s) => {
        if (s.transits.has(input.transitId)) return "ID_TAKEN";
        const sample = active(s, input.sampleId);
        if (typeof sample === "string") return sample;
        if (inTransit(s, input.sampleId)) return "IN_TRANSIT";
        const target = storage.get(input.targetContainerId);
        if (!target) return "UNKNOWN_CONTAINER";
        if (!storage.supports(input.targetContainerId, sample.storageGrade)) return "STORAGE_INCOMPATIBLE";
        if (!storage.isFree(s, input.targetContainerId)) return "CONTAINER_OCCUPIED";
        if (sample.remaining > target.capacity) return "CAPACITY_EXCEEDED";
        s.transits.set(input.transitId, {
          transitId: input.transitId,
          sampleId: input.sampleId,
          sourceContainerId: sample.containerId!,
          targetContainerId: input.targetContainerId,
          status: "ACTIVE",
        });
        s.reserved.set(input.targetContainerId, input.transitId);
        return null;
      });
    },
    confirmArrival(eventId, transitId) {
      return engine.run(eventId, "confirmArrival", { transitId }, (s) => {
        const t = s.transits.get(transitId);
        if (!t) return "UNKNOWN_TRANSIT";
        if (t.status !== "ACTIVE") return "TRANSIT_CLOSED";
        const sample = s.samples.get(t.sampleId)!;
        s.occupied.delete(t.sourceContainerId);
        s.occupied.set(t.targetContainerId, t.sampleId);
        sample.containerId = t.targetContainerId;
        t.status = "ARRIVED";
        s.reserved.delete(t.targetContainerId);
        return null;
      });
    },
    cancelTransit(eventId, transitId) {
      return engine.run(eventId, "cancelTransit", { transitId }, (s) => {
        const t = s.transits.get(transitId);
        if (!t) return "UNKNOWN_TRANSIT";
        if (t.status !== "ACTIVE") return "TRANSIT_CLOSED";
        t.status = "CANCELED";
        s.reserved.delete(t.targetContainerId);
        return null;
      });
    },
  };
}
