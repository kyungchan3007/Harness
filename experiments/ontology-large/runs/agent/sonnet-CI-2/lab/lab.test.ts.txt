import { describe, expect, it } from "vitest";
import { createLab } from "./lab.js";
import type { Catalog, Lab } from "./types.js";

const catalog: Catalog = {
  locationIds: ["L1"],
  custodianIds: ["u1", "u2"],
  containers: [
    { containerId: "A", locationId: "L1", capacity: 100, supportedGrades: ["AMBIENT"] },
    { containerId: "B", locationId: "L1", capacity: 100, supportedGrades: ["AMBIENT", "CHILLED"] },
    { containerId: "C", locationId: "L1", capacity: 100, supportedGrades: ["FROZEN"] },
    { containerId: "D", locationId: "L1", capacity: 100, supportedGrades: ["AMBIENT"] },
    { containerId: "E", locationId: "L1", capacity: 100, supportedGrades: ["AMBIENT"] },
    { containerId: "S", locationId: "L1", capacity: 10, supportedGrades: ["AMBIENT"] },
  ],
};
let n = 0;
const id = () => `ev${++n}`;
const mk = (): Lab => {
  const lab = createLab(catalog);
  expect(lab.receive(id(), { sampleId: "s1", containerId: "A", custodianId: "u1", qty: 50, storageGrade: "AMBIENT" }).ok).toBe(true);
  return lab;
};
const code = (r: { ok: boolean; code?: string }) => (r.ok ? "OK" : r.code);

describe("격리", () => {
  it("같은 값도 성공, 예약은 그대로, 분취·새 예약·소비만 QUARANTINED", () => {
    const lab = mk();
    expect(lab.reserve(id(), { reservationId: "r1", sampleId: "s1", qty: 10, expiresAt: null }).ok).toBe(true);
    expect(lab.quarantine(id(), "s1").ok).toBe(true);
    expect(lab.quarantine(id(), "s1").ok).toBe(true);
    expect(lab.snapshot().reservations[0]).toMatchObject({ status: "ACTIVE", qty: 10 });
    expect(code(lab.split(id(), "s1", [{ sampleId: "c", containerId: "D", qty: 5 }]))).toBe("QUARANTINED");
    expect(code(lab.reserve(id(), { reservationId: "r2", sampleId: "s1", qty: 1, expiresAt: null }))).toBe("QUARANTINED");
    expect(code(lab.consume(id(), "r1"))).toBe("QUARANTINED");
    expect(code(lab.move(id(), "s1", "D"))).toBe("OK");
    expect(code(lab.transferCustody(id(), "s1", "u2"))).toBe("OK");
    expect(code(lab.cancel(id(), "r1"))).toBe("OK");
    expect(lab.snapshot().samples[0]).toMatchObject({ quarantined: true, remaining: 50, reserved: 0 });
    expect(lab.release(id(), "s1").ok).toBe(true);
    expect(code(lab.quarantine(id(), "nope"))).toBe("UNKNOWN_SAMPLE");
    expect(code(lab.discard(id(), "s1"))).toBe("OK");
    expect(code(lab.release(id(), "s1"))).toBe("NOT_ACTIVE");
  });
});

describe("보관 등급", () => {
  it("정확 일치만 허용, 자식은 부모 등급 상속", () => {
    const lab = createLab(catalog);
    expect(code(lab.receive(id(), { sampleId: "x", containerId: "C", custodianId: "u1", qty: 5, storageGrade: "CHILLED" }))).toBe("STORAGE_INCOMPATIBLE");
    expect(code(lab.receive(id(), { sampleId: "x", containerId: "B", custodianId: "u1", qty: 5, storageGrade: "CHILLED" }))).toBe("OK");
    expect(code(lab.split(id(), "x", [{ sampleId: "y", containerId: "A", qty: 1 }]))).toBe("STORAGE_INCOMPATIBLE");
    expect(code(lab.split(id(), "x", [{ sampleId: "y", containerId: "D", qty: 1 }]))).toBe("STORAGE_INCOMPATIBLE");
    expect(code(lab.move(id(), "x", "C"))).toBe("STORAGE_INCOMPATIBLE");
    expect(lab.snapshot().samples).toHaveLength(1);
  });
  it("자식이 부모 등급을 이어받는다", () => {
    const lab = createLab({ ...catalog, containers: [...catalog.containers, { containerId: "B2", locationId: "L1", capacity: 100, supportedGrades: ["CHILLED"] }] });
    lab.receive(id(), { sampleId: "x", containerId: "B", custodianId: "u1", qty: 5, storageGrade: "CHILLED" });
    expect(code(lab.split(id(), "x", [{ sampleId: "y", containerId: "B2", qty: 1 }]))).toBe("OK");
    expect(lab.snapshot().samples.find((q) => q.sampleId === "y")?.storageGrade).toBe("CHILLED");
  });
});

describe("예약 만료", () => {
  it("expiresAt <= now 에서 만료, 한 이벤트, 실제 양 불변", () => {
    const lab = mk();
    lab.reserve(id(), { reservationId: "r1", sampleId: "s1", qty: 10, expiresAt: 5 });
    lab.reserve(id(), { reservationId: "r2", sampleId: "s1", qty: 10, expiresAt: 5 });
    lab.reserve(id(), { reservationId: "r3", sampleId: "s1", qty: 10, expiresAt: null });
    const before = lab.snapshot().events.length;
    expect(code(lab.advanceTime(id(), 4))).toBe("OK");
    expect(lab.snapshot().reservations.map((r) => r.status)).toEqual(["ACTIVE", "ACTIVE", "ACTIVE"]);
    expect(code(lab.advanceTime(id(), 5))).toBe("OK");
    const s = lab.snapshot();
    expect(s.reservations.map((r) => r.status)).toEqual(["EXPIRED", "EXPIRED", "ACTIVE"]);
    expect(s.events.length).toBe(before + 2);
    expect(s.samples[0]).toMatchObject({ remaining: 50, reserved: 10, consumedQty: 0, discardedQty: 0 });
    expect(code(lab.cancel(id(), "r1"))).toBe("RESERVATION_CLOSED");
    expect(code(lab.consume(id(), "r2"))).toBe("RESERVATION_CLOSED");
  });
  it("과거·잘못된 시각은 INVALID_INPUT, 같은 시각은 성공", () => {
    const lab = mk();
    lab.advanceTime(id(), 10);
    for (const t of [9, -1, 1.5, NaN, Number.MAX_SAFE_INTEGER + 1]) expect(code(lab.advanceTime(id(), t))).toBe("INVALID_INPUT");
    expect(code(lab.advanceTime(id(), 10))).toBe("OK");
    for (const e of [10, 3, 1.5, NaN, Number.MAX_SAFE_INTEGER + 1]) expect(code(lab.reserve(id(), { reservationId: "r", sampleId: "s1", qty: 1, expiresAt: e }))).toBe("INVALID_INPUT");
    expect(code(lab.reserve(id(), { reservationId: "r", sampleId: "s1", qty: 1, expiresAt: 11 }))).toBe("OK");
  });
});

describe("운송", () => {
  it("시작·확보·도착", () => {
    const lab = mk();
    expect(code(lab.startTransit(id(), { transitId: "t1", sampleId: "s1", targetContainerId: "D" }))).toBe("OK");
    expect(code(lab.receive(id(), { sampleId: "z", containerId: "D", custodianId: "u1", qty: 1, storageGrade: "AMBIENT" }))).toBe("CONTAINER_OCCUPIED");
    expect(code(lab.receive(id(), { sampleId: "z", containerId: "A", custodianId: "u1", qty: 1, storageGrade: "AMBIENT" }))).toBe("CONTAINER_OCCUPIED");
    expect(code(lab.startTransit(id(), { transitId: "t2", sampleId: "s1", targetContainerId: "E" }))).toBe("IN_TRANSIT");
    expect(code(lab.split(id(), "s1", [{ sampleId: "c", containerId: "E", qty: 1 }]))).toBe("IN_TRANSIT");
    expect(code(lab.discard(id(), "s1"))).toBe("IN_TRANSIT");
    expect(code(lab.move(id(), "s1", "E"))).toBe("IN_TRANSIT");
    expect(code(lab.reserve(id(), { reservationId: "r1", sampleId: "s1", qty: 5, expiresAt: null }))).toBe("OK");
    expect(code(lab.consume(id(), "r1"))).toBe("IN_TRANSIT");
    expect(code(lab.transferCustody(id(), "s1", "u2"))).toBe("OK");
    expect(code(lab.quarantine(id(), "s1"))).toBe("OK");
    expect(code(lab.cancel(id(), "r1"))).toBe("OK");
    const arrive = id();
    expect(code(lab.confirmArrival(arrive, "t1"))).toBe("OK");
    const len = lab.snapshot().events.length;
    expect(code(lab.confirmArrival(arrive, "t1"))).toBe("OK");
    expect(lab.snapshot().events.length).toBe(len);
    expect(code(lab.confirmArrival(id(), "t1"))).toBe("TRANSIT_CLOSED");
    expect(code(lab.cancelTransit(id(), "t1"))).toBe("TRANSIT_CLOSED");
    expect(code(lab.startTransit(id(), { transitId: "t1", sampleId: "s1", targetContainerId: "E" }))).toBe("ID_TAKEN");
    expect(code(lab.confirmArrival(id(), "nope"))).toBe("UNKNOWN_TRANSIT");
    const s = lab.snapshot();
    expect(s.samples[0]).toMatchObject({ containerId: "D", quarantined: true, custodianId: "u2", remaining: 50 });
    expect(code(lab.receive(id(), { sampleId: "z", containerId: "A", custodianId: "u1", qty: 1, storageGrade: "AMBIENT" }))).toBe("OK");
  });
  it("취소는 출발 용기 유지·확보 해제, 목적지 부적합·용량 초과는 기록 없이 실패", () => {
    const lab = mk();
    expect(code(lab.startTransit(id(), { transitId: "t0", sampleId: "s1", targetContainerId: "C" }))).toBe("STORAGE_INCOMPATIBLE");
    expect(code(lab.startTransit(id(), { transitId: "t0", sampleId: "s1", targetContainerId: "S" }))).toBe("CAPACITY_EXCEEDED");
    expect(code(lab.startTransit(id(), { transitId: "t0", sampleId: "s1", targetContainerId: "A" }))).toBe("CONTAINER_OCCUPIED");
    expect(lab.snapshot().transits).toHaveLength(0);
    lab.startTransit(id(), { transitId: "t1", sampleId: "s1", targetContainerId: "D" });
    expect(code(lab.cancelTransit(id(), "t1"))).toBe("OK");
    expect(lab.snapshot().transits[0]?.status).toBe("CANCELED");
    expect(lab.snapshot().samples[0]?.containerId).toBe("A");
    expect(code(lab.move(id(), "s1", "D"))).toBe("OK");
  });
  it("두 운송이 같은 목적지를 잡을 수 없다", () => {
    const lab = mk();
    lab.receive(id(), { sampleId: "s2", containerId: "E", custodianId: "u1", qty: 5, storageGrade: "AMBIENT" });
    lab.startTransit(id(), { transitId: "t1", sampleId: "s1", targetContainerId: "D" });
    expect(code(lab.startTransit(id(), { transitId: "t2", sampleId: "s2", targetContainerId: "D" }))).toBe("CONTAINER_OCCUPIED");
  });
});

describe("재전송·원자성·이력", () => {
  it("성공한 eventId 재전송, 충돌, 실패한 eventId 재사용, 이력 불변", () => {
    const lab = mk();
    const input = { reservationId: "r1", sampleId: "s1", qty: 5, expiresAt: 9 };
    const ev = id();
    const first = lab.reserve(ev, input);
    input.qty = 99;
    expect(lab.reserve(ev, { reservationId: "r1", sampleId: "s1", qty: 5, expiresAt: 9 })).toEqual(first);
    expect(code(lab.reserve(ev, { reservationId: "r1", sampleId: "s1", qty: 6, expiresAt: 9 }))).toBe("EVENT_CONFLICT");
    expect(code(lab.quarantine(ev, "s1"))).toBe("EVENT_CONFLICT");
    expect(lab.snapshot().events.at(-1)?.args).toMatchObject({ qty: 5 });
    const bad = id();
    expect(code(lab.advanceTime(bad, -1))).toBe("INVALID_INPUT");
    expect(code(lab.advanceTime(bad, 1))).toBe("OK");
    const s = lab.snapshot();
    expect(s.events.map((e) => e.seq)).toEqual(s.events.map((_, i) => i + 1));
    expect(s.events.at(-1)).toMatchObject({ op: "advanceTime", args: { to: 1 } });
  });
});
