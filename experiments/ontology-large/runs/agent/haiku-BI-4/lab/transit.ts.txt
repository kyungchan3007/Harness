// 운송 — 시작·도착 확정·취소 (docs/transit.md)
// 시작: 출발 용기는 점유 유지, 목적 용기는 claimed로 배타 확보. 도착: 시료를 목적 용기로 옮기고 확보를 푼다. 취소: 확보만 푼다.
import { active, inTransit, type State } from "./state.js";
import type { Engine } from "./events.js";
import type { Storage } from "./storage.js";
import type { ID, Lab, TransitView } from "./types.js";

export function transitCommands(engine: Engine, storage: Storage): Pick<Lab, "startTransit" | "confirmArrival" | "cancelTransit"> {
  return {
    startTransit(eventId, input) {
      return engine.run(eventId, "startTransit", input, (s) => {
        if (s.transits.has(input.transitId)) return "ID_TAKEN";
        const x = active(s, input.sampleId);
        if (typeof x === "string") return x;
        if (inTransit(s, x.sampleId)) return "IN_TRANSIT";
        const c = storage.get(input.targetContainerId);
        if (!c) return "UNKNOWN_CONTAINER";
        if (!storage.isFree(s, c.containerId)) return "CONTAINER_OCCUPIED";
        if (x.remaining > c.capacity) return "CAPACITY_EXCEEDED";
        if (!storage.supports(c.containerId, x.storageGrade)) return "STORAGE_INCOMPATIBLE";
        const t: TransitView = {
          transitId: input.transitId, sampleId: x.sampleId, sourceContainerId: x.containerId!, targetContainerId: c.containerId, status: "ACTIVE",
        };
        s.transits.set(t.transitId, t);
        s.claimed.set(c.containerId, t.transitId);
        return null;
      });
    },

    confirmArrival(eventId, transitId) {
      return engine.run(eventId, "confirmArrival", { transitId }, (s) => {
        const t = findTransit(s, transitId);
        if (typeof t === "string") return t;
        const x = s.samples.get(t.sampleId)!;
        s.occupied.delete(t.sourceContainerId);
        s.claimed.delete(t.targetContainerId);
        s.occupied.set(t.targetContainerId, x.sampleId);
        x.containerId = t.targetContainerId;
        t.status = "ARRIVED";
        return null;
      });
    },

    cancelTransit(eventId, transitId) {
      return engine.run(eventId, "cancelTransit", { transitId }, (s) => {
        const t = findTransit(s, transitId);
        if (typeof t === "string") return t;
        s.claimed.delete(t.targetContainerId);
        t.status = "CANCELED";
        return null;
      });
    },
  };
}

/** 진행 중인 운송만 돌려준다. 없으면 UNKNOWN_TRANSIT, 끝났으면 TRANSIT_CLOSED */
function findTransit(s: State, id: ID): TransitView | "UNKNOWN_TRANSIT" | "TRANSIT_CLOSED" {
  const t = s.transits.get(id);
  if (!t) return "UNKNOWN_TRANSIT";
  if (t.status !== "ACTIVE") return "TRANSIT_CLOSED";
  return t;
}
