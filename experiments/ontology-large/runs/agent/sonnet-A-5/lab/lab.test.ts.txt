import { describe, expect, it } from "vitest";
import { createLab } from "./lab.js";

const catalog = () => ({
  locationIds: ["L1"],
  custodianIds: ["c1", "c2"],
  containers: [
    { containerId: "A", locationId: "L1", capacity: 100, supportedGrades: ["AMBIENT" as const] },
    { containerId: "B", locationId: "L1", capacity: 100, supportedGrades: ["AMBIENT" as const, "CHILLED" as const] },
    { containerId: "C", locationId: "L1", capacity: 100, supportedGrades: ["FROZEN" as const] },
    { containerId: "D", locationId: "L1", capacity: 100, supportedGrades: ["AMBIENT" as const] },
    { containerId: "E", locationId: "L1", capacity: 10, supportedGrades: ["AMBIENT" as const] },
  ],
});
const setup = () => {
  const lab = createLab(catalog());
  lab.receive("e0", { sampleId: "s1", containerId: "A", custodianId: "c1", qty: 50, storageGrade: "AMBIENT" });
  return lab;
};
const code = (r: { ok: boolean } & Record<string, unknown>) => (r.ok ? "OK" : r.code);

describe("격리", () => {
  it("분취·예약·소비 거절, 예약 취소·이동·폐기는 허용, 멱등", () => {
    const lab = setup();
    lab.reserve("e1", { reservationId: "r1", sampleId: "s1", qty: 10, expiresAt: null });
    expect(lab.quarantine("q1", "s1").ok).toBe(true);
    expect(lab.quarantine("q2", "s1").ok).toBe(true);
    expect(code(lab.split("e2", "s1", [{ sampleId: "k", containerId: "D", qty: 1 }]))).toBe("QUARANTINED");
    expect(code(lab.reserve("e3", { reservationId: "r2", sampleId: "s1", qty: 1, expiresAt: null }))).toBe("QUARANTINED");
    expect(code(lab.consume("e4", "r1"))).toBe("QUARANTINED");
    expect(lab.snapshot().reservations[0]!.status).toBe("ACTIVE");
    expect(lab.move("e5", "s1", "B").ok).toBe(true);
    expect(lab.cancel("e6", "r1").ok).toBe(true);
    expect(lab.release("e7", "s1").ok).toBe(true);
    expect(lab.snapshot().samples[0]!.quarantined).toBe(false);
    expect(code(lab.quarantine("e8", "nope"))).toBe("UNKNOWN_SAMPLE");
    lab.discard("e9", "s1");
    expect(code(lab.quarantine("e10", "s1"))).toBe("NOT_ACTIVE");
  });
});

describe("보관 등급", () => {
  it("받기·이동·분취 거절, 자식은 부모 등급 상속", () => {
    const lab = setup();
    expect(code(lab.receive("e1", { sampleId: "x", containerId: "C", custodianId: "c1", qty: 1, storageGrade: "CHILLED" }))).toBe("STORAGE_INCOMPATIBLE");
    expect(code(lab.move("e2", "s1", "C"))).toBe("STORAGE_INCOMPATIBLE");
    expect(code(lab.split("e3", "s1", [{ sampleId: "k", containerId: "C", qty: 1 }]))).toBe("STORAGE_INCOMPATIBLE");
    expect(lab.split("e4", "s1", [{ sampleId: "k", containerId: "D", qty: 1 }]).ok).toBe(true);
    expect(lab.snapshot().samples.find((x) => x.sampleId === "k")!.storageGrade).toBe("AMBIENT");
    expect(lab.snapshot().events).toHaveLength(2);
  });
});

describe("예약 만료", () => {
  it("검증·만료·양 해제·재전송", () => {
    const lab = setup();
    expect(code(lab.reserve("e1", { reservationId: "r", sampleId: "s1", qty: 5, expiresAt: 0 }))).toBe("INVALID_INPUT");
    expect(lab.reserve("e2", { reservationId: "r", sampleId: "s1", qty: 5, expiresAt: 5 }).ok).toBe(true);
    lab.reserve("e3", { reservationId: "r2", sampleId: "s1", qty: 5, expiresAt: 9 });
    expect(lab.advanceTime("t1", 5).ok).toBe(true);
    expect(code(lab.advanceTime("t2", 4))).toBe("INVALID_INPUT");
    let snap = lab.snapshot();
    expect(snap.reservations.map((r) => r.status)).toEqual(["EXPIRED", "ACTIVE"]);
    expect(snap.samples[0]!).toMatchObject({ remaining: 50, reserved: 5, consumedQty: 0 });
    expect(code(lab.cancel("e4", "r"))).toBe("RESERVATION_CLOSED");
    expect(lab.advanceTime("t1", 5).ok).toBe(true);
    expect(code(lab.advanceTime("t1", 6))).toBe("EVENT_CONFLICT");
    expect(lab.advanceTime("t3", 5).ok).toBe(true);
    snap = lab.snapshot();
    expect(snap.now).toBe(5);
    expect(snap.events.filter((e) => e.op === "advanceTime")).toHaveLength(2);
  });
});

describe("운송", () => {
  it("도착: 목적 용기 배타 확보 후 이동", () => {
    const lab = setup();
    expect(lab.startTransit("e1", { transitId: "t", sampleId: "s1", targetContainerId: "B" }).ok).toBe(true);
    expect(code(lab.receive("e2", { sampleId: "y", containerId: "B", custodianId: "c1", qty: 1, storageGrade: "AMBIENT" }))).toBe("CONTAINER_OCCUPIED");
    expect(code(lab.startTransit("e3", { transitId: "t2", sampleId: "s1", targetContainerId: "D" }))).toBe("IN_TRANSIT");
    expect(code(lab.move("e4", "s1", "D"))).toBe("IN_TRANSIT");
    expect(code(lab.discard("e5", "s1"))).toBe("IN_TRANSIT");
    expect(code(lab.split("e6", "s1", [{ sampleId: "k", containerId: "D", qty: 1 }]))).toBe("IN_TRANSIT");
    lab.reserve("e7", { reservationId: "r", sampleId: "s1", qty: 5, expiresAt: null });
    expect(code(lab.consume("e8", "r"))).toBe("IN_TRANSIT");
    expect(lab.confirmArrival("e9", "t").ok).toBe(true);
    expect(code(lab.cancelTransit("e10", "t"))).toBe("TRANSIT_CLOSED");
    expect(lab.confirmArrival("e9", "t").ok).toBe(true);
    expect(code(lab.startTransit("e11", { transitId: "t", sampleId: "s1", targetContainerId: "D" }))).toBe("ID_TAKEN");
    expect(code(lab.confirmArrival("e12", "zz"))).toBe("UNKNOWN_TRANSIT");
    const snap = lab.snapshot();
    expect(snap.samples[0]!.containerId).toBe("B");
    expect(snap.transits[0]!.status).toBe("ARRIVED");
    expect(lab.receive("e13", { sampleId: "y", containerId: "A", custodianId: "c1", qty: 1, storageGrade: "AMBIENT" }).ok).toBe(true);
  });
  it("취소: 출발 용기 유지, 목적 확보 해제, 실패는 아무것도 안 남김", () => {
    const lab = setup();
    expect(code(lab.startTransit("f0", { transitId: "t", sampleId: "s1", targetContainerId: "C" }))).toBe("STORAGE_INCOMPATIBLE");
    expect(code(lab.startTransit("f1", { transitId: "t", sampleId: "s1", targetContainerId: "A" }))).toBe("CONTAINER_OCCUPIED");
    expect(lab.snapshot().transits).toHaveLength(0);
    lab.startTransit("e1", { transitId: "t", sampleId: "s1", targetContainerId: "D" });
    lab.quarantine("q", "s1");
    expect(lab.cancelTransit("e2", "t").ok).toBe(true);
    const snap = lab.snapshot();
    expect(snap.samples[0]!).toMatchObject({ containerId: "A", quarantined: true });
    expect(snap.transits[0]!.status).toBe("CANCELED");
    expect(lab.move("e3", "s1", "D").ok).toBe(true);
  });
});
