import { describe, expect, it } from "vitest";
import { createLab } from "./lab.js";
import type { Catalog, Lab } from "./types.js";

const catalog: Catalog = {
  locationIds: ["L1"],
  custodianIds: ["u1", "u2"],
  containers: [
    { containerId: "A", locationId: "L1", capacity: 100, supportedGrades: ["AMBIENT"] },
    { containerId: "B", locationId: "L1", capacity: 100, supportedGrades: ["AMBIENT", "CHILLED"] },
    { containerId: "C", locationId: "L1", capacity: 100, supportedGrades: ["FROZEN"] },
    { containerId: "D", locationId: "L1", capacity: 100, supportedGrades: ["AMBIENT"] },
    { containerId: "E", locationId: "L1", capacity: 100, supportedGrades: ["AMBIENT"] },
    { containerId: "S", locationId: "L1", capacity: 5, supportedGrades: ["AMBIENT"] },
  ],
};
let n = 0;
const id = () => `ev${++n}`;
const setup = (grade: "AMBIENT" | "CHILLED" | "FROZEN" = "AMBIENT", container = "A"): Lab => {
  const lab = createLab(catalog);
  expect(lab.receive(id(), { sampleId: "s1", containerId: container, custodianId: "u1", qty: 50, storageGrade: grade }).ok).toBe(true);
  return lab;
};
const code = (r: { ok: boolean; code?: string }) => (r.ok ? "OK" : r.code);

describe("격리", () => {
  it("격리 중 분취·새 예약·소비 거절, 같은 상태 재설정 성공, 예약은 그대로", () => {
    const lab = setup();
    lab.reserve(id(), { reservationId: "r1", sampleId: "s1", qty: 10, expiresAt: null });
    expect(lab.quarantine(id(), "s1").ok).toBe(true);
    expect(lab.quarantine(id(), "s1").ok).toBe(true);
    expect(code(lab.split(id(), "s1", [{ sampleId: "c", containerId: "D", qty: 1 }]))).toBe("QUARANTINED");
    expect(code(lab.reserve(id(), { reservationId: "r2", sampleId: "s1", qty: 1, expiresAt: null }))).toBe("QUARANTINED");
    expect(code(lab.consume(id(), "r1"))).toBe("QUARANTINED");
    expect(lab.snapshot().reservations[0]).toMatchObject({ status: "ACTIVE", qty: 10 });
    expect(lab.cancel(id(), "r1").ok).toBe(true);
    expect(lab.move(id(), "s1", "D").ok).toBe(true);
    expect(lab.release(id(), "s1").ok).toBe(true);
    expect(lab.release(id(), "s1").ok).toBe(true);
    expect(lab.snapshot().samples[0]!.quarantined).toBe(false);
  });
  it("끝난 시료는 NOT_ACTIVE, 없으면 UNKNOWN_SAMPLE", () => {
    const lab = setup();
    expect(code(lab.quarantine(id(), "zz"))).toBe("UNKNOWN_SAMPLE");
    lab.discard(id(), "s1");
    expect(code(lab.quarantine(id(), "s1"))).toBe("NOT_ACTIVE");
  });
});

describe("보관 등급", () => {
  it("받기·이동은 정확 일치만, 분취 자식은 부모 등급 상속·적합성 검사", () => {
    const lab = createLab(catalog);
    expect(code(lab.receive(id(), { sampleId: "x", containerId: "A", custodianId: "u1", qty: 1, storageGrade: "CHILLED" }))).toBe("STORAGE_INCOMPATIBLE");
    expect(code(lab.receive(id(), { sampleId: "x", containerId: "A", custodianId: "u1", qty: 1, storageGrade: "BAD" as never }))).toBe("INVALID_INPUT");
    lab.receive(id(), { sampleId: "s1", containerId: "B", custodianId: "u1", qty: 50, storageGrade: "CHILLED" });
    expect(code(lab.split(id(), "s1", [{ sampleId: "c", containerId: "A", qty: 1 }]))).toBe("STORAGE_INCOMPATIBLE");
    expect(code(lab.move(id(), "s1", "C"))).toBe("STORAGE_INCOMPATIBLE");
    expect(code(lab.split(id(), "s1", [{ sampleId: "c", containerId: "B", qty: 1 }]))).toBe("CONTAINER_OCCUPIED");
    const l2 = setup("AMBIENT", "A");
    l2.split(id(), "s1", [{ sampleId: "c", containerId: "D", qty: 1 }]);
    expect(l2.snapshot().samples.find((s) => s.sampleId === "c")!.storageGrade).toBe("AMBIENT");
  });
  it("분취 실패는 아무것도 바꾸지 않는다", () => {
    const lab = createLab(catalog);
    lab.receive(id(), { sampleId: "s1", containerId: "B", custodianId: "u1", qty: 50, storageGrade: "CHILLED" });
    const before = lab.snapshot();
    expect(code(lab.split(id(), "s1", [{ sampleId: "c1", containerId: "A", qty: 1 }, { sampleId: "c2", containerId: "D", qty: 1 }]))).toBe("STORAGE_INCOMPATIBLE");
    expect(lab.snapshot()).toEqual(before);
  });
});

describe("예약 만료", () => {
  it("경계 포함 만료, 양만 풀림, 이벤트 하나, 이후 cancel/consume은 RESERVATION_CLOSED", () => {
    const lab = setup();
    lab.reserve(id(), { reservationId: "r1", sampleId: "s1", qty: 10, expiresAt: 5 });
    lab.reserve(id(), { reservationId: "r2", sampleId: "s1", qty: 10, expiresAt: 5 });
    lab.reserve(id(), { reservationId: "r3", sampleId: "s1", qty: 10, expiresAt: 6 });
    const ev = lab.snapshot().events.length;
    expect(lab.advanceTime(id(), 5).ok).toBe(true);
    const s = lab.snapshot();
    expect(s.now).toBe(5);
    expect(s.events.length).toBe(ev + 1);
    expect(s.reservations.map((r) => r.status)).toEqual(["EXPIRED", "EXPIRED", "ACTIVE"]);
    expect(s.samples[0]).toMatchObject({ remaining: 50, reserved: 10, consumedQty: 0, discardedQty: 0 });
    expect(code(lab.cancel(id(), "r1"))).toBe("RESERVATION_CLOSED");
    expect(code(lab.consume(id(), "r2"))).toBe("RESERVATION_CLOSED");
  });
  it("시각 검증", () => {
    const lab = setup();
    lab.advanceTime(id(), 5);
    expect(lab.advanceTime(id(), 5).ok).toBe(true);
    expect(code(lab.advanceTime(id(), 4))).toBe("INVALID_INPUT");
    expect(code(lab.advanceTime(id(), 5.5))).toBe("INVALID_INPUT");
    expect(code(lab.advanceTime(id(), Number.NaN))).toBe("INVALID_INPUT");
    for (const e of [5, 4, 1.5, Number.NaN, Number.MAX_SAFE_INTEGER + 1]) {
      expect(code(lab.reserve(id(), { reservationId: `q${e}`, sampleId: "s1", qty: 1, expiresAt: e }))).toBe("INVALID_INPUT");
    }
    expect(lab.reserve(id(), { reservationId: "ok", sampleId: "s1", qty: 1, expiresAt: 6 }).ok).toBe(true);
  });
  it("격리·운송 중 만료도 상태를 바꾸지 않는다", () => {
    const lab = setup();
    lab.reserve(id(), { reservationId: "r1", sampleId: "s1", qty: 10, expiresAt: 3 });
    lab.quarantine(id(), "s1");
    lab.startTransit(id(), { transitId: "t1", sampleId: "s1", targetContainerId: "D" });
    lab.advanceTime(id(), 3);
    const s = lab.snapshot();
    expect(s.reservations[0]!.status).toBe("EXPIRED");
    expect(s.samples[0]).toMatchObject({ quarantined: true, containerId: "A" });
    expect(code(lab.move(id(), "s1", "E"))).toBe("IN_TRANSIT");
    expect(code(lab.startTransit(id(), { transitId: "t2", sampleId: "x", targetContainerId: "D" }))).toBe("UNKNOWN_SAMPLE");
  });
});

describe("운송", () => {
  it("확보·금지·허용·도착·취소", () => {
    const lab = setup();
    expect(lab.startTransit("T1", { transitId: "t1", sampleId: "s1", targetContainerId: "D" }).ok).toBe(true);
    expect(code(lab.startTransit(id(), { transitId: "t1", sampleId: "s1", targetContainerId: "E" }))).toBe("ID_TAKEN");
    expect(code(lab.startTransit(id(), { transitId: "t2", sampleId: "s1", targetContainerId: "E" }))).toBe("IN_TRANSIT");
    // 목적·출발 용기는 다른 시료가 못 씀
    expect(code(lab.receive(id(), { sampleId: "o", containerId: "D", custodianId: "u1", qty: 1, storageGrade: "AMBIENT" }))).toBe("CONTAINER_OCCUPIED");
    expect(code(lab.receive(id(), { sampleId: "o", containerId: "A", custodianId: "u1", qty: 1, storageGrade: "AMBIENT" }))).toBe("CONTAINER_OCCUPIED");
    lab.receive(id(), { sampleId: "o", containerId: "E", custodianId: "u1", qty: 1, storageGrade: "AMBIENT" });
    expect(code(lab.startTransit(id(), { transitId: "t3", sampleId: "o", targetContainerId: "D" }))).toBe("CONTAINER_OCCUPIED");
    expect(code(lab.move(id(), "o", "D"))).toBe("CONTAINER_OCCUPIED");
    // 금지
    lab.reserve(id(), { reservationId: "r1", sampleId: "s1", qty: 5, expiresAt: null });
    expect(code(lab.split(id(), "s1", [{ sampleId: "c", containerId: "B", qty: 1 }]))).toBe("IN_TRANSIT");
    expect(code(lab.consume(id(), "r1"))).toBe("IN_TRANSIT");
    expect(code(lab.discard(id(), "s1"))).toBe("IN_TRANSIT");
    expect(code(lab.move(id(), "s1", "B"))).toBe("IN_TRANSIT");
    // 허용
    expect(lab.transferCustody(id(), "s1", "u2").ok).toBe(true);
    expect(lab.cancel(id(), "r1").ok).toBe(true);
    // 재전송
    expect(lab.confirmArrival("C1", "t1").ok).toBe(true);
    const events = lab.snapshot().events.length;
    expect(lab.confirmArrival("C1", "t1")).toEqual({ ok: true, value: { eventId: "C1", seq: events } });
    expect(code(lab.confirmArrival(id(), "t1"))).toBe("TRANSIT_CLOSED");
    expect(code(lab.cancelTransit(id(), "t1"))).toBe("TRANSIT_CLOSED");
    expect(code(lab.cancelTransit(id(), "nope"))).toBe("UNKNOWN_TRANSIT");
    const s = lab.snapshot();
    expect(s.samples.find((x) => x.sampleId === "s1")).toMatchObject({ containerId: "D", custodianId: "u2" });
    expect(s.transits[0]!.status).toBe("ARRIVED");
    // 출발 용기는 비고, 끝난 ID는 재사용 불가
    expect(lab.receive(id(), { sampleId: "p", containerId: "A", custodianId: "u1", qty: 1, storageGrade: "AMBIENT" }).ok).toBe(true);
    expect(code(lab.startTransit(id(), { transitId: "t1", sampleId: "s1", targetContainerId: "B" }))).toBe("ID_TAKEN");
  });
  it("취소는 목적 확보를 풀고 시료는 출발 용기에", () => {
    const lab = setup();
    lab.startTransit(id(), { transitId: "t1", sampleId: "s1", targetContainerId: "D" });
    expect(lab.cancelTransit(id(), "t1").ok).toBe(true);
    const s = lab.snapshot();
    expect(s.samples[0]!.containerId).toBe("A");
    expect(s.transits[0]!.status).toBe("CANCELED");
    expect(lab.receive(id(), { sampleId: "o", containerId: "D", custodianId: "u1", qty: 1, storageGrade: "AMBIENT" }).ok).toBe(true);
    expect(lab.discard(id(), "s1").ok).toBe(true);
  });
  it("실패한 시작은 아무것도 남기지 않는다", () => {
    const lab = setup();
    const before = lab.snapshot();
    expect(code(lab.startTransit(id(), { transitId: "t1", sampleId: "s1", targetContainerId: "C" }))).toBe("STORAGE_INCOMPATIBLE");
    expect(code(lab.startTransit(id(), { transitId: "t1", sampleId: "s1", targetContainerId: "S" }))).toBe("CAPACITY_EXCEEDED");
    expect(code(lab.startTransit(id(), { transitId: "t1", sampleId: "s1", targetContainerId: "A" }))).toBe("CONTAINER_OCCUPIED");
    expect(code(lab.startTransit(id(), { transitId: "t1", sampleId: "s1", targetContainerId: "ZZ" }))).toBe("UNKNOWN_CONTAINER");
    expect(lab.snapshot()).toEqual(before);
    expect(lab.startTransit(id(), { transitId: "t1", sampleId: "s1", targetContainerId: "D" }).ok).toBe(true);
  });
  it("격리 중에도 운송 가능, 도착은 격리·등급을 바꾸지 않음", () => {
    const lab = setup();
    lab.quarantine(id(), "s1");
    lab.startTransit(id(), { transitId: "t1", sampleId: "s1", targetContainerId: "D" });
    lab.confirmArrival(id(), "t1");
    expect(lab.snapshot().samples[0]).toMatchObject({ quarantined: true, storageGrade: "AMBIENT", containerId: "D" });
  });
});
