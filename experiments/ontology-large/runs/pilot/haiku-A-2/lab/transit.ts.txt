import { active } from "./state.js";
import { isQty } from "./samples.js";
import type { Engine } from "./events.js";
import type { Storage } from "./storage.js";
import type { ID, Lab } from "./types.js";

function isInTransit(sampleId: ID, transitId: ID | null = null): (engine: InstanceType<typeof Engine>) => boolean {
  return (_engine) => {
    const st = (_engine as any).st;
    for (const t of st.transits.values()) {
      if (t.sampleId === sampleId && t.status === "ACTIVE") {
        if (transitId === null || t.transitId !== transitId) return true;
      }
    }
    return false;
  };
}

export function transitCommands(engine: Engine, storage: Storage): Pick<Lab, "startTransit" | "confirmArrival" | "cancelTransit"> {
  return {
    startTransit(eventId, input) {
      return engine.run(eventId, "startTransit", input, (s) => {
        const x = active(s, input.sampleId);
        if (typeof x === "string") return x;
        if (s.transits.has(input.transitId)) return "ID_TAKEN";
        const c = storage.get(input.targetContainerId);
        if (!c) return "UNKNOWN_CONTAINER";
        if (!storage.supports(input.targetContainerId, x.storageGrade)) return "STORAGE_INCOMPATIBLE";
        if (!storage.isFree(s, input.targetContainerId)) return "CONTAINER_OCCUPIED";
        for (const t of s.transits.values()) {
          if (t.sampleId === input.sampleId && t.status === "ACTIVE") return "IN_TRANSIT";
        }
        s.transits.set(input.transitId, {
          transitId: input.transitId, sampleId: input.sampleId, sourceContainerId: x.containerId!, targetContainerId: input.targetContainerId, status: "ACTIVE",
        });
        s.occupied.set(input.targetContainerId, input.sampleId);
        return null;
      });
    },

    confirmArrival(eventId, transitId) {
      return engine.run(eventId, "confirmArrival", { transitId }, (s) => {
        const t = s.transits.get(transitId);
        if (!t) return "UNKNOWN_TRANSIT";
        if (t.status !== "ACTIVE") return "TRANSIT_CLOSED";
        const x = s.samples.get(t.sampleId);
        if (!x || x.status !== "ACTIVE") return "UNKNOWN_SAMPLE";
        s.occupied.delete(t.sourceContainerId);
        x.containerId = t.targetContainerId;
        t.status = "ARRIVED";
        return null;
      });
    },

    cancelTransit(eventId, transitId) {
      return engine.run(eventId, "cancelTransit", { transitId }, (s) => {
        const t = s.transits.get(transitId);
        if (!t) return "UNKNOWN_TRANSIT";
        if (t.status !== "ACTIVE") return "TRANSIT_CLOSED";
        s.occupied.delete(t.targetContainerId);
        t.status = "CANCELED";
        return null;
      });
    },
  };
}
