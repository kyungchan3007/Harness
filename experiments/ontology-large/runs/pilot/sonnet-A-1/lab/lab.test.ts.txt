import { describe, expect, it } from "vitest";
import { createLab } from "./lab.js";
import type { Catalog, Lab, Result, Receipt } from "./types.js";

const catalog: Catalog = {
  locationIds: ["L1", "L2"],
  custodianIds: ["alice", "bob"],
  containers: [
    { containerId: "A", locationId: "L1", capacity: 100, supportedGrades: ["AMBIENT", "CHILLED"] },
    { containerId: "B", locationId: "L1", capacity: 100, supportedGrades: ["AMBIENT"] },
    { containerId: "C", locationId: "L2", capacity: 100, supportedGrades: ["AMBIENT", "CHILLED"] },
    { containerId: "D", locationId: "L2", capacity: 100, supportedGrades: ["FROZEN"] },
    { containerId: "S", locationId: "L2", capacity: 5, supportedGrades: ["AMBIENT"] },
  ],
};

let n = 0;
const ev = () => `e${++n}`;
const code = (r: Result<Receipt>) => (r.ok ? "OK" : r.code);

function setup(grade: "AMBIENT" | "CHILLED" | "FROZEN" = "AMBIENT", container = "A", qty = 50): Lab {
  const lab = createLab(catalog);
  expect(lab.receive(ev(), { sampleId: "s1", containerId: container, custodianId: "alice", qty, storageGrade: grade }).ok).toBe(true);
  return lab;
}
const sample = (lab: Lab, id = "s1") => lab.snapshot().samples.find((x) => x.sampleId === id)!;

describe("격리", () => {
  it("멱등으로 설정하고 ACTIVE만 허용한다", () => {
    const lab = setup();
    expect(code(lab.quarantine(ev(), "s1"))).toBe("OK");
    expect(code(lab.quarantine(ev(), "s1"))).toBe("OK");
    expect(sample(lab).quarantined).toBe(true);
    expect(code(lab.release(ev(), "s1"))).toBe("OK");
    expect(code(lab.release(ev(), "s1"))).toBe("OK");
    expect(sample(lab).quarantined).toBe(false);
    expect(code(lab.quarantine(ev(), "zz"))).toBe("UNKNOWN_SAMPLE");
    lab.discard(ev(), "s1");
    expect(code(lab.quarantine(ev(), "s1"))).toBe("NOT_ACTIVE");
  });

  it("분취·새 예약·소비를 막되 취소·이동·폐기·책임자 변경·운송은 허용한다", () => {
    const lab = setup();
    lab.reserve(ev(), { reservationId: "r1", sampleId: "s1", qty: 10, expiresAt: 5 });
    expect(code(lab.quarantine(ev(), "s1"))).toBe("OK"); // 예약이 있어도 격리 가능
    expect(lab.snapshot().reservations[0]).toMatchObject({ status: "ACTIVE", qty: 10, expiresAt: 5 });
    expect(code(lab.split(ev(), "s1", [{ sampleId: "c1", containerId: "C", qty: 5 }]))).toBe("QUARANTINED");
    expect(code(lab.reserve(ev(), { reservationId: "r2", sampleId: "s1", qty: 1, expiresAt: null }))).toBe("QUARANTINED");
    expect(code(lab.consume(ev(), "r1"))).toBe("QUARANTINED");
    expect(code(lab.transferCustody(ev(), "s1", "bob"))).toBe("OK");
    expect(code(lab.move(ev(), "s1", "C"))).toBe("OK");
    expect(code(lab.startTransit(ev(), { transitId: "t1", sampleId: "s1", targetContainerId: "A" }))).toBe("OK");
    expect(code(lab.cancelTransit(ev(), "t1"))).toBe("OK");
    expect(code(lab.cancel(ev(), "r1"))).toBe("OK");
    expect(sample(lab)).toMatchObject({ quarantined: true, remaining: 50, reserved: 0 });
    expect(code(lab.discard(ev(), "s1"))).toBe("OK");
  });
});

describe("보관 등급", () => {
  it("받기·이동은 정확히 일치할 때만 허용한다", () => {
    const lab = createLab(catalog);
    expect(code(lab.receive(ev(), { sampleId: "x", containerId: "D", custodianId: "alice", qty: 1, storageGrade: "CHILLED" }))).toBe("STORAGE_INCOMPATIBLE");
    expect(lab.snapshot().samples).toHaveLength(0);
    const lab2 = setup("CHILLED");
    expect(code(lab2.move(ev(), "s1", "B"))).toBe("STORAGE_INCOMPATIBLE");
    expect(code(lab2.move(ev(), "s1", "C"))).toBe("OK");
  });

  it("분취 자식은 부모 등급을 상속하고 용기가 맞아야 한다", () => {
    const lab = setup("CHILLED");
    expect(code(lab.split(ev(), "s1", [{ sampleId: "c1", containerId: "B", qty: 5 }]))).toBe("STORAGE_INCOMPATIBLE");
    expect(lab.snapshot().samples).toHaveLength(1);
    expect(code(lab.split(ev(), "s1", [{ sampleId: "c1", containerId: "C", qty: 5 }]))).toBe("OK");
    expect(sample(lab, "c1").storageGrade).toBe("CHILLED");
  });

  it("운송 목적지도 검사하고 실패하면 아무것도 남기지 않는다", () => {
    const lab = setup("CHILLED");
    const before = lab.snapshot();
    expect(code(lab.startTransit(ev(), { transitId: "t1", sampleId: "s1", targetContainerId: "B" }))).toBe("STORAGE_INCOMPATIBLE");
    expect(lab.snapshot()).toEqual(before);
    expect(code(lab.receive(ev(), { sampleId: "s2", containerId: "B", custodianId: "bob", qty: 1, storageGrade: "AMBIENT" }))).toBe("OK");
  });
});

describe("예약 만료", () => {
  it("시각은 과거로 갈 수 없고 같은 시각은 허용한다", () => {
    const lab = setup();
    expect(code(lab.advanceTime(ev(), 5))).toBe("OK");
    expect(code(lab.advanceTime(ev(), 4))).toBe("INVALID_INPUT");
    expect(code(lab.advanceTime(ev(), 5))).toBe("OK");
    expect(code(lab.advanceTime(ev(), -1))).toBe("INVALID_INPUT");
    expect(code(lab.advanceTime(ev(), 1.5))).toBe("INVALID_INPUT");
    expect(lab.snapshot().now).toBe(5);
  });

  it("만료 시각은 null이거나 현재보다 미래여야 한다", () => {
    const lab = setup();
    lab.advanceTime(ev(), 5);
    for (const [i, e] of ([5, 4, -1, 6.5, NaN] as number[]).entries()) {
      expect(code(lab.reserve(ev(), { reservationId: `bad${i}`, sampleId: "s1", qty: 1, expiresAt: e }))).toBe("INVALID_INPUT");
    }
    expect(code(lab.reserve(ev(), { reservationId: "ok1", sampleId: "s1", qty: 1, expiresAt: 6 }))).toBe("OK");
    expect(code(lab.reserve(ev(), { reservationId: "ok2", sampleId: "s1", qty: 1, expiresAt: null }))).toBe("OK");
  });

  it("expiresAt <= now인 ACTIVE 예약만 만료하며 양은 풀기만 한다", () => {
    const lab = setup();
    lab.reserve(ev(), { reservationId: "r1", sampleId: "s1", qty: 10, expiresAt: 3 });
    lab.reserve(ev(), { reservationId: "r2", sampleId: "s1", qty: 10, expiresAt: 3 });
    lab.reserve(ev(), { reservationId: "r3", sampleId: "s1", qty: 10, expiresAt: 4 });
    lab.reserve(ev(), { reservationId: "r4", sampleId: "s1", qty: 10, expiresAt: null });
    lab.cancel(ev(), "r2");
    const seq = lab.snapshot().events.length;
    expect(code(lab.advanceTime(ev(), 3))).toBe("OK");
    const snap = lab.snapshot();
    expect(snap.events.length).toBe(seq + 1);
    const st = Object.fromEntries(snap.reservations.map((r) => [r.reservationId, r.status]));
    expect(st).toEqual({ r1: "EXPIRED", r2: "CANCELED", r3: "ACTIVE", r4: "ACTIVE" });
    expect(sample(lab)).toMatchObject({ remaining: 50, reserved: 20, consumedQty: 0, discardedQty: 0 });
    expect(code(lab.cancel(ev(), "r1"))).toBe("RESERVATION_CLOSED");
    expect(code(lab.consume(ev(), "r1"))).toBe("RESERVATION_CLOSED");
  });

  it("격리·운송 중에도 만료되고 확보는 그대로다", () => {
    const lab = setup();
    lab.reserve(ev(), { reservationId: "r1", sampleId: "s1", qty: 10, expiresAt: 2 });
    lab.quarantine(ev(), "s1");
    lab.startTransit(ev(), { transitId: "t1", sampleId: "s1", targetContainerId: "C" });
    lab.advanceTime(ev(), 2);
    expect(lab.snapshot().reservations[0]?.status).toBe("EXPIRED");
    expect(sample(lab)).toMatchObject({ quarantined: true, containerId: "A", reserved: 0 });
    expect(code(lab.receive(ev(), { sampleId: "s2", containerId: "C", custodianId: "bob", qty: 1, storageGrade: "AMBIENT" }))).toBe("CONTAINER_OCCUPIED");
  });
});

describe("운송", () => {
  it("도착하면 목적 용기로 옮기고 출발 용기를 비운다", () => {
    const lab = setup();
    expect(code(lab.startTransit(ev(), { transitId: "t1", sampleId: "s1", targetContainerId: "C" }))).toBe("OK");
    expect(sample(lab).containerId).toBe("A");
    expect(code(lab.confirmArrival(ev(), "t1"))).toBe("OK");
    expect(sample(lab).containerId).toBe("C");
    expect(lab.snapshot().transits[0]).toEqual({ transitId: "t1", sampleId: "s1", sourceContainerId: "A", targetContainerId: "C", status: "ARRIVED" });
    expect(code(lab.receive(ev(), { sampleId: "s2", containerId: "A", custodianId: "bob", qty: 1, storageGrade: "AMBIENT" }))).toBe("OK");
  });

  it("취소하면 목적 확보를 풀고 시료는 출발 용기에 남는다", () => {
    const lab = setup();
    lab.startTransit(ev(), { transitId: "t1", sampleId: "s1", targetContainerId: "C" });
    expect(code(lab.cancelTransit(ev(), "t1"))).toBe("OK");
    expect(sample(lab).containerId).toBe("A");
    expect(lab.snapshot().transits[0]?.status).toBe("CANCELED");
    expect(code(lab.receive(ev(), { sampleId: "s2", containerId: "C", custodianId: "bob", qty: 1, storageGrade: "AMBIENT" }))).toBe("OK");
  });

  it("목적 용기는 배타적으로 확보된다", () => {
    const lab = setup();
    lab.receive(ev(), { sampleId: "s2", containerId: "B", custodianId: "bob", qty: 1, storageGrade: "AMBIENT" });
    lab.startTransit(ev(), { transitId: "t1", sampleId: "s1", targetContainerId: "C" });
    expect(code(lab.startTransit(ev(), { transitId: "t2", sampleId: "s2", targetContainerId: "C" }))).toBe("CONTAINER_OCCUPIED");
    expect(code(lab.move(ev(), "s2", "C"))).toBe("CONTAINER_OCCUPIED");
    expect(code(lab.receive(ev(), { sampleId: "s3", containerId: "C", custodianId: "bob", qty: 1, storageGrade: "AMBIENT" }))).toBe("CONTAINER_OCCUPIED");
    expect(code(lab.startTransit(ev(), { transitId: "t3", sampleId: "s2", targetContainerId: "B" }))).toBe("CONTAINER_OCCUPIED");
  });

  it("운송 중에는 분취·소비·폐기·이동·추가 운송을 막고 나머지는 허용한다", () => {
    const lab = setup();
    lab.reserve(ev(), { reservationId: "r1", sampleId: "s1", qty: 10, expiresAt: null });
    lab.startTransit(ev(), { transitId: "t1", sampleId: "s1", targetContainerId: "C" });
    const before = lab.snapshot();
    expect(code(lab.split(ev(), "s1", [{ sampleId: "c1", containerId: "B", qty: 1 }]))).toBe("IN_TRANSIT");
    expect(code(lab.consume(ev(), "r1"))).toBe("IN_TRANSIT");
    expect(code(lab.discard(ev(), "s1"))).toBe("IN_TRANSIT");
    expect(code(lab.move(ev(), "s1", "B"))).toBe("IN_TRANSIT");
    expect(code(lab.startTransit(ev(), { transitId: "t2", sampleId: "s1", targetContainerId: "B" }))).toBe("IN_TRANSIT");
    expect(lab.snapshot()).toEqual(before);
    expect(code(lab.reserve(ev(), { reservationId: "r2", sampleId: "s1", qty: 1, expiresAt: null }))).toBe("OK");
    expect(code(lab.cancel(ev(), "r2"))).toBe("OK");
    expect(code(lab.transferCustody(ev(), "s1", "bob"))).toBe("OK");
    expect(code(lab.quarantine(ev(), "s1"))).toBe("OK");
    expect(code(lab.release(ev(), "s1"))).toBe("OK");
    expect(code(lab.advanceTime(ev(), 1))).toBe("OK");
  });

  it("끝난 운송의 식별자와 오류 코드", () => {
    const lab = setup();
    lab.startTransit(ev(), { transitId: "t1", sampleId: "s1", targetContainerId: "C" });
    expect(code(lab.startTransit(ev(), { transitId: "t1", sampleId: "s1", targetContainerId: "B" }))).toBe("ID_TAKEN");
    lab.confirmArrival(ev(), "t1");
    expect(code(lab.startTransit(ev(), { transitId: "t1", sampleId: "s1", targetContainerId: "A" }))).toBe("ID_TAKEN");
    expect(code(lab.confirmArrival(ev(), "t1"))).toBe("TRANSIT_CLOSED");
    expect(code(lab.cancelTransit(ev(), "t1"))).toBe("TRANSIT_CLOSED");
    expect(code(lab.confirmArrival(ev(), "nope"))).toBe("UNKNOWN_TRANSIT");
    expect(code(lab.cancelTransit(ev(), "nope"))).toBe("UNKNOWN_TRANSIT");
  });

  it("목적지 검사: 없는 용기·자기 용기·용량 부족·ACTIVE 아님", () => {
    const lab = setup();
    expect(code(lab.startTransit(ev(), { transitId: "t1", sampleId: "s1", targetContainerId: "X" }))).toBe("UNKNOWN_CONTAINER");
    expect(code(lab.startTransit(ev(), { transitId: "t1", sampleId: "s1", targetContainerId: "A" }))).toBe("CONTAINER_OCCUPIED");
    expect(code(lab.startTransit(ev(), { transitId: "t1", sampleId: "s1", targetContainerId: "S" }))).toBe("CAPACITY_EXCEEDED");
    expect(code(lab.startTransit(ev(), { transitId: "t1", sampleId: "zz", targetContainerId: "C" }))).toBe("UNKNOWN_SAMPLE");
    expect(lab.snapshot().transits).toHaveLength(0);
    lab.discard(ev(), "s1");
    expect(code(lab.startTransit(ev(), { transitId: "t1", sampleId: "s1", targetContainerId: "C" }))).toBe("NOT_ACTIVE");
  });
});

describe("재전송·이력", () => {
  it("성공한 새 명령의 재전송은 처음 Receipt를 돌려주고 상태를 바꾸지 않는다", () => {
    const lab = setup();
    const e1 = ev();
    const first = lab.startTransit(e1, { transitId: "t1", sampleId: "s1", targetContainerId: "C" });
    const e2 = ev();
    const arr = lab.confirmArrival(e2, "t1");
    const snap = lab.snapshot();
    expect(lab.startTransit(e1, { targetContainerId: "C", sampleId: "s1", transitId: "t1" })).toEqual(first);
    expect(lab.confirmArrival(e2, "t1")).toEqual(arr);
    expect(lab.snapshot()).toEqual(snap);
    expect(code(lab.cancelTransit(e2, "t1"))).toBe("EVENT_CONFLICT");
    expect(code(lab.confirmArrival(e2, "t2"))).toBe("EVENT_CONFLICT");
  });

  it("advanceTime·quarantine 재전송과 충돌, 실패한 eventId는 소모되지 않는다", () => {
    const lab = setup();
    const a = ev();
    const r = lab.advanceTime(a, 3);
    expect(lab.advanceTime(a, 3)).toEqual(r);
    expect(code(lab.advanceTime(a, 4))).toBe("EVENT_CONFLICT");
    const f = ev();
    expect(code(lab.advanceTime(f, 1))).toBe("INVALID_INPUT");
    expect(code(lab.advanceTime(f, 10))).toBe("OK");
    const q = ev();
    lab.quarantine(q, "s1");
    expect(code(lab.release(q, "s1"))).toBe("EVENT_CONFLICT");
  });

  it("이력에 새 명령의 op·args를 남기고 seq는 1씩 늘어난다", () => {
    const lab = setup();
    lab.quarantine("q", "s1");
    lab.advanceTime("a", 2);
    lab.startTransit("t", { transitId: "t1", sampleId: "s1", targetContainerId: "C" });
    lab.cancelTransit("c", "t1");
    const evs = lab.snapshot().events;
    expect(evs.map((e) => e.seq)).toEqual([1, 2, 3, 4, 5]);
    expect(evs.slice(1).map((e) => [e.op, e.args])).toEqual([
      ["quarantine", { sampleId: "s1" }],
      ["advanceTime", { to: 2 }],
      ["startTransit", { transitId: "t1", sampleId: "s1", targetContainerId: "C" }],
      ["cancelTransit", { transitId: "t1" }],
    ]);
  });
});
