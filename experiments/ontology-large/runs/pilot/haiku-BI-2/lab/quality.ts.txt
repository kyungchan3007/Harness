// 품질관리 — 격리·해제 (docs/quality.md)
import { active } from "./state.js";
import type { Engine } from "./events.js";
import type { ID, Lab } from "./types.js";

export function qualityCommands(engine: Engine): Pick<Lab, "quarantine" | "release"> {
  return {
    quarantine(eventId: ID, sampleId: ID) {
      return engine.run(eventId, "quarantine", { sampleId }, (s) => {
        const x = active(s, sampleId);
        if (typeof x === "string") return x;
        x.quarantined = true;
        return null;
      });
    },
    release(eventId: ID, sampleId: ID) {
      return engine.run(eventId, "release", { sampleId }, (s) => {
        const x = active(s, sampleId);
        if (typeof x === "string") return x;
        x.quarantined = false;
        return null;
      });
    },
  };
}
