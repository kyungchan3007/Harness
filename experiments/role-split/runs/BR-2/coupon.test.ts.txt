import { describe, expect, it } from "vitest";
import type { LineItem } from "../cart/cart.js";
import { Coupon, CouponError, priceWithCoupons } from "./coupon.js";

describe("Coupon", () => {
  const defaultItem: LineItem = {
    sku: "SKU001",
    name: "Test Item",
    unitPrice: 10_000,
    quantity: 1,
  };

  const defaultPolicy = { fee: 3_000, freeThreshold: 50_000 };

  describe("rate coupon basic calculations", () => {
    it("should apply rate discount correctly", () => {
      const items: readonly LineItem[] = [defaultItem];
      const coupons: readonly Coupon[] = [
        { scope: "order", type: "rate", value: 10 },
      ];

      const result = priceWithCoupons(items, coupons, defaultPolicy);
      expect(result.discount).toBe(1_000); // 10000 * 10 / 100 = 1000
      expect(result.subtotal).toBe(10_000);
      expect(result.total).toBe(10_000 - 1_000 + 3_000); // 12000
    });

    it("should floor rate discount (not round)", () => {
      const items: readonly LineItem[] = [
        {
          sku: "SKU002",
          name: "Test",
          unitPrice: 100,
          quantity: 1,
        },
      ];
      const coupons: readonly Coupon[] = [
        { scope: "order", type: "rate", value: 33 },
      ];

      const result = priceWithCoupons(items, coupons, defaultPolicy);
      expect(result.discount).toBe(33); // floor(100 * 33 / 100) = 33
    });
  });

  describe("fixed coupon calculations", () => {
    it("should apply fixed discount correctly", () => {
      const items: readonly LineItem[] = [defaultItem];
      const coupons: readonly Coupon[] = [
        { scope: "order", type: "fixed", value: 5_000 },
      ];

      const result = priceWithCoupons(items, coupons, defaultPolicy);
      expect(result.discount).toBe(5_000);
      expect(result.total).toBe(10_000 - 5_000 + 3_000); // 8000
    });

    it("should not exceed base amount in fixed discount", () => {
      const items: readonly LineItem[] = [
        {
          sku: "SKU003",
          name: "Small Item",
          unitPrice: 1_000,
          quantity: 1,
        },
      ];
      const coupons: readonly Coupon[] = [
        { scope: "order", type: "fixed", value: 5_000 },
      ];

      const result = priceWithCoupons(items, coupons, defaultPolicy);
      expect(result.discount).toBe(1_000); // min(5000, 1000) = 1000
    });
  });

  describe("maxDiscount constraint", () => {
    it("should limit discount with maxDiscount", () => {
      const items: readonly LineItem[] = [defaultItem];
      const coupons: readonly Coupon[] = [
        { scope: "order", type: "rate", value: 50, maxDiscount: 4_000 },
      ];

      const result = priceWithCoupons(items, coupons, defaultPolicy);
      expect(result.discount).toBe(4_000); // min(5000, 4000) = 4000
    });

    it("should apply maxDiscount to fixed coupons", () => {
      const items: readonly LineItem[] = [defaultItem];
      const coupons: readonly Coupon[] = [
        { scope: "order", type: "fixed", value: 10_000, maxDiscount: 6_000 },
      ];

      const result = priceWithCoupons(items, coupons, defaultPolicy);
      expect(result.discount).toBe(6_000);
    });
  });

  describe("minOrder condition", () => {
    it("should apply coupon when minOrder is met", () => {
      const items: readonly LineItem[] = [
        {
          sku: "SKU004",
          name: "Item",
          unitPrice: 10_000,
          quantity: 1,
        },
      ];
      const coupons: readonly Coupon[] = [
        { scope: "order", type: "fixed", value: 1_000, minOrder: 10_000 },
      ];

      const result = priceWithCoupons(items, coupons, defaultPolicy);
      expect(result.discount).toBe(1_000); // subtotal >= minOrder
    });

    it("should not apply coupon when minOrder is not met (returns 0, not error)", () => {
      const items: readonly LineItem[] = [
        {
          sku: "SKU005",
          name: "Item",
          unitPrice: 4_999,
          quantity: 1,
        },
      ];
      const coupons: readonly Coupon[] = [
        { scope: "order", type: "fixed", value: 1_000, minOrder: 5_000 },
      ];

      const result = priceWithCoupons(items, coupons, defaultPolicy);
      expect(result.discount).toBe(0); // subtotal < minOrder, so no discount
    });

    it("should apply coupon when minOrder equals subtotal", () => {
      const items: readonly LineItem[] = [
        {
          sku: "SKU006",
          name: "Item",
          unitPrice: 5_000,
          quantity: 1,
        },
      ];
      const coupons: readonly Coupon[] = [
        { scope: "order", type: "rate", value: 10, minOrder: 5_000 },
      ];

      const result = priceWithCoupons(items, coupons, defaultPolicy);
      expect(result.discount).toBe(500); // subtotal >= minOrder (equal case)
    });
  });

  describe("item coupon (SKU-specific)", () => {
    it("should apply item coupon to specific product", () => {
      const items: readonly LineItem[] = [
        {
          sku: "ITEM1",
          name: "First",
          unitPrice: 10_000,
          quantity: 1,
        },
        {
          sku: "ITEM2",
          name: "Second",
          unitPrice: 20_000,
          quantity: 1,
        },
      ];
      const coupons: readonly Coupon[] = [
        { scope: "item", sku: "ITEM1", type: "rate", value: 10 },
      ];

      const result = priceWithCoupons(items, coupons, defaultPolicy);
      expect(result.discount).toBe(1_000); // Only ITEM1: 10000 * 10 / 100
    });

    it("should apply multiple item coupons to different products", () => {
      const items: readonly LineItem[] = [
        {
          sku: "ITEM1",
          name: "First",
          unitPrice: 10_000,
          quantity: 1,
        },
        {
          sku: "ITEM2",
          name: "Second",
          unitPrice: 20_000,
          quantity: 1,
        },
      ];
      const coupons: readonly Coupon[] = [
        { scope: "item", sku: "ITEM1", type: "rate", value: 10 },
        { scope: "item", sku: "ITEM2", type: "fixed", value: 5_000 },
      ];

      const result = priceWithCoupons(items, coupons, defaultPolicy);
      expect(result.discount).toBe(6_000); // 1000 + 5000
    });
  });

  describe("item + order coupon combination", () => {
    it("should apply item coupon first, then order coupon on remaining", () => {
      const items: readonly LineItem[] = [
        {
          sku: "ITEM1",
          name: "Item",
          unitPrice: 10_000,
          quantity: 1,
        },
      ];
      const coupons: readonly Coupon[] = [
        { scope: "item", sku: "ITEM1", type: "rate", value: 20 }, // 2000 discount
        { scope: "order", type: "rate", value: 10 }, // 10% on (10000 - 2000) = 800
      ];

      const result = priceWithCoupons(items, coupons, defaultPolicy);
      expect(result.discount).toBe(2_800); // 2000 + 800
      expect(result.subtotal).toBe(10_000);
    });

    it("should apply order coupon minOrder based on original subtotal", () => {
      const items: readonly LineItem[] = [
        {
          sku: "ITEM1",
          name: "Item",
          unitPrice: 6_000,
          quantity: 1,
        },
      ];
      const coupons: readonly Coupon[] = [
        { scope: "item", sku: "ITEM1", type: "fixed", value: 2_000 },
        { scope: "order", type: "fixed", value: 1_000, minOrder: 5_000 }, // minOrder based on original subtotal
      ];

      const result = priceWithCoupons(items, coupons, defaultPolicy);
      expect(result.discount).toBe(3_000); // Both applied (6000 >= 5000)
    });
  });

  describe("constraint validation", () => {
    it("should throw error for duplicate item coupons on same SKU", () => {
      const items: readonly LineItem[] = [defaultItem];
      const coupons: readonly Coupon[] = [
        { scope: "item", sku: "SKU001", type: "rate", value: 10 },
        { scope: "item", sku: "SKU001", type: "rate", value: 20 },
      ];

      expect(() => priceWithCoupons(items, coupons, defaultPolicy)).toThrow(CouponError);
    });

    it("should throw error for more than one order coupon", () => {
      const items: readonly LineItem[] = [defaultItem];
      const coupons: readonly Coupon[] = [
        { scope: "order", type: "rate", value: 10 },
        { scope: "order", type: "fixed", value: 1_000 },
      ];

      expect(() => priceWithCoupons(items, coupons, defaultPolicy)).toThrow(CouponError);
    });

    it("should throw error for item coupon on non-existent SKU", () => {
      const items: readonly LineItem[] = [defaultItem];
      const coupons: readonly Coupon[] = [
        { scope: "item", sku: "NONEXISTENT", type: "rate", value: 10 },
      ];

      expect(() => priceWithCoupons(items, coupons, defaultPolicy)).toThrow(CouponError);
    });
  });

  describe("validation errors", () => {
    it("should throw error for item coupon without SKU", () => {
      const items: readonly LineItem[] = [defaultItem];
      const coupons: readonly Coupon[] = [
        { scope: "item", type: "rate", value: 10 },
      ];

      expect(() => priceWithCoupons(items, coupons, defaultPolicy)).toThrow(CouponError);
    });

    it("should throw error for invalid rate value (not 1-100)", () => {
      const items: readonly LineItem[] = [defaultItem];
      const coupons: readonly Coupon[] = [
        { scope: "order", type: "rate", value: 0 },
      ];

      expect(() => priceWithCoupons(items, coupons, defaultPolicy)).toThrow(CouponError);
    });

    it("should throw error for invalid rate value (> 100)", () => {
      const items: readonly LineItem[] = [defaultItem];
      const coupons: readonly Coupon[] = [
        { scope: "order", type: "rate", value: 101 },
      ];

      expect(() => priceWithCoupons(items, coupons, defaultPolicy)).toThrow(CouponError);
    });

    it("should throw error for invalid fixed value (< 1)", () => {
      const items: readonly LineItem[] = [defaultItem];
      const coupons: readonly Coupon[] = [
        { scope: "order", type: "fixed", value: 0 },
      ];

      expect(() => priceWithCoupons(items, coupons, defaultPolicy)).toThrow(CouponError);
    });

    it("should throw error for non-integer rate value", () => {
      const items: readonly LineItem[] = [defaultItem];
      const coupons: readonly Coupon[] = [
        { scope: "order", type: "rate", value: 10.5 },
      ];

      expect(() => priceWithCoupons(items, coupons, defaultPolicy)).toThrow(CouponError);
    });

    it("should throw error for non-integer fixed value", () => {
      const items: readonly LineItem[] = [defaultItem];
      const coupons: readonly Coupon[] = [
        { scope: "order", type: "fixed", value: 1_000.5 },
      ];

      expect(() => priceWithCoupons(items, coupons, defaultPolicy)).toThrow(CouponError);
    });

    it("should throw error for invalid maxDiscount (negative)", () => {
      const items: readonly LineItem[] = [defaultItem];
      const coupons: readonly Coupon[] = [
        { scope: "order", type: "rate", value: 10, maxDiscount: -1 },
      ];

      expect(() => priceWithCoupons(items, coupons, defaultPolicy)).toThrow(CouponError);
    });

    it("should throw error for invalid minOrder (negative)", () => {
      const items: readonly LineItem[] = [defaultItem];
      const coupons: readonly Coupon[] = [
        { scope: "order", type: "rate", value: 10, minOrder: -1 },
      ];

      expect(() => priceWithCoupons(items, coupons, defaultPolicy)).toThrow(CouponError);
    });
  });

  describe("empty cart", () => {
    it("should return all zeros for empty cart regardless of coupons", () => {
      const items: readonly LineItem[] = [];
      const coupons: readonly Coupon[] = [
        { scope: "order", type: "rate", value: 10 },
      ];

      const result = priceWithCoupons(items, coupons, defaultPolicy);
      expect(result).toEqual({ subtotal: 0, discount: 0, shipping: 0, total: 0 });
    });
  });

  describe("shipping integration", () => {
    it("should charge shipping when discounted subtotal < threshold", () => {
      const items: readonly LineItem[] = [defaultItem];
      const coupons: readonly Coupon[] = [
        { scope: "order", type: "rate", value: 50 }, // 5000 discount
      ];

      const result = priceWithCoupons(items, coupons, defaultPolicy);
      expect(result.discount).toBe(5_000);
      expect(result.shipping).toBe(3_000); // (10000 - 5000) < 50000
      expect(result.total).toBe(5_000 + 3_000); // 8000
    });

    it("should not charge shipping when discounted subtotal >= threshold", () => {
      const items: readonly LineItem[] = [
        {
          sku: "EXPENSIVE",
          name: "Expensive",
          unitPrice: 60_000,
          quantity: 1,
        },
      ];
      const coupons: readonly Coupon[] = [
        { scope: "order", type: "rate", value: 10 }, // 6000 discount
      ];

      const result = priceWithCoupons(items, coupons, defaultPolicy);
      expect(result.discount).toBe(6_000);
      expect(result.shipping).toBe(0); // (60000 - 6000) >= 50000
      expect(result.total).toBe(54_000);
    });
  });

  describe("quantity in line item", () => {
    it("should calculate item coupon based on quantity", () => {
      const items: readonly LineItem[] = [
        {
          sku: "MULTI",
          name: "Multi",
          unitPrice: 1_000,
          quantity: 10,
        },
      ];
      const coupons: readonly Coupon[] = [
        { scope: "item", sku: "MULTI", type: "rate", value: 50 }, // 50% of (1000 * 10) = 5000
      ];

      const result = priceWithCoupons(items, coupons, defaultPolicy);
      expect(result.discount).toBe(5_000); // floor(10000 * 50 / 100)
      expect(result.subtotal).toBe(10_000);
    });
  });

  describe("immutability", () => {
    it("should not modify input arrays", () => {
      const items: LineItem[] = [{ ...defaultItem }];
      const coupons: Coupon[] = [{ scope: "order", type: "rate", value: 10 }];

      const originalItems = JSON.stringify(items);
      const originalCoupons = JSON.stringify(coupons);

      priceWithCoupons(items, coupons, defaultPolicy);

      expect(JSON.stringify(items)).toBe(originalItems);
      expect(JSON.stringify(coupons)).toBe(originalCoupons);
    });
  });
});
