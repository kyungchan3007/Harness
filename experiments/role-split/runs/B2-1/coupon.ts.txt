import type { LineItem } from "../cart/cart.js";
import { assertWon, type Won } from "../money.js";
import { DEFAULT_SHIPPING_POLICY, priceCart, type PriceBreakdown, type ShippingPolicy } from "./price-cart.js";

export interface Coupon {
  /** item: 한 상품 줄에 적용 / order: 주문 전체에 적용 */
  scope: "item" | "order";
  /** scope가 "item"일 때 대상 상품 (필수) */
  sku?: string;
  /** rate: 정률(%) / fixed: 정액(원) */
  type: "rate" | "fixed";
  /** rate면 할인율(%), fixed면 할인 금액(원) */
  value: number;
  /** 할인 금액의 상한 (원) */
  maxDiscount?: Won;
  /** 사용 조건: 최소 주문 금액 (원) */
  minOrder?: Won;
}

export class CouponError extends Error {}

/** 쿠폰 유효성 검증 */
function validateCoupon(coupon: Coupon): void {
  // 정률 쿠폰: value는 1~100의 정수
  if (coupon.type === "rate") {
    if (!Number.isInteger(coupon.value) || coupon.value < 1 || coupon.value > 100) {
      throw new CouponError(
        `정률 쿠폰의 value는 1~100의 정수여야 합니다: ${coupon.value}`
      );
    }
  }
  // 정액 쿠폰: value는 1 이상의 정수
  else if (coupon.type === "fixed") {
    if (!Number.isInteger(coupon.value) || coupon.value < 1) {
      throw new CouponError(
        `정액 쿠폰의 value는 1 이상의 정수여야 합니다: ${coupon.value}`
      );
    }
  }

  // maxDiscount는 0 이상의 정수 (있으면)
  if (coupon.maxDiscount !== undefined) {
    if (!Number.isSafeInteger(coupon.maxDiscount) || coupon.maxDiscount < 0) {
      throw new CouponError(
        `maxDiscount는 0 이상의 정수여야 합니다: ${coupon.maxDiscount}`
      );
    }
  }

  // minOrder는 0 이상의 정수 (있으면)
  if (coupon.minOrder !== undefined) {
    if (!Number.isSafeInteger(coupon.minOrder) || coupon.minOrder < 0) {
      throw new CouponError(
        `minOrder는 0 이상의 정수여야 합니다: ${coupon.minOrder}`
      );
    }
  }

  // 상품 쿠폰: sku는 필수
  if (coupon.scope === "item" && !coupon.sku) {
    throw new CouponError(`상품 쿠폰(scope="item")에는 sku가 필수입니다`);
  }
}

/** 할인액 계산 */
function calculateDiscount(coupon: Coupon, baseAmount: Won): Won {
  let discount: Won;

  if (coupon.type === "rate") {
    // 정률: 기준금액 × value / 100 (원 미만 버림)
    discount = Math.floor((baseAmount * coupon.value) / 100);
  } else {
    // 정액: min(value, 기준금액)
    discount = Math.min(coupon.value, baseAmount);
  }

  // maxDiscount 적용
  if (coupon.maxDiscount !== undefined) {
    discount = Math.min(discount, coupon.maxDiscount);
  }

  return discount;
}

export function priceWithCoupons(
  items: readonly LineItem[],
  coupons: readonly Coupon[],
  shippingPolicy: ShippingPolicy = DEFAULT_SHIPPING_POLICY,
): PriceBreakdown {
  // 빈 장바구니 처리
  if (items.length === 0) {
    return { subtotal: 0, discount: 0, shipping: 0, total: 0 };
  }

  // 쿠폰 유효성 검증
  for (const coupon of coupons) {
    validateCoupon(coupon);
  }

  // 상품 합계 계산
  const subtotal = items.reduce((sum, item) => sum + item.unitPrice * item.quantity, 0);
  assertWon(subtotal, "subtotal");

  // 상품쿠폰 적용 전 중복 검증
  const itemCouponsMap = new Map<string, Coupon>();
  let orderCouponCount = 0;

  for (const coupon of coupons) {
    if (coupon.scope === "item") {
      const sku = coupon.sku!;
      // 장바구니에 없는 상품의 쿠폰
      if (!items.some((item) => item.sku === sku)) {
        throw new CouponError(`장바구니에 없는 상품의 쿠폰입니다: ${sku}`);
      }
      // 상품당 1장 초과 검증
      if (itemCouponsMap.has(sku)) {
        throw new CouponError(
          `상품 ${sku}에 여러 쿠폰을 적용할 수 없습니다`
        );
      }
      itemCouponsMap.set(sku, coupon);
    } else if (coupon.scope === "order") {
      orderCouponCount++;
      // 주문 쿠폰은 1장만
      if (orderCouponCount > 1) {
        throw new CouponError(`주문 쿠폰은 1장만 사용할 수 있습니다`);
      }
    }
  }

  // 상품 쿠폰 적용
  let itemDiscount: Won = 0;
  for (const item of items) {
    const coupon = itemCouponsMap.get(item.sku);
    if (!coupon) continue;

    // minOrder 체크 (조건 미충족 시 0원 할인, 에러 아님)
    if (coupon.minOrder !== undefined && subtotal < coupon.minOrder) {
      continue;
    }

    // 상품 쿠폰의 기준 금액 = 줄 금액
    const lineAmount = item.unitPrice * item.quantity;
    const discount = calculateDiscount(coupon, lineAmount);
    itemDiscount += discount;
  }

  // 주문 쿠폰 적용
  let orderDiscount: Won = 0;
  const orderCoupon = coupons.find((c) => c.scope === "order");
  if (orderCoupon) {
    // minOrder 체크
    if (orderCoupon.minOrder === undefined || subtotal >= orderCoupon.minOrder) {
      // 주문 쿠폰의 기준 금액 = subtotal - 상품쿠폰할인합
      const baseAmount = subtotal - itemDiscount;
      orderDiscount = calculateDiscount(orderCoupon, baseAmount);
    }
  }

  const totalDiscount = itemDiscount + orderDiscount;
  assertWon(totalDiscount, "discount");

  // priceCart로 배송비 계산
  return priceCart(items, { discount: totalDiscount, shippingPolicy });
}
