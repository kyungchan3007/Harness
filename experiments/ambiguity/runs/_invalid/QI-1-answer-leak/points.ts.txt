import type { PriceBreakdown } from "./price-cart.js";
import type { Won } from "../money.js";

export interface PointsInput {
  priceBreakdown: PriceBreakdown;
  member: { grade: "VIP" | "NORMAL" };
}

/**
 * 포인트를 계산한다. domain.md 포인트 규칙 1~5를 따른다.
 *
 * 규칙:
 * 1. 적립 기준 금액: subtotal - discount (배송비 제외)
 * 2. 기본 적립률: 1%
 * 3. 쿠폰 사용 시(discount > 0): 적립률 절반 → 0.5%
 * 4. VIP 회원: 적립률 2배 → 2%
 * 5. VIP + 쿠폰: 2%를 먼저 적용 후 절반 → 1%
 *
 * 최종적으로 소수점은 버린다(내림).
 */
export function calculatePoints(input: PointsInput): Won {
  const { priceBreakdown, member } = input;
  const { subtotal, discount } = priceBreakdown;

  // 적립 기준 금액 (배송비 제외)
  const basisAmount = subtotal - discount;

  // 기본 적립률 1%
  let rate = 1;

  // VIP 회원: 적립률 2배
  if (member.grade === "VIP") {
    rate = 2;
  }

  // 쿠폰 사용: 적립률 절반
  if (discount > 0) {
    rate = rate / 2;
  }

  // 포인트 계산 후 내림
  const points = Math.floor((basisAmount * rate) / 100);

  return points;
}
