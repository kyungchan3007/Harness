import { calcPoints } from "./points.js";
import type { PriceBreakdown } from "./price-cart.js";

export function earn(breakdown: PriceBreakdown, member: { grade: "VIP" | "NORMAL" }): number {
  return calcPoints({
    total: breakdown.total,
    tier: member.grade === "VIP" ? "vip" : "regular",
    couponUsed: breakdown.discount > 0,
  });
}
