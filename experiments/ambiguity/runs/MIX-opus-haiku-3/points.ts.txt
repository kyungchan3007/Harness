import type { PriceBreakdown } from "./price-cart.js";
import { assertWon } from "../money.js";

export type MemberGrade = "VIP" | "NORMAL";

export interface Member {
  grade: MemberGrade;
}

/** 적립 포인트. 항상 0 이상의 정수 */
export type Points = number;

export function calculateEarnedPoints(
  breakdown: PriceBreakdown,
  member: Member
): Points {
  // 규칙 8: 검증
  assertWon(breakdown.subtotal, "subtotal");
  assertWon(breakdown.discount, "discount");

  if (breakdown.discount > breakdown.subtotal) {
    throw new RangeError(
      `할인(${breakdown.discount})이 상품 합계(${breakdown.subtotal})보다 클 수 없습니다`
    );
  }

  if (member.grade !== "VIP" && member.grade !== "NORMAL") {
    throw new RangeError(
      `알 수 없는 회원 등급: ${member.grade}`
    );
  }

  // 규칙 1: 기준 금액
  const base = breakdown.subtotal - breakdown.discount;

  // 규칙 2~5: 적립률 (천분율 정수)
  let rate = 10; // 기본 1% = 10‰
  if (member.grade === "VIP") {
    rate *= 2; // VIP: 2% = 20‰
  }
  if (breakdown.discount > 0) {
    rate /= 2; // 쿠폰 주문: 절반
  }

  // 규칙 6·7: 내림 (안전한 정수 계산)
  const points = Math.floor(base / 1000) * rate +
    Math.floor((base % 1000) * rate / 1000);

  return points;
}
