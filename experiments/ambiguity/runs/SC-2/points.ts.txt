import { assertWon, type Won } from "../money.js";

export interface PointsCalculationInput {
  /** 결제 금액 (원) */
  totalAmount: Won;
  /** VIP 회원 여부 */
  isVip: boolean;
  /** 쿠폰 사용 여부 */
  usedCoupon: boolean;
}

/**
 * 주문 결제 금액에 따라 포인트를 계산한다.
 * - 기본 적립률: 1% (결제 금액의 1%)
 * - 쿠폰 사용: 적립률 절반 (1% → 0.5%)
 * - VIP 회원: 적립률 2배 (1% → 2%)
 * - 결합 예: VIP + 쿠폰 = 1% × 0.5 × 2 = 1%
 */
export function calculatePoints(input: PointsCalculationInput): number {
  assertWon(input.totalAmount, "totalAmount");

  let rate = 1;

  if (input.usedCoupon) {
    rate *= 0.5;
  }

  if (input.isVip) {
    rate *= 2;
  }

  return Math.floor((input.totalAmount * rate) / 100);
}
