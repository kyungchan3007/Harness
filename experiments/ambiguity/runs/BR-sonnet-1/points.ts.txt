import { assertWon } from "../money.js";
import type { PriceBreakdown } from "./price-cart.js";

/** 포인트. 0 이상의 정수 (domain.md 포인트 규칙 1) */
export type Points = number;

/** 회원 등급. 판정은 엔진 밖 (원문에 등급 이름 없음 — 설계 결정) */
export type MemberTier = "regular" | "vip";

/** 적립률: 만분율 정수. 1% = 100 */
export type RateBp = number;

export interface TierRates {
  /** 쿠폰을 쓰지 않은 주문 */
  standard: RateBp;
  /** 쿠폰 주문 (discount > 0) */
  withCoupon: RateBp;
}

export interface PointsPolicy {
  rates: Record<MemberTier, TierRates>;
}

/** 값은 domain.md "포인트 규칙 (0030)" 표와 같아야 한다. regular.standard만 원문 확정, 나머지는 가정 */
export const DEFAULT_POINTS_POLICY: PointsPolicy = {
  rates: {
    regular: { standard: 100, withCoupon: 50 },
    vip: { standard: 200, withCoupon: 100 },
  },
};

export interface PointsOptions {
  /** 기본 "regular" */
  tier?: MemberTier;
  /** 기본 DEFAULT_POINTS_POLICY */
  policy?: PointsPolicy;
}

/** 만분율의 분모. 100% = 10,000bp */
const BP_DENOMINATOR = 10_000;

function assertRate(rate: RateBp, label: string): void {
  if (!Number.isInteger(rate) || rate < 0 || rate > BP_DENOMINATOR) {
    throw new RangeError(`${label} 적립률은 0 이상 ${BP_DENOMINATOR} 이하의 정수(bp)여야 합니다: ${rate}`);
  }
}

/** 정책이 domain.md 규칙 6·8을 지키는지 검사. 위반 시 RangeError */
export function assertPointsPolicy(policy: PointsPolicy): void {
  const { regular, vip } = policy.rates;

  assertRate(regular.standard, "regular.standard");
  assertRate(regular.withCoupon, "regular.withCoupon");
  assertRate(vip.standard, "vip.standard");
  assertRate(vip.withCoupon, "vip.withCoupon");

  if (regular.withCoupon >= regular.standard) {
    throw new RangeError(
      `regular: 쿠폰 주문 적립률(${regular.withCoupon})은 쿠폰 없는 주문(${regular.standard})보다 낮아야 합니다`,
    );
  }
  if (vip.withCoupon >= vip.standard) {
    throw new RangeError(`vip: 쿠폰 주문 적립률(${vip.withCoupon})은 쿠폰 없는 주문(${vip.standard})보다 낮아야 합니다`);
  }
  if (vip.standard <= regular.standard) {
    throw new RangeError(`쿠폰 없는 주문에서 vip 적립률(${vip.standard})은 regular(${regular.standard})보다 높아야 합니다`);
  }
  if (vip.withCoupon <= regular.withCoupon) {
    throw new RangeError(`쿠폰 주문에서 vip 적립률(${vip.withCoupon})은 regular(${regular.withCoupon})보다 높아야 합니다`);
  }
}

/**
 * 결제 금액(total)에 등급·쿠폰 여부별 적립률을 곱해 내림한 포인트를 돌려준다 (domain.md 포인트 규칙 0030).
 * 쿠폰 주문은 discount > 0.
 */
export function calculatePoints(price: Pick<PriceBreakdown, "total" | "discount">, options: PointsOptions = {}): Points {
  const { tier = "regular", policy = DEFAULT_POINTS_POLICY } = options;

  assertWon(price.total, "total");
  assertWon(price.discount, "discount");
  if (!Object.hasOwn(policy.rates, tier)) {
    throw new RangeError(`알 수 없는 회원 등급입니다: ${String(tier)}`);
  }
  assertPointsPolicy(policy);

  const tierRates = policy.rates[tier];
  const rate = price.discount > 0 ? tierRates.withCoupon : tierRates.standard;

  // floor(total * rate / D)를 total * rate가 안전 정수를 넘지 않게 몫·나머지로 나눠 계산한다.
  const quotient = Math.floor(price.total / BP_DENOMINATOR);
  const remainder = price.total % BP_DENOMINATOR;
  return quotient * rate + Math.floor((remainder * rate) / BP_DENOMINATOR);
}
