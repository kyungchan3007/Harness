/**
 * 포인트 적립 계산 (domain.md 포인트 규칙 0030)
 *
 * 공식:
 * 1. 효과적 금액 = total - (discount || 0)
 * 2. 기본 포인트 = round(효과적 금액 × 0.01)
 * 3. VIP인 경우 = round(기본 포인트 × 1.5)
 * 4. 음수 결과는 0으로 조정
 */

/**
 * 주문 금액에 따라 포인트를 계산한다
 *
 * @param total 결제 금액 (원)
 * @param discount 할인액 (선택사항, 기본값 0)
 * @param isVip VIP 회원 여부 (선택사항, 기본값 false)
 * @returns 적립 포인트 (정수, 0 이상)
 */
export function calculatePoints(total: number, discount?: number, isVip?: boolean): number {
  // 할인액이 제공되지 않으면 0으로 처리
  const discountAmount = discount ?? 0;

  // 효과적인 금액 계산
  const effectiveAmount = total - discountAmount;

  // 효과적인 금액이 음수이면 0 반환
  if (effectiveAmount < 0) {
    return 0;
  }

  // 기본 포인트 계산 (1%)
  const basePoints = Math.round(effectiveAmount * 0.01);

  // VIP 배율 적용
  const points = isVip ? Math.round(basePoints * 1.5) : basePoints;

  // 최종 포인트가 음수가 되지 않도록 보정 (이론적으로는 발생하지 않음)
  return Math.max(0, points);
}
