import type { PriceBreakdown } from "./price-cart.js";
import { calculatePoints } from "./points.js";

export function earn(breakdown: PriceBreakdown, member: { grade: "VIP" | "NORMAL" }): number {
  return calculatePoints(breakdown.total, {
    hasCoupon: breakdown.discount > 0,
    isVip: member.grade === "VIP",
  });
}
