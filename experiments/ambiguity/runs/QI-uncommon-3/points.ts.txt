import { assertWon, type Won } from "../money.js";
import type { PriceBreakdown } from "./price-cart.js";

export interface Member {
  grade: "VIP" | "NORMAL";
}

export function calculatePoints(priceBreakdown: PriceBreakdown, member: Member): Won {
  assertWon(priceBreakdown.total, "priceBreakdown.total");

  if (member.grade !== "VIP" && member.grade !== "NORMAL") {
    throw new Error(`회원 등급이 유효하지 않습니다: ${member.grade}`);
  }

  // NORMAL 회원이 쿠폰을 사용한 경우 포인트 적립 안 함
  if (member.grade === "NORMAL" && priceBreakdown.discount > 0) {
    return 0;
  }

  // 적립률 결정
  const rate = member.grade === "VIP" ? 0.03 : 0.01;

  // 포인트 계산 및 반올림
  const points = Math.round(priceBreakdown.total * rate);

  assertWon(points, "points");
  return points;
}
